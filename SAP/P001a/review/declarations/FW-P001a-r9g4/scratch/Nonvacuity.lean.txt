-- No audited theorem is used to construct these probability experiments.
theorem finiteLaw_probability {n : ℕ} (P : FiniteLaw n) :
    IsProbabilityMeasure (finiteLawMeasure P) := by
  constructor
  simp only [finiteLawMeasure, Measure.finset_sum_apply, Measure.smul_apply,
    Measure.dirac_apply_of_mem (Set.mem_univ _), smul_eq_mul, mul_one]
  rw [← ENNReal.ofReal_sum_of_nonneg (fun i _ => P.nonneg i), P.total]
  norm_num

theorem iid_coordinates {E : Type*} [MeasurableSpace E] (ν : Measure E)
    [IsProbabilityMeasure ν] :
    IID (Measure.infinitePi (fun _ : ℕ => ν)) (fun n ω => ω n) := by
  refine ⟨fun i => measurable_pi_apply i, ?_, ?_⟩
  · exact iIndepFun_infinitePi (X := fun _ => id) (fun _ => measurable_id)
  · intro i
    refine ⟨(measurable_pi_apply i).aemeasurable,
      (measurable_pi_apply 0).aemeasurable, ?_⟩
    simp only [Measure.infinitePi_map_eval]

-- T17: all hypotheses of the universal iid experiment are realized.
example : ∃ μ : Measure (ℕ → Fin 12), IsProbabilityMeasure μ ∧
    IID μ (fun n ω => ω n) ∧
    μ.map (fun ω => ω 0) = finiteLawMeasure placeboLaw := by
  letI := finiteLaw_probability placeboLaw
  refine ⟨Measure.infinitePi (fun _ : ℕ => finiteLawMeasure placeboLaw),
    inferInstance, iid_coordinates _, ?_⟩
  exact Measure.infinitePi_map_eval _ _

-- T19: check genuine interior cells, rather than a zero denominator.
example : overlapT 0 = 1/3 ∧ overlapC 0 = 1/4 ∧ overlapPropensity 0 = 4/7 := by
  norm_num [overlapT, overlapC, overlapPropensity]
example (k : ℕ) : 0 < overlapT k ∧ 0 < overlapC k ∧
    0 < overlapPropensity k ∧ overlapPropensity k < 1 := by
  have ht : 0 < overlapT k := pow_pos (by norm_num) _
  have hc : 0 < overlapC k := pow_pos (by norm_num) _
  exact ⟨ht, hc, div_pos ht (add_pos ht hc),
    (div_lt_one (add_pos ht hc)).2 (lt_add_of_pos_right _ hc)⟩
