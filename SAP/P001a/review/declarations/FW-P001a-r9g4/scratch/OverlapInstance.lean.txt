-- Independent verification of the complete countable witness, not an application of T19.
example : T19Statement := by
  have h3 : ‖(1/3 : ℝ)‖ < 1 := by norm_num
  have h2 : ‖(1/2 : ℝ)‖ < 1 := by norm_num
  have ht : Summable overlapT := by
    unfold overlapT
    simpa only [pow_succ] using
      (summable_geometric_of_norm_lt_one h3).mul_right (1/3 : ℝ)
  have hc : Summable overlapC := by
    unfold overlapC
    simpa only [pow_add] using
      (summable_geometric_of_norm_lt_one h2).mul_right ((1/2 : ℝ)^2)
  have hts : (∑' k, overlapT k) = 1/2 := by
    unfold overlapT
    simp_rw [pow_succ]
    rw [tsum_mul_right, tsum_geometric_of_norm_lt_one h3]
    norm_num
  have hcs : (∑' k, overlapC k) = 1/2 := by
    unfold overlapC
    simp_rw [pow_add]
    rw [tsum_mul_right, tsum_geometric_of_norm_lt_one h2]
    norm_num
  have hp (k : ℕ) : 0 < overlapT k ∧ 0 < overlapC k ∧
      0 < overlapPropensity k ∧ overlapPropensity k < 1 := by
    have htk : 0 < overlapT k := pow_pos (by norm_num) _
    have hck : 0 < overlapC k := pow_pos (by norm_num) _
    exact ⟨htk, hck, div_pos htk (add_pos htk hck),
      (div_lt_one (add_pos htk hck)).2 (lt_add_of_pos_right _ hck)⟩
  have hformula (k : ℕ) : overlapPropensity k =
      ((4/3 : ℝ)*(2/3 : ℝ)^k)/(1+(4/3 : ℝ)*(2/3 : ℝ)^k) := by
    have h1 : (1/2 : ℝ)^k ≠ 0 := pow_ne_zero _ (by norm_num)
    have hden : 1+(4/3 : ℝ)*(2/3 : ℝ)^k ≠ 0 := by positivity
    have hden' : overlapT k + overlapC k ≠ 0 := ne_of_gt (add_pos (hp k).1 (hp k).2.1)
    apply (div_eq_div_iff hden' hden).2
    unfold overlapT overlapC
    have hpow : (1/2 : ℝ)^k * (2/3 : ℝ)^k = (1/3 : ℝ)^k := by
      rw [← mul_pow]
      norm_num
    simp only [pow_succ]
    nlinarith [hpow]
  have hlim : Tendsto overlapPropensity atTop (nhds 0) := by
    have hr := (tendsto_const_nhds (x := (4/3 : ℝ))).mul
      (tendsto_pow_atTop_nhds_zero_of_lt_one (r := (2/3 : ℝ)) (by norm_num) (by norm_num))
    have hd := (tendsto_const_nhds (x := (1 : ℝ))).add hr
    have h := hr.div hd (by norm_num)
    change Tendsto (fun k => (4/3 : ℝ)*(2/3 : ℝ)^k /
      (1+(4/3 : ℝ)*(2/3 : ℝ)^k)) atTop
      (nhds ((4/3 : ℝ)*0/(1+(4/3 : ℝ)*0))) at h
    change Tendsto (fun k => overlapPropensity k) atTop (nhds 0)
    simp_rw [hformula]
    simpa only [Pi.div_apply, mul_zero, add_zero, zero_div] using h
  refine ⟨ht, hc, hts, hcs, hp, ?_⟩
  rintro ⟨ε, hε, hb⟩
  obtain ⟨k, hk⟩ := (hlim.eventually_lt_const hε).exists
  exact (not_lt_of_ge (le_of_lt (hb k).1)) hk
