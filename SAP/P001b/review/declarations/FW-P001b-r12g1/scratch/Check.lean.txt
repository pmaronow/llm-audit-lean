-- Checks below use definitions and Mathlib, never the audited theorem stubs.
lemma fair_halves : (2⁻¹ : ℝ≥0∞) + 2⁻¹ = 1 := by
  simpa using (ENNReal.add_halves 1)

local instance fair_probability : IsProbabilityMeasure W1Fair := by
  constructor
  norm_num [W1Fair, Measure.add_apply, Measure.smul_apply, fair_halves]

lemma fair_real (H : Set Bool) :
    probability W1Fair H =
      (if false ∈ H then (1 / 2 : ℝ) else 0) +
      (if true ∈ H then (1 / 2 : ℝ) else 0) := by
  by_cases h0 : false ∈ H <;> by_cases h1 : true ∈ H <;>
    norm_num [probability, W1Fair, Measure.add_apply, Measure.smul_apply,
      Measure.dirac_apply, h0, h1, fair_halves]

lemma fair_arm (d : ℝ) (hd : d = 0 ∨ d = 1) :
    probability W1Fair {b | W1BinaryValue b = d} = 1 / 2 := by
  rcases hd with rfl | rfl <;> norm_num [fair_real, W1BinaryValue]

lemma fair_propensity (x : Unit) :
    W1CellPropensity W1Fair (fun _ => ()) W1BinaryValue x = 1 / 2 := by
  cases x
  norm_num [W1CellPropensity, W1JointMass, fair_real, W1BinaryValue]

lemma fair_integral (f : Bool → ℝ) :
    (∫ b, f b ∂W1Fair) = (f false + f true) / 2 := by
  rw [W1Fair, integral_add_measure
    ((integrable_dirac (by finiteness)).smul_measure (by norm_num))
    ((integrable_dirac (by finiteness)).smul_measure (by norm_num))]
  simp only [integral_smul_measure, integral_dirac]
  norm_num
  ring

lemma fair_cef : W1CEF W1Fair (fun _ : Bool => ()) W1BinaryValue
    (fun _ : Unit => (1 / 2 : ℝ)) := by
  refine ⟨measurable_const, Integrable.of_finite, integrable_const _, ?_⟩
  intro A _
  by_cases h : () ∈ A
  · have hp : (fun _ : Bool => ()) ⁻¹' A = Set.univ := by ext b; simp [h]
    rw [hp]
    simp only [Measure.restrict_univ]
    rw [fair_integral, fair_integral]
    norm_num [W1BinaryValue]
  · have hp : (fun _ : Bool => ()) ⁻¹' A = ∅ := by ext b; simp [h]
    rw [hp]
    simp

-- T20: all hypotheses of the general overlap clause hold together.
example :
    Measurable (fun _ : Bool => ()) ∧ Measurable W1BinaryValue ∧
    (∀ b, W1BinaryValue b = 0 ∨ W1BinaryValue b = 1) ∧
    0 < probability W1Fair {b | W1BinaryValue b = 0} ∧
    0 < probability W1Fair {b | W1BinaryValue b = 1} ∧
    W1CEF W1Fair (fun _ : Bool => ()) W1BinaryValue
      (fun _ : Unit => (1 / 2 : ℝ)) ∧
    (∃ ε : ℝ, 0 < ε ∧ ∀ᵐ _b ∂W1Fair,
      ε < (1 / 2 : ℝ) ∧ (1 / 2 : ℝ) < 1 - ε) := by
  refine ⟨measurable_const, Measurable.of_discrete, ?_, ?_, ?_, fair_cef, ?_⟩
  · intro b; cases b <;> simp [W1BinaryValue]
  · rw [fair_arm 0 (Or.inl rfl)]; norm_num
  · rw [fair_arm 1 (Or.inr rfl)]; norm_num
  · exact ⟨1 / 4, by norm_num, Filter.Eventually.of_forall (by intro _; norm_num)⟩

lemma constant_pair_ci :
    W1CI W1Fair (fun _ : Bool => ()) measurable_const
      (fun _ => ((0, 1) : ℝ × ℝ)) W1BinaryValue := by
  intro A B _ _
  have hm : MeasurableSpace.comap (fun _ : Bool => ()) inferInstance ≤
      (inferInstance : MeasurableSpace Bool) :=
    (measurable_const : Measurable (fun _ : Bool => ())).comap_le
  by_cases h : ((0, 1) : ℝ × ℝ) ∈ A
  · simp [h, condExp_const hm]
  · simp [h]

lemma finite_count (n : Fin 100) :
    ((Finset.univ.filter (fun k : Fin 100 =>
      (k.val : ℝ) + 1 ≤ (n.val : ℝ))).card : ℝ) = n.val := by
  have heq : Finset.univ.filter (fun k : Fin 100 =>
      (k.val : ℝ) + 1 ≤ (n.val : ℝ)) = Finset.Iio n := by
    ext k
    simp only [Finset.mem_filter, Finset.mem_univ, true_and, Finset.mem_Iio]
    change (k.val : ℝ) + 1 ≤ (n.val : ℝ) ↔ k.val < n.val
    rw [← Nat.cast_one, ← Nat.cast_add, Nat.cast_le]
    omega
  rw [heq, Fin.card_Iio]

lemma voting_count_of_threshold (i : Fin 4) (d : Bool) (n : Fin 100)
    (h : 20 + 40 * electorateType i + 5 * (if d then 1 else 0) = (n.val : ℝ)) :
    countedVoteShare i d = potentialVoteShare (if d then 1 else 0) i := by
  unfold countedVoteShare votesNo potentialVoteShare
  rw [h]
  exact finite_count n

lemma voting_counts : ∀ i d,
    countedVoteShare i d = potentialVoteShare (if d then 1 else 0) i := by
  intro i d
  have hu : electorateType i = 0 ∨ electorateType i = 1 := by
    fin_cases i <;> norm_num [electorateType]
  rcases hu with hu | hu
  · cases d
    · exact voting_count_of_threshold _ _ ⟨20, by norm_num⟩ (by norm_num [hu])
    · exact voting_count_of_threshold _ _ ⟨25, by norm_num⟩ (by norm_num [hu])
  · cases d
    · exact voting_count_of_threshold _ _ ⟨60, by norm_num⟩ (by norm_num [hu])
    · exact voting_count_of_threshold _ _ ⟨65, by norm_num⟩ (by norm_num [hu])

lemma voting_dependence :
    ¬ FiniteConditionalIndependence voterMass
      (fun i => (potentialVoteShare 0 i, potentialVoteShare 1 i))
      displayedGender (fun _ => (0 : ℝ)) := by
  intro h
  have hw : 0 < finiteProbability voterMass (fun _ => (0 : ℝ) = 0) := by
    norm_num [finiteProbability, voterMass, Fin.sum_univ_succ]
  have bad := h 0 (60, 65) 1 hw
  norm_num [finiteProbability, voterMass, potentialVoteShare, electorateType,
    displayedGender, Fin.sum_univ_succ] at bad

-- T24: a genuine two-cell positive partition, with a measurable event B={true}.

    (⋃ i, A i) = Set.univ ∧
    (∀ i, 0 < probability W1Fair (A i)) ∧ MeasurableSet ({true} : Set Bool) := by
  refine ⟨fun b => {b}, ?_, ?_, ?_, ?_, ?_, measurableSet_singleton _⟩
  · intro b; exact measurableSet_singleton _
  · intro b; exact ⟨b, rfl⟩
  · intro b c h; simpa using h
  · ext b; simp
  · intro b; cases b <;> norm_num [fair_real]
