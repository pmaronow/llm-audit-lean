-- T26: an integer atom of full mass and a noninteger value at a null sample point.
def nullNonintegerX (b : Bool) : ℝ := if b then 1/2 else 0

example : IsProbabilityMeasure (Measure.dirac false : Measure Bool) ∧
    Measurable nullNonintegerX ∧ (Set.range nullNonintegerX).Countable ∧
    (∀ x : ℝ, (¬ ∃ k : ℤ, (k : ℝ) = x) →
      probability (Measure.dirac false) {ω | nullNonintegerX ω = x} = 0) ∧
    nullNonintegerX true = 1/2 ∧
    (Measure.dirac false : Measure Bool) {true} = 0 := by
  refine ⟨inferInstance, Measurable.of_discrete, (Set.finite_range _).countable, ?_, rfl, ?_⟩
  · intro x hx
    have h0 : (0 : ℝ) ≠ x := fun h => hx ⟨0, by simpa using h⟩
    simp [probability, nullNonintegerX, h0]
  · simp

-- T27: prove normalization of the exact dyadic weights independently of T27.
lemma dyadic_level (m : ℕ) :
    (∑ k : Fin (2 ^ m), dyadicWeight ⟨m, k⟩) = (1/2 : ℝ) ^ (m+1) := by
  simp only [dyadicWeight, Finset.sum_const, Finset.card_univ, Fintype.card_fin,
    nsmul_eq_mul, Nat.cast_pow, Nat.cast_ofNat]
  have he : 2 * m + 1 = m + (m + 1) := by omega
  rw [he, pow_add, ← mul_assoc, ← mul_pow]
  norm_num

lemma dyadic_summable : Summable dyadicWeight := by
  apply (summable_sigma_of_nonneg (fun i => by unfold dyadicWeight; positivity : ∀ i : DyadicIndex, 0 ≤ dyadicWeight i)).2
  constructor
  · intro m
    exact hasSum_fintype (fun k : Fin (2 ^ m) => dyadicWeight ⟨m, k⟩) |>.summable
  · simp only [tsum_fintype, dyadic_level]
    simpa [pow_succ] using summable_geometric_two.mul_right (1/2 : ℝ)

lemma dyadic_total : (∑' i : DyadicIndex, dyadicWeight i) = 1 := by
  rw [Summable.tsum_sigma' (fun m => (hasSum_fintype _).summable) dyadic_summable]
  simp only [tsum_fintype, dyadic_level]
  have h := (hasSum_geometric_two.mul_right (1/2 : ℝ)).tsum_eq
  norm_num at h
  simpa [pow_succ] using h

local instance : MeasurableSpace DyadicIndex := ⊤

example : (∀ s : Set DyadicIndex, MeasurableSet s) ∧
    Measurable dyadicLocation ∧ IsProbabilityMeasure W1DyadicLaw ∧
    (∀ i : DyadicIndex, 0 < dyadicWeight i) ∧
    dyadicLocation ⟨0, 0⟩ = 1/2 := by
  refine ⟨fun _ => trivial, Measurable.of_discrete, ?_, ?_, ?_⟩
  · constructor
    simp only [W1DyadicLaw, Measure.sum_apply _ MeasurableSet.univ,
      Measure.smul_apply, Measure.dirac_apply_of_mem (Set.mem_univ _), smul_eq_mul, mul_one]
    rw [← ENNReal.ofReal_tsum_of_nonneg (fun i => by unfold dyadicWeight; positivity : ∀ i : DyadicIndex, 0 ≤ dyadicWeight i)
      dyadic_summable, dyadic_total]
    norm_num
  · intro i
    unfold dyadicWeight
    positivity
  · norm_num [dyadicLocation]

-- T29: the explicit functions are genuine integrable densities, not default integrals.
lemma closed_indicator : closedUniformDensity =
    (Set.Icc (0:ℝ) 1).indicator (fun _ : ℝ => (1:ℝ)) := by
  funext x
  simp [closedUniformDensity, Set.indicator]

lemma open_indicator : openUniformDensity =
    (Set.Ioo (0:ℝ) 1).indicator (fun _ : ℝ => (1:ℝ)) := by
  funext x
  simp [openUniformDensity, Set.indicator]

lemma closed_integrable : Integrable closedUniformDensity volume := by
  have h : IntegrableOn (fun _ : ℝ => (1 : ℝ)) (Set.Icc (0:ℝ) 1) volume :=
    integrableOn_const (by norm_num [Real.volume_Icc])
  rw [closed_indicator]
  exact h.integrable_indicator measurableSet_Icc

lemma open_integrable : Integrable openUniformDensity volume := by
  have h : IntegrableOn (fun _ : ℝ => (1 : ℝ)) (Set.Ioo (0:ℝ) 1) volume :=
    integrableOn_const (by norm_num [Real.volume_Ioo])
  rw [open_indicator]
  exact h.integrable_indicator measurableSet_Ioo

lemma modified_ae : closedUniformDensity =ᵐ[volume] modifiedUniformDensity := by
  apply ae_iff.mpr
  apply measure_mono_null (t := {(1/3 : ℝ)}) ?_ (measure_singleton _)
  intro x hx
  simp only [Set.mem_ofPred_eq] at hx
  by_contra hn
  have hxne : x ≠ (1/3 : ℝ) := by simpa using hn
  have hxne_inv : x ≠ (3 : ℝ)⁻¹ := by simpa only [one_div] using hxne
  exact hx (by simp [modifiedUniformDensity, hxne_inv])

example : Integrable closedUniformDensity volume ∧ Integrable openUniformDensity volume ∧
    Integrable modifiedUniformDensity volume ∧
    (∫ x : ℝ, closedUniformDensity x) = 1 ∧
    closedUniformDensity 0 ≠ openUniformDensity 0 ∧
    closedUniformDensity (1/3) ≠ modifiedUniformDensity (1/3) := by
  refine ⟨closed_integrable, open_integrable, closed_integrable.congr modified_ae, ?_, ?_, ?_⟩
  · rw [closed_indicator, integral_indicator measurableSet_Icc]
    norm_num [Real.volume_Icc]
  · norm_num [closedUniformDensity, openUniformDensity]
  · norm_num [closedUniformDensity, modifiedUniformDensity]

-- Identify the clamped function with the CDF of an actual uniform probability measure.
def uniformLaw : Measure ℝ := volume.restrict (Set.Icc 0 1)

example : IsProbabilityMeasure uniformLaw := by
  constructor
  norm_num [uniformLaw, Real.volume_Icc]

example (x : ℝ) : distributionFunction uniformLaw id x = uniformDistributionFunction x := by
  change (uniformLaw (Set.Iic x)).toReal = max 0 (min 1 x)
  rw [uniformLaw, Measure.restrict_apply measurableSet_Iic]
  have he : Set.Iic x ∩ Set.Icc (0:ℝ) 1 = Set.Icc 0 (min 1 x) := by
    ext u
    simp only [Set.mem_inter_iff, Set.mem_Iic, Set.mem_Icc, le_min_iff]
    tauto
  rw [he, Real.volume_Icc, sub_zero, ENNReal.toReal_ofReal']
  exact max_comm _ _
