-- T68: a genuine probability space, nonlinear signal, and independent mean-zero noise.
example : ∃ (μ : Measure ℝ) (X ε g : ℝ → ℝ),
    IsProbabilityMeasure μ ∧ Measurable X ∧ Measurable ε ∧
    IndepFun X ε μ ∧ Integrable ε μ ∧ (∫ ω, ε ω ∂μ) = 0 ∧
    Measurable g ∧ Integrable (g ∘ X) μ := by
  let μ : Measure ℝ := Measure.dirac 0
  let _ : IsProbabilityMeasure μ := inferInstance
  refine ⟨μ, id, (fun _ => 0), (fun x => x^2+1), inferInstance,
    measurable_id, measurable_const, ?_, integrable_const 0, ?_, ?_, ?_⟩
  · exact indepFun_const_right id 0
  · simp
  · fun_prop
  · exact integrable_dirac (by simp)

-- T70: a nonconstant, full-rank polynomial design at positive degree.

    0 < (2 : ℕ) ∧
    IsUnit (((SAP.P001d.polynomialDesign x 1).transpose *
      SAP.P001d.polynomialDesign x 1).det) := by
  refine ⟨![0, 1], ![2, 3], by norm_num, ?_⟩
  have hdet : (SAP.P001d.polynomialDesign (![0, 1] : Fin 2 → ℝ) 1).det = 1 := by
    rw [Matrix.det_fin_two]
    norm_num [SAP.P001d.polynomialDesign]
  rw [Matrix.det_mul, Matrix.det_transpose]
  rw [hdet]
  norm_num

-- T69 population example: independent coordinates on the uniform square.
lemma uniform01_eq : SAP.P001d.uniformLaw 0 1 =
    volume.restrict (Set.Icc (0 : ℝ) 1) := by
  unfold SAP.P001d.uniformLaw
  convert withDensity_indicator_one (μ := (volume : Measure ℝ)) measurableSet_Icc using 1
  congr 1
  funext x
  by_cases hx : x ∈ Set.Icc (0 : ℝ) 1 <;> simp [Set.indicator, hx]

def squareLaw : Measure (ℝ × ℝ) :=
  (SAP.P001d.uniformLaw 0 1).prod (SAP.P001d.uniformLaw 0 1)

instance uniform01_probability : IsProbabilityMeasure (SAP.P001d.uniformLaw 0 1) := by
  constructor
  rw [uniform01_eq]
  simp

instance squareLaw_probability : IsProbabilityMeasure squareLaw := by
  unfold squareLaw
  infer_instance

lemma piecewise_measurable : Measurable
    (fun xy : ℝ × ℝ => SAP.P001d.apdExample xy.1 xy.2) := by
  unfold SAP.P001d.apdExample
  exact Measurable.ite (measurableSet_le measurable_snd measurable_const)
    measurable_fst.neg (measurable_const.mul measurable_fst)

lemma piecewise_integrable : Integrable
    (fun xy : ℝ × ℝ => SAP.P001d.apdExample xy.1 xy.2) squareLaw := by
  have hx : Integrable (fun x : ℝ => x) (SAP.P001d.uniformLaw 0 1) := by
    rw [uniform01_eq]
    exact continuous_id.integrableOn_Icc
  have hf : Integrable (fun xy : ℝ × ℝ => xy.1) squareLaw := hx.comp_fst _
  apply (hf.norm.const_mul 9).mono' piecewise_measurable.aestronglyMeasurable
  filter_upwards [] with xy
  unfold SAP.P001d.apdExample
  by_cases h : xy.2 ≤ (4/5 : ℝ)
  · simp only [h, if_pos, Real.norm_eq_abs, abs_neg]
    nlinarith [abs_nonneg xy.1]
  · simp [h, Real.norm_eq_abs, abs_mul]

example : ∃ (μ : Measure (ℝ × ℝ)) (X1 X2 Y : ℝ × ℝ → ℝ),
    IsProbabilityMeasure μ ∧ Measurable X1 ∧ Measurable X2 ∧ Integrable Y μ ∧
    IndepFun X1 X2 μ ∧ μ.map X1 = SAP.P001d.uniformLaw 0 1 ∧
    μ.map X2 = SAP.P001d.uniformLaw 0 1 ∧
    SAP.P001d.IsCEF μ (fun ω => (X1 ω, X2 ω)) Y
      (fun xy => SAP.P001d.apdExample xy.1 xy.2) := by
  refine ⟨squareLaw, Prod.fst, Prod.snd,
    (fun xy => SAP.P001d.apdExample xy.1 xy.2), inferInstance,
    measurable_fst, measurable_snd, piecewise_integrable, ?_, ?_, ?_, ?_⟩
  · exact indepFun_prod measurable_id measurable_id
  · simp [squareLaw]
  · simp [squareLaw]
  · refine ⟨piecewise_measurable, piecewise_integrable, ?_, ?_⟩
    · simpa only [Function.comp_def, Prod.eta] using piecewise_integrable
    · intro A hA
      rfl

-- T69 plug-in clause: a nonlinear fitted function with genuine sample derivatives.
example : ∃ (x : Fin 2 → Fin 2 → ℝ) (mhat : (Fin 2 → ℝ) → ℝ) (k : Fin 2),
    0 < (2 : ℕ) ∧ ∀ i,
      DifferentiableAt ℝ (fun t : ℝ => mhat (x i + t • Pi.single k 1)) 0 := by
  refine ⟨![![0, 0], ![1, 1]], (fun z => z 0 ^ 2 + z 1), 0, by norm_num, ?_⟩
  intro i
  simp only [Pi.add_apply, Pi.smul_apply, Pi.single_apply, smul_eq_mul]
  fun_prop
