def fourLaw : Measure (Fin 4) :=
  ∑ i : Fin 4, (1/4 : ℝ≥0) • Measure.dirac i

instance fourProb : IsProbabilityMeasure fourLaw := by
  constructor
  apply (ENNReal.toReal_eq_one_iff _).mp
  norm_num [fourLaw, Fin.sum_univ_succ, ENNReal.toReal_add]

lemma four_integral (f : Fin 4 → ℝ) :
    (∫ i, f i ∂fourLaw) = finiteMean f := by
  rw [fourLaw, integral_finsetSum_measure]
  · simp [finiteMean, NNReal.smul_def, smul_eq_mul]
  · intro i _
    exact Integrable.of_finite

    ENNReal.toReal_add, ENNReal.toReal_mul] at ht

-- T74: a genuine CEF in the probability measure of the four atoms.
lemma four_cef : IsCEF fourLaw heteroX heteroY (fun x => 2+x) := by
  classical
  refine ⟨by fun_prop, Integrable.of_finite, Integrable.of_finite, ?_⟩
  intro A hA
  rw [← integral_indicator (measurable_heteroX hA),
      ← integral_indicator (measurable_heteroX hA)]
  rw [four_integral, four_integral]
  by_cases h0 : (0 : ℝ) ∈ A <;> by_cases h1 : (1 : ℝ) ∈ A <;>
    norm_num [finiteMean, Fin.sum_univ_succ, Set.indicator,
      heteroX, heteroY, h0, h1]
where
  measurable_heteroX {A : Set ℝ} (hA : MeasurableSet A) :
      MeasurableSet (heteroX ⁻¹' A) := (measurable_of_finite heteroX) hA

example : Measurable heteroX ∧ Measurable heteroY ∧
    IsCEF fourLaw heteroX heteroY (fun x => 2+x) ∧
    Measurable (fun x : ℝ => x) ∧
    Integrable ((fun x : ℝ => x) ∘ heteroX) fourLaw ∧
    Integrable (fun i => heteroX i * heteroResidual i) fourLaw ∧
    Integrable (fun i => (heteroResidual i)^2) fourLaw := by
  exact ⟨measurable_of_finite _, measurable_of_finite _, four_cef,
    measurable_id, Integrable.of_finite, Integrable.of_finite,
    Integrable.of_finite⟩

lemma conditional_of_indep
    {Ω E : Type} [MeasurableSpace Ω] [MeasurableSpace E]
    (μ : Measure Ω) [IsProbabilityMeasure μ]
    (X : Ω → E) (Y : Ω → ℝ) (ν : Measure ℝ) [IsProbabilityMeasure ν]
    (hX : Measurable X) (hY : Measurable Y)
    (hi : IndepFun X Y μ) (hmap : μ.map Y = ν) :
    WholeConditionalLaw μ X Y (fun _ => ν) := by
  refine ⟨fun _ => inferInstance, fun _ _ => measurable_const, ?_⟩
  intro A B hA hB
  change μ (X ⁻¹' A ∩ Y ⁻¹' B) = _
  rw [hi.measure_inter_preimage_eq_mul A B hA hB,
    setLIntegral_const, Measure.map_apply hX hA]
  have hb : μ (Y ⁻¹' B) = ν B := by
    rw [← hmap, Measure.map_apply hY hB]
  rw [hb, mul_comm]

    Measurable (fun _ : ℝ => (![1] : Fin 1 → ℝ)) ∧
    Measurable (id : ℝ → ℝ) ∧
    (∀ ω : ℝ, (![1] : Fin 1 → ℝ) 0 = 1) ∧

    WholeConditionalLaw (gaussianReal 0 1)
      (fun _ : ℝ => (![1] : Fin 1 → ℝ)) id
      (wholeNormalKernel (0 : Fin 1 → ℝ) 1) := by
  refine ⟨measurable_const, measurable_id, by simp, by norm_num, ?_⟩
  have hk : wholeNormalKernel (0 : Fin 1 → ℝ) 1 =
      fun _ : Fin 1 → ℝ => gaussianReal 0 1 := by
    funext x
    simp [wholeNormalKernel, wholeLinearMean]
    congr 1
  rw [hk]
  exact conditional_of_indep _ _ _ _ measurable_const measurable_id
    (indepFun_const_left _ _) (by simp)
