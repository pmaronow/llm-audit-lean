import SAP.P001g.Defs

open MeasureTheory ProbabilityTheory Filter
open scoped BigOperators ENNReal NNReal Topology

noncomputable section
namespace ReviewSieve

def intervalLaw : Measure ℝ := volume.restrict (Set.Icc (0 : ℝ) 1)
instance : IsProbabilityMeasure intervalLaw :=
  ⟨by norm_num [intervalLaw, Real.volume_Icc]⟩
def law : Measure (ℕ → ℝ) := Measure.infinitePi (fun _ : ℕ => intervalLaw)
instance : IsProbabilityMeasure law := by unfold law; infer_instance
def X (i : ℕ) (ω : ℕ → ℝ) : ℝ := ω i
def Y (_ : ℕ) (_ : ℕ → ℝ) : ℝ := 0
def density : ℝ → ℝ := (Set.Icc (0 : ℝ) 1).indicator (fun _ => 1)

lemma iid_pairs : SAP.P001g.IID law (fun i ω => (X i ω, Y i ω)) := by
  refine ⟨fun _ => by unfold X Y; fun_prop, ?_, ?_⟩
  · exact iIndepFun_infinitePi (P := fun _ : ℕ => intervalLaw)
      (X := fun _ x => (x, (0 : ℝ))) (by intro i; fun_prop)
  · intro i
    have h : IdentDistrib (fun ω : ℕ → ℝ => ω i) (fun ω => ω 0) law law :=
      ⟨(measurable_pi_apply i).aemeasurable, (measurable_pi_apply 0).aemeasurable,
        by simp [law, Measure.infinitePi_map_eval]⟩
    exact h.comp (measurable_id.prodMk measurable_const)

lemma marginal_density : SAP.P001g.HasPDF (law.map (X 0)) density := by
  have hm : Measurable density := measurable_const.indicator measurableSet_Icc
  refine ⟨hm.aemeasurable, ?_, ?_, ?_⟩
  · intro x
    classical
    simp only [density, Set.indicator_apply]
    split_ifs <;> norm_num
  · apply (integrable_indicator_iff measurableSet_Icc).2
    exact integrable_const _
  · have hd : (fun x => ENNReal.ofReal (density x)) =
        (Set.Icc (0 : ℝ) 1).indicator (fun _ => (1 : ℝ≥0∞)) := by
      funext x
      classical
      by_cases h : x ∈ Set.Icc (0 : ℝ) 1 <;> simp [density, h]
    rw [hd, withDensity_indicator measurableSet_Icc, withDensity_const, one_smul]
    exact Measure.infinitePi_map_eval (fun _ : ℕ => intervalLaw) 0

lemma support_density : ∀ x, 0 < density x ↔ x ∈ Set.Icc (0 : ℝ) 1 := by
  intro x
  classical
  by_cases h : x ∈ Set.Icc (0 : ℝ) 1 <;> simp [density, h]

lemma zero_cef : SAP.P001g.IsCEF law (X 0) (Y 0) (fun _ => 0) := by
  refine ⟨measurable_const, integrable_const _, integrable_const _, ?_⟩
  intro A hA
  simp [Y]

lemma degree_atTop : Tendsto Nat.sqrt atTop atTop := by
  apply tendsto_atTop.2
  intro m
  filter_upwards [eventually_ge_atTop (m*m)] with n hn
  exact Nat.le_sqrt.mpr hn

lemma degree_rate : Tendsto (fun n : ℕ => (Nat.sqrt n : ℝ)/(n : ℝ)) atTop (nhds 0) := by
  have hu : ∀ n : ℕ, (Nat.sqrt n : ℝ)/(n : ℝ) ≤ (Nat.sqrt n : ℝ)⁻¹ := by
    intro n
    by_cases hs : Nat.sqrt n = 0
    · simp [hs]
    · have hspos : (0 : ℝ) < (Nat.sqrt n : ℝ) := by exact_mod_cast Nat.pos_of_ne_zero hs
      have hn : n ≠ 0 := by intro hn; simp [hn] at hs
      have hnpos : (0 : ℝ) < (n : ℝ) := by exact_mod_cast Nat.pos_of_ne_zero hn
      rw [inv_eq_one_div, div_le_div_iff₀ hnpos hspos]
      simpa [pow_two] using
        (show (Nat.sqrt n : ℝ) ^ 2 ≤ (n : ℝ) by exact_mod_cast Nat.sqrt_le' n)
  apply squeeze_zero (fun n => div_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _)) hu
  exact tendsto_inv_atTop_zero.comp (tendsto_natCast_atTop_atTop.comp degree_atTop)

lemma zero_fit : SAP.P001g.polynomialSieveLeastSquares X Y Nat.sqrt
    (fun _ _ => 0) := by
  intro n ω
  refine ⟨by simp, ?_⟩
  intro p hp
  simp only [Y, Polynomial.eval_zero, sub_zero, zero_pow (by decide : 2 ≠ 0),
    Finset.sum_const_zero]
  exact Finset.sum_nonneg (fun i hi => sq_nonneg _)

-- All of T119's hypotheses, including both rates and exact sample minimization.
-- This is a nonvacuity instance, not a counterexample to consistency.
example :
    SAP.P001g.IID law (fun i ω => (X i ω, Y i ω)) ∧
    Integrable (Y 0) law ∧ MemLp (Y 0) 2 law ∧ (0 : ℝ) < 1 ∧
    SAP.P001g.HasPDF (law.map (X 0)) density ∧
    (∀ x, 0 < density x ↔ x ∈ Set.Icc (0 : ℝ) 1) ∧
    ContDiff ℝ ⊤ (fun _ : ℝ => (0 : ℝ)) ∧
    SAP.P001g.IsCEF law (X 0) (Y 0) (fun _ => 0) ∧
    Tendsto Nat.sqrt atTop atTop ∧
    Tendsto (fun n : ℕ => (Nat.sqrt n : ℝ)/(n : ℝ)) atTop (nhds 0) ∧
    SAP.P001g.polynomialSieveLeastSquares X Y Nat.sqrt (fun _ _ => 0) := by
  exact ⟨iid_pairs, integrable_const _, memLp_const _, by norm_num, marginal_density,
    support_density, contDiff_const, zero_cef, degree_atTop, degree_rate, zero_fit⟩

end ReviewSieve
