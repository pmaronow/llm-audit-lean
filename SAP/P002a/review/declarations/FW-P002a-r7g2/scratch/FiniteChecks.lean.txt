example : ∃ x : Fin 2 → Fin 1 → ℝ,
    (∀ s, ∑ i, x i s = 0) ∧
    (∀ s, mean (fun i => x i s^4) = 16) ∧
    covariateGram x = (fun _ _ => 4) ∧
    gram x Finset.univ = Matrix.diagonal
      (fun j : Option (Fin 1) => match j with | none => 1 | some _ => 4) ∧
    ∃ b : Option (Fin 1) → ℝ,
      normSq b = 1 ∧ dotProduct b (gram x Finset.univ *ᵥ b) = 1 ∧
      dotProduct b (gram x Finset.univ *ᵥ b) < 2 * normSq b := by
  let x : Fin 2 → Fin 1 → ℝ := fun i _ => if i = 0 then -2 else 2
  have hx : gram x Finset.univ = Matrix.diagonal
      (fun j : Option (Fin 1) => match j with | none => 1 | some _ => 4) := by
    ext s t
    fin_cases s <;> fin_cases t <;>
      norm_num [gram, groupMean, augmented, x, Fin.sum_univ_succ]
  refine ⟨x, ?_, ?_, ?_, hx, ?_⟩
  · intro s
    norm_num [x, Fin.sum_univ_succ]
  · intro s
    norm_num [mean, x, Fin.sum_univ_succ]
  · ext s t
    norm_num [covariateGram, mean, x, Fin.sum_univ_succ]
  · refine ⟨fun j => if j = none then 1 else 0, ?_⟩
    rw [hx]
    norm_num [normSq, dotProduct, Matrix.mulVec, Fin.sum_univ_succ,
      Fintype.sum_option]
