def px : Fin 4 → Fin 1 → ℝ := fun i _ => if i.val < 2 then -1 else 1
def signal : Fin 4 → ℝ := ![-1, 1, -1, 1]
def P : CovPopulation 4 1 where
  pop := {y := fun i _ => signal i, d := fun _ a => a, n1 := 2}
  x := px

lemma P_valid : ValidDesign P.pop := by norm_num [P, ValidDesign]
lemma P_mono : MonotoneReceipt P.pop := by simp [P, MonotoneReceipt]
lemma P_compliers : (compliers P.pop).Nonempty := by
  exact ⟨0, by simp [P, compliers]⟩
lemma P_late : late P.pop = 0 := by simp [late, P, groupMean]

lemma population_gram : gram px Finset.univ = 1 := by
  ext a b
  fin_cases a <;> fin_cases b <;>
    norm_num [gram, groupMean, augmented, px, Fin.sum_univ_succ, Matrix.one_apply]

lemma population_score : score px Finset.univ signal = 0 := by
  ext a
  fin_cases a <;>
    norm_num [score, groupMean, augmented, px, signal, Fin.sum_univ_succ]

lemma P_residual (a : Bool) : residual P a (late P.pop) = signal := by
  have hw : w P.pop a (late P.pop) = signal := by
    ext i
    rw [P_late]
    simp [w, P]
  have hp : psi P a (late P.pop) = 0 := by
    change ols px Finset.univ (w P.pop a (late P.pop)) = 0
    rw [hw]
    simp [ols, population_gram, population_score]
  ext i
  simp only [SAP.P002b.residual, hp, dotProduct_zero, sub_zero]
  exact congrFun hw i

    norm_num [mean, P, px, Fin.sum_univ_succ]

def Qzero : QuadraticAR := ⟨0, 0, 0, 0, 0, 1⟩
def Qsquare : QuadraticAR := ⟨1, 0, 0, 0, 0, 1⟩
def Qfamily (m : Fin 2) : QuadraticAR := if m = 0 then Qsquare else Qzero
def gs (m : Fin 2) (b : ℝ) : ℝ := if m = 0 then b^2 else 0

lemma gs_eq : (fun m => arFunction (Qfamily m)) = gs := by
  ext m b
  fin_cases m <;> simp [gs, Qfamily, Qsquare, Qzero, arFunction]

lemma Qregular : ARRegular Qsquare ∧ (∀ m, ARRegular (Qfamily m)) := by
  have hs : ARRegular Qsquare := by
    constructor
    · intro b; norm_num [Qsquare]
    · intro b; simpa [Qsquare] using sq_nonneg b
  refine ⟨hs, ?_⟩
  intro m
  fin_cases m
  · simpa [Qfamily] using hs
  · constructor <;> intro b <;> norm_num [Qfamily, Qzero]

lemma gs_crossing (b : ℝ) : crossing gs b ↔ b = 0 := by
  constructor
  · rintro ⟨s, t, hne, he⟩
    fin_cases s <;> fin_cases t
    · exact False.elim (hne rfl)
    · simpa [gs, sq_eq_zero_iff] using he
    · have hb2 : b ^ 2 = 0 := by simpa [gs] using he.symm
      exact (sq_eq_zero_iff).mp hb2
    · exact False.elim (hne rfl)
  · intro hb
    subst b
    refine ⟨0, 1, ?_, by norm_num [gs]⟩
    intro h
    have h1 := congrFun h 1
    norm_num [gs] at h1

lemma gs_intersections : IntersectionList gs [0] := by
  constructor
  · simp
  · intro b
    simp [gs_crossing]

lemma gs_quantile (b : ℝ) : mcCritical gs (1/2) b = 0 := by
  let C := (Finset.univ.image (fun m : Fin 2 => Real.sqrt (gs m b))).filter fun q =>
    (1-(1/2 : ℝ))*(2 : ℝ) ≤
      ((Finset.univ.filter (fun m => Real.sqrt (gs m b) ≤ q)).card : ℝ)
  have hz : 0 ∈ C := by
    dsimp [C]
    apply Finset.mem_filter.mpr
    constructor
    · exact Finset.mem_image.mpr ⟨1, Finset.mem_univ _, by norm_num [gs]⟩
    · have hm : (1 : Fin 2) ∈ Finset.univ.filter (fun m => Real.sqrt (gs m b) ≤ 0) := by
        apply Finset.mem_filter.mpr
        exact ⟨Finset.mem_univ _, by norm_num [gs]⟩
      have hc := Finset.card_pos.mpr ⟨1, hm⟩
      have hc' : 1 ≤ (Finset.univ.filter (fun m : Fin 2 => Real.sqrt (gs m b) ≤ 0)).card :=
        Nat.succ_le_of_lt hc
      calc
        (1-(1/2 : ℝ))*(2 : ℝ) = 1 := by norm_num
        _ ≤ _ := by exact_mod_cast hc'
  have hne : C.Nonempty := ⟨0, hz⟩
  have hnonneg : ∀ q ∈ C, (0 : ℝ) ≤ q := by
    intro q hq
    obtain ⟨m, _, rfl⟩ := Finset.mem_image.mp (Finset.mem_filter.mp hq).1
    exact Real.sqrt_nonneg _
  change (if h : C.Nonempty then C.min' h else 0) = 0
  rw [dite_eq_left hne]
  exact le_antisymm (Finset.min'_le C 0 hz) (hnonneg _ (Finset.min'_mem C hne))

lemma gs_index (b : ℝ) : (1 : Fin 2) ∈ mcIndices gs (1/2) b := by
  apply Finset.mem_filter.mpr
  exact ⟨Finset.mem_univ _, by rw [gs_quantile]; norm_num [gs]⟩

def rawSignal : Fin 4 → ℝ := ![1, -1, -1, 1]
def Praw : CovPopulation 4 1 where
  pop := {y := fun i _ => rawSignal i, d := fun _ a => a, n1 := 2}
  x := px
def Oraw : Finset (Fin 4) := {0, 1}
def Traw (m : Fin 2) : Finset (Fin 4) := if m = 0 then Oraw else S
def RQ0 : QuadraticAR := ⟨1/16, 0, 0, 0, 0, 1/16⟩
def RQ1 : QuadraticAR := ⟨0, 0, 0, 1/64, 0, 1/16⟩
def RQg (m : Fin 2) : QuadraticAR := if m = 0 then RQ0 else RQ1

lemma raw_coeff0 : RQ0 = dataCoefficients Praw false Oraw Oraw := by
  have hc : Orawᶜ = ({2, 3} : Finset (Fin 4)) := by decide
  have hd : (fun i => binary (observedD Praw.pop Oraw i)) = ![(1 : ℝ), 1, 0, 0] := by
    ext i
    fin_cases i <;> norm_num [binary, observedD, Praw, Oraw]
  change RQ0 = rawCoefficients Praw.pop rawSignal
    (fun i => binary (observedD Praw.pop Oraw i)) Oraw
  rw [hd]
  dsimp only [RQ0, rawCoefficients, rawTau, rawSigma2, rawVarianceCross,
    fraction, controlSize, armMean, groupMean]
  simp only [hc]
  simp only [Fin.sum_univ_succ]
  congr 1 <;>
    norm_num [Praw, Oraw, rawSignal, Fin.sum_univ_succ]

lemma raw_coeff1 : RQ1 = dataCoefficients Praw false Oraw S := by
  have hc : Sᶜ = ({1, 3} : Finset (Fin 4)) := by decide
  have hd : (fun i => binary (observedD Praw.pop Oraw i)) = ![(1 : ℝ), 1, 0, 0] := by
    ext i
    fin_cases i <;> norm_num [binary, observedD, Praw, Oraw]
  change RQ1 = rawCoefficients Praw.pop rawSignal
    (fun i => binary (observedD Praw.pop Oraw i)) S
  rw [hd]
  dsimp only [RQ1, rawCoefficients, rawTau, rawSigma2, rawVarianceCross,
    fraction, controlSize, armMean, groupMean]
  simp only [hc]
  simp only [Fin.sum_univ_succ]
  congr 1 <;>
    norm_num [Praw, S, rawSignal, Fin.sum_univ_succ]

lemma raw_generated : DataGeneratedCoefficients RQ0 RQg := by
  refine ⟨4, 1, Praw, false, Oraw, Traw, ?_, ?_, ?_, ?_, ?_, ?_, raw_coeff0, ?_⟩
  · norm_num [ValidDesign, Praw]
  · simp [MonotoneReceipt, Praw]
  · exact ⟨0, by simp [compliers, Praw]⟩
  · simp [assignments, Oraw, Praw, Finset.mem_powersetCard]
  · intro m
    fin_cases m <;> simp [Traw, Oraw, S, assignments, Praw, Finset.mem_powersetCard]
  · simp
  · intro m
    fin_cases m
    · simpa [RQg, Traw] using raw_coeff0
    · simpa [RQg, Traw] using raw_coeff1

lemma raw_regular : ARRegular RQ0 ∧ (∀ m, ARRegular (RQg m)) := by
  constructor
  · constructor
    · intro b; norm_num [RQ0]
    · intro b
      dsimp [RQ0]
      nlinarith [sq_nonneg b]
  · intro m
    fin_cases m <;> constructor <;> intro b <;> dsimp [RQg, RQ0, RQ1]
    · norm_num
    · nlinarith [sq_nonneg b]
    · nlinarith [sq_nonneg b]
    · norm_num

lemma raw_curves : (fun m => arFunction (RQg m)) = gs := by
  ext m b
  fin_cases m <;> dsimp [RQg, RQ0, RQ1, arFunction, gs] <;> ring

def rawPieces : List (RecordedPiece 2) := [⟨⊥, 0, 1⟩, ⟨0, ⊤, 1⟩]

lemma raw_algorithm1 : Algorithm1Run gs (1/2) [0] rawPieces := by
  refine ⟨gs_intersections, ?_, ?_⟩
  · simp [rawPieces, gapEndpoints]
  · intro p hp
    simp only [rawPieces, List.mem_cons, List.not_mem_nil, or_false] at hp
    rcases hp with rfl | rfl <;> exact gs_index _

lemma gs_meets (b : ℝ) : (∃ m, gs m ≠ gs 1 ∧ gs m b = gs 1 b) ↔ b = 0 := by
  constructor
  · rintro ⟨m, hn, he⟩
    exact (gs_crossing b).mp ⟨m, 1, hn, he⟩
  · intro hb
    subst b
    refine ⟨0, ?_, by norm_num [gs]⟩
    intro h
    have h1 := congrFun h 1
    norm_num [gs] at h1

lemma raw_algorithm2 : Algorithm2Run gs (1/2) [0] rawPieces := by
  refine ⟨gs_intersections, by simp, by simp [rawPieces], ?_, ?_⟩
  · intro p hp
    simp only [rawPieces, List.head?_cons, Option.some.injEq] at hp
    subst p
    exact ⟨rfl, gs_index _⟩
  · intro i
    fin_cases i
    · change Algorithm2Step gs (1/2) [0] ⟨⊥, 0, 1⟩ (some ⟨0, ⊤, 1⟩)
      dsimp only [Algorithm2Step]
      simp only [gs_meets]
      simp [IsLeast, lowerBounds]
      simpa only [one_div] using gs_index (1 : ℝ)
    · change Algorithm2Step gs (1/2) [0] ⟨0, ⊤, 1⟩ none
      dsimp only [Algorithm2Step]
      simp only [gs_meets]
      simp [Set.Nonempty]
