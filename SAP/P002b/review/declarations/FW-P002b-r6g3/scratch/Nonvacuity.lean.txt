def optionOneEquiv : Option (Fin 1) ≃ Fin 2 where
  toFun a := match a with | none => 0 | some _ => 1
  invFun i := if i = 0 then none else some 0
  left_inv := by
    intro a
    cases a with
    | none => rfl
    | some i => have h : i = 0 := Subsingleton.elim _ _; subst i; rfl
  right_inv := by intro i; fin_cases i <;> rfl

lemma det_option_one (G : Matrix (Option (Fin 1)) (Option (Fin 1)) ℝ) :
    G.det = G none none * G (some 0) (some 0) -
      G none (some 0) * G (some 0) none := by
  rw [← Matrix.det_reindex_self optionOneEquiv G, Matrix.det_fin_two]
  rfl

lemma ols_eq_of_normal {n k : ℕ} (x : Fin n → Fin k → ℝ)
    (S : Finset (Fin n)) (q : Fin n → ℝ) (c : Option (Fin k) → ℝ)
    (hG : (gram x S).det ≠ 0) (heq : gram x S *ᵥ c = score x S q) :
    ols x S q = c := by
  rw [ols, if_pos hG, ← heq, Matrix.mulVec_mulVec,
    Matrix.nonsing_inv_mul _ (isUnit_iff_ne_zero.mpr hG), Matrix.one_mulVec]

def ivFourEquiv : IVIndex 1 ≃ Fin 4 where
  toFun j := match j with
    | Sum.inl i => if i = 0 then 0 else 1
    | Sum.inr (false, _) => 2
    | Sum.inr (true, _) => 3
  invFun i := (![Sum.inl 0, Sum.inl 1, Sum.inr (false,0), Sum.inr (true,0)] :
    Fin 4 → IVIndex 1) i
  left_inv := by
    intro j
    rcases j with i | ⟨a,i⟩
    · fin_cases i <;> rfl
    · fin_cases i; cases a <;> rfl
  right_inv := by intro i; fin_cases i <;> rfl
