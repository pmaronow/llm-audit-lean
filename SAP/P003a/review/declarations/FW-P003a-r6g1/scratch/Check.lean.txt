-- T2: a balanced law with assignment-dependent nonempty missingness.
def balancedV : Counts := ⟨1,0,0,1⟩
def dependentMask (q : Counts) : Counts :=
  if q.a = 1 then observed balancedV q else ⟨0,0,0,0⟩
noncomputable def dependentLaw (q r : Counts) : ℝ :=
  if r = dependentMask q then 1/2 else 0

lemma dependentLawValid : IsMissingLaw balancedV 1 dependentLaw := by
  constructor
  · intro q _ r _
    unfold dependentLaw
    split <;> norm_num
  · intro q hq
    have hs : randomStates balancedV 1 = [⟨0,0,0,1⟩,⟨1,0,0,0⟩] := by decide
    rw [hs] at hq
    simp only [List.mem_cons, List.not_mem_nil, or_false] at hq
    rcases hq with rfl | rfl
    all_goals norm_num [dependentLaw, dependentMask, balancedV, observed, missingMasks,
      List.range_succ, SAP.P003a.multiplicity, binomial, Counts.total, Nat.fast_choose]

example : dependentMask ⟨1,0,0,0⟩ ≠ dependentMask ⟨0,0,0,1⟩ := by decide

example : let v := balancedV; let m := 1; let α : ℝ := 1/2;
    0 < m ∧ v.total = 2*m ∧ 0 < α ∧ α < 1 ∧
      IsMissingLaw v m dependentLaw := by
  norm_num [Counts.total, balancedV]
  exact dependentLawValid

-- T2 clause 3: d=1, strict estimate inequality, and positive acceptance.
def coupleZ : Finset (Fin 4) := {0,1}
def coupleR : Finset (Fin 4) := {0}
def coupleY (i : Fin 4) : Bool := decide (2 ≤ i.val)
def coupleW (i : Fin 4) : Bool × Bool := (coupleY i,coupleY i)

example : coupleZ.card = 2 ∧
    (∀ i, i ∉ coupleR → coupleY i = coupleY i) ∧
    PossibleTable coupleY coupleZ coupleW := by
  constructor
  · decide
  constructor
  · intro i _; rfl
  · funext i
    simp [unitOutcomes, coupleW]

lemma coupleStrict : outcomeEstimator (unitPlus coupleY coupleZ coupleR) coupleZ <
    tableTau coupleW := by
  simp only [outcomeEstimator, tableTau, coupleW, sub_self, Finset.sum_const_zero,
    zero_div, Fin.sum_univ_four]
  norm_num [unitPlus, coupleZ, coupleR, coupleY, bit]

example : (Finset.univ.filter fun i => coupleY i ≠
    unitPlus coupleY coupleZ coupleR i).card = 1 := by native_decide

lemma couplePValue : unitPValue coupleW coupleY coupleZ = (1/3 : ℝ) := by
  let s : Finset (Assignment 4 2) :=
    {⟨{0,1},by decide⟩,⟨{0,2},by decide⟩,⟨{0,3},by decide⟩,
     ⟨{1,2},by decide⟩,⟨{1,3},by decide⟩,⟨{2,3},by decide⟩}
  have hs : (Finset.univ : Finset (Assignment 4 2)) = s := by native_decide
  unfold unitPValue uniformProb
  change ((Finset.univ.filter fun q : Assignment 4 2 =>
    |tableEstimator coupleW q.val-tableTau coupleW| ≥
    |outcomeEstimator coupleY coupleZ-tableTau coupleW|).card : ℝ) /
      Fintype.card (Assignment 4 2) = 1/3
  rw [hs]
  simp only [tableEstimator, tableTau, outcomeEstimator, coupleW, sub_self,
    Finset.sum_const_zero, zero_div, Fin.sum_univ_four]
  norm_num [s, Finset.filter_insert, Finset.filter_singleton, coupleY, coupleZ,
    bit, Assignment, Nat.choose]
  have hc : ({(⟨{0,1},by decide⟩ : Assignment 4 2),⟨{2,3},by decide⟩} :
      Finset (Assignment 4 2)).card = 2 := by decide
  norm_num only [hc]

example : (0 : ℕ) < 2 ∧ 4 = 2*2 ∧ coupleZ.card = 2 ∧
    (∀ i, i ∉ coupleR → coupleY i = coupleY i) ∧
    PossibleTable coupleY coupleZ coupleW ∧

    outcomeEstimator (unitPlus coupleY coupleZ coupleR) coupleZ < tableTau coupleW := by
  norm_num [couplePValue, coupleStrict, PossibleTable]
  constructor
  · decide
  · funext i; simp [unitOutcomes, coupleW]
