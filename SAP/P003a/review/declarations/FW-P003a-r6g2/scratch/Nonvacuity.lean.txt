-- T11: the printed reading asserts a least REAL threshold for the inclusive
-- tail, in a positive complete-randomization binary experiment at 0<alpha<1.
-- Here n=2,m=1,alpha=1/2, both potential pairs are (0,0), and the uniform
-- design has exactly two assignments. All printed conditions hold.
def zeroY : PotentialTable 2 := fun _ => (false,false)
def zeroZ : Assignment 2 1 := ⟨{0}, by simp⟩

example : 0 < (1 : ℕ) ∧ 1 < (2 : ℕ) ∧ (0 : ℝ) < 1/2 ∧ (1/2 : ℝ) < 1 := by

  simp [tableEstimator, zeroY, bit]

theorem zeroDeviation (z : Assignment 2 1) : w2Deviation zeroY z = 0 := by
  simp [w2Deviation, tableEstimator, tableTau, zeroY, bit]

theorem zeroTail (x : ℝ) : w2Tail 1 zeroY x = if x ≤ 0 then 1 else 0 := by
  classical
  have hc : Fintype.card (Assignment 2 1) = 2 := by decide
  by_cases hx : x ≤ 0 <;> simp [w2Tail, uniformProb, zeroDeviation, hx, hc]

example : {x : ℝ | w2Tail 1 zeroY x < 1/2} = Set.Ioi 0 := by
  ext x
  simp only [Set.mem_ofPred_eq, Set.mem_Ioi, zeroTail]
  by_cases hx : x ≤ 0 <;> norm_num [hx] <;> linarith

example : ¬ ∃ x : ℝ, w2Tail 1 zeroY x < 1/2 ∧
    ∀ u : ℝ, w2Tail 1 zeroY u < 1/2 → x ≤ u := by
  rintro ⟨x, hx, hmin⟩
  have hxpos : 0 < x := by
    by_contra hn
    norm_num [zeroTail, le_of_not_gt hn] at hx
  have hhalf : w2Tail 1 zeroY (x/2) < 1/2 := by
    rw [zeroTail, if_neg (by linarith)]
    norm_num
  have := hmin (x/2) hhalf
  linarith

example (x : ℝ) : designProb ⟨0,0,0,2⟩ 1

theorem halfPoint_eq_delta (x : W2Lattice 2) :
    w2HalfPointMass x = if x = w2Point 2 1 then 1 else 0 := by
  have he : x.val = (1/2 : ℝ) ↔ x = w2Point 2 1 := by
    constructor
    · intro h
      apply Subtype.ext
      simpa [w2Point] using h
    · intro h
      subst x
      norm_num [w2Point]
  simp only [w2HalfPointMass, he]

example : (0 : ℕ) < 2 ∧ W2LatticePMF w2HalfPointMass ∧
    W2LatticeMoment w2HalfPointMass := by
  classical
  refine ⟨by norm_num, ?_, ?_⟩
  · refine ⟨?_, ?_, ?_⟩
    · intro x
      rw [halfPoint_eq_delta]
      split_ifs <;> norm_num
    · have hf : w2HalfPointMass =
          fun x => if x = w2Point 2 1 then (1 : ℝ) else 0 := by
        funext x
        exact halfPoint_eq_delta x
      rw [hf]
      exact (hasSum_ite_eq (w2Point 2 1) (1 : ℝ)).summable
    · simp [halfPoint_eq_delta]
  · unfold W2LatticeMoment
    have he : (fun x : W2Lattice 2 => x.val*w2HalfPointMass x) =
        fun x => if x = w2Point 2 1 then (1/2 : ℝ) else 0 := by
      funext x
      rw [halfPoint_eq_delta]
      by_cases hx : x = w2Point 2 1
      · subst x
        norm_num [w2Point]
      · simp [hx]
    rw [he]
    exact (hasSum_ite_eq (w2Point 2 1) (1/2 : ℝ)).summable

example (j : ℤ) : w2Rescale w2HalfPointMass j = if j = 1 then 1 else 0 := by
  have he : (j : ℝ)/2 = 1/2 ↔ j = 1 := by
    constructor
    · intro h
      have : (j : ℝ) = 1 := by linarith
      exact_mod_cast this
    · intro h
      subst j
      norm_num
  simp only [w2Rescale, w2HalfPointMass, w2Point, Nat.cast_ofNat, he]

example (j : ℤ) : w2PrintedRescale j = 0 := by
  have hn : (2 : ℝ)*(j : ℝ) ≠ 1/2 := by
    intro h
    have hj : (4 : ℝ)*(j : ℝ) = 1 := by linarith
    have hz : (4 : ℤ)*j = 1 := by exact_mod_cast hj
    omega
  have he : (((4*j : ℤ) : ℝ)/2) = 2*(j : ℝ) := by push_cast; ring
  simp only [w2PrintedRescale, w2HalfPointMass, w2Point, Nat.cast_ofNat]
  simp only [he, if_neg hn]
