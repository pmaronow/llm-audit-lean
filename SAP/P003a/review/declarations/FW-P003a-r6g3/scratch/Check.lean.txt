lemma supportedMass (s : Finset ℤ) (f : ℤ → ℝ)
    (hz : ∀ j ∉ s, f j = 0) (hn : ∀ j, 0 ≤ f j)
    (hs : ∑ j ∈ s, f j = 1) : IsMass f := by
  refine ⟨hn, summable_of_ne_finset_zero hz, ?_, ?_⟩
  · rw [tsum_eq_sum hz]
    exact hs
  · apply summable_of_ne_finset_zero (s := s)
    intro j hj
    simp [hz j hj]

lemma fairSupported : ∀ j ∉ ({0,1} : Finset ℤ), fairMass j = 0 := by
  intro j hj
  simp only [Finset.mem_insert, Finset.mem_singleton, not_or] at hj
  simp [fairMass, hj.1, hj.2]

lemma fairMean : (∑' j : ℤ, (j : ℝ)*fairMass j) = 1/2 := by
  rw [tsum_eq_sum (s := ({0,1} : Finset ℤ)) (by
    intro j hj
    simp [fairSupported j hj])]
  norm_num [fairMass]

lemma fairShape : SDShape fairMass 1 := by
  constructor
  · intro i
    simp only [fairMass]
    split_ifs <;> norm_num at * <;> omega
  · intro i j hi hij
    simp only [fairMass]
    split_ifs <;> norm_num at * <;> omega

lemma fairSD : IsSD fairMass := by
  refine ⟨supportedMass {0,1} fairMass fairSupported ?_ ?_, 1, ?_, fairShape⟩
  · intro j
    unfold fairMass
    positivity
  · norm_num [fairMass]
  · simpa using fairMean

  refine ⟨fairSD, ?_, ?_⟩
  · intro j
    simp [fairMass, eq_comm, add_comm]
  · norm_num [dependentSumMass]

lemma dependentMean : (∑' j : ℤ, (j : ℝ)*dependentSumMass j) = 1 := by
  rw [tsum_eq_sum (s := ({0,2} : Finset ℤ)) (by
    intro j hj
    simp only [Finset.mem_insert, Finset.mem_singleton, not_or] at hj
    simp [dependentSumMass, hj.1, hj.2])]
  norm_num [dependentSumMass]

example : ¬ IsSD dependentSumMass := by
  rintro ⟨_, c, hc, hshape⟩
  have hcr : (c : ℝ) = 2 := by rw [dependentMean] at hc; linarith
  have hcz : c = 2 := by exact_mod_cast hcr
  subst c
  have h := hshape.2 1 2 (by norm_num) (by norm_num)
  norm_num [dependentSumMass] at h
