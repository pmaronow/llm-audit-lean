lemma constant_rank (K : ℕ) (o v : Counts) (s : Fin K → Counts) :
    rankValue (fun _ _ => 0) o v s = 1 := by
  simp only [rankValue, abs_zero, le_refl, ite_true, Finset.sum_const,
    Finset.card_univ, Fintype.card_fin, nsmul_eq_mul, mul_one]
  have h : (K : ℝ) + 1 ≠ 0 := by positivity
  simpa only [add_comm] using div_self h

def o23 : Counts := ⟨0,2,0,2⟩
def draw23a : Assignment 4 2 := ⟨{0,2}, by decide⟩
def draw23b : Assignment 4 2 := ⟨{1,3}, by decide⟩
def pool23 : Fin (2*w1TestBudget o23.total+1) → Fin 1 →
    Assignment o23.total (o23.a+o23.b) :=
  fun slot _ => if slot.val % 2 = 0 then draw23a else draw23b

example : (0 < o23.a+o23.b ∧ o23.a+o23.b = o23.c+o23.d) ∧
    (0 : ℕ) < 1 ∧ (0 < (1/2 : ℝ) ∧ (1/2 : ℝ) < 1) ∧
    (0 ≤ (0 : ℝ) ∧ (0 : ℝ) < 1/2) := by
  norm_num [o23]

example : (pool23 ⟨0,by decide⟩ 0).val ≠ (pool23 ⟨1,by decide⟩ 0).val := by
  change ({0,2} : Finset (Fin 4)) ≠ {1,3}
  decide

def reject23Q (v : Counts) (idx : ℕ) : Bool :=
  decide (w1MonteMean v o23 (w1Batch pool23 idx) < (1/2 : ℚ))

lemma actual_reject23 : w1ApproxReject (1/2) 0 o23 pool23 = reject23Q := by
  funext v idx
  unfold w1ApproxReject reject23Q
  simp only [add_zero]
  have hhalf : ((1/2 : ℚ) : ℝ) = (1/2 : ℝ) := by norm_num
  rw [← hhalf]
  simp only [Rat.cast_lt]

example : w1MonteMean ⟨0,1,0,3⟩ o23 (w1Batch pool23 0) = 1 ∧
    w1MonteMean ⟨0,2,0,2⟩ o23 (w1Batch pool23 1) = 0 := by native_decide

example : w1EffectUpperRun o23 reject23Q =
    ⟨some 1, [⟨1,0,false⟩,⟨2,1,true⟩],
      [(⟨0,1,0,3⟩,false),(⟨0,2,0,2⟩,true)]⟩ := by native_decide

example : (w1EffectUpperRun o23 (w1ApproxReject (1/2) 0 o23 pool23)).value = some 1 ∧
    (w1EffectUpperRun o23 (w1ApproxReject (1/2) 0 o23 pool23)).queries =
      [⟨1,0,false⟩,⟨2,1,true⟩] := by
  rw [actual_reject23]
  native_decide

example : intervalQ (1/2) o23 = some (-1/4,1/4) := by native_decide
