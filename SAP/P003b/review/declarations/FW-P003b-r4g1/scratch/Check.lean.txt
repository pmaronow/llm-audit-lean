-- T24: a genuinely varying estimator in a positive balanced experiment.
def auditW : PotentialTable 2 := fun i => if i.val = 0 then (true,true) else (false,false)
def auditZ : Finset (Fin 2) := {0}

example : 0 < (1 : ℕ) ∧ (2 : ℕ) = 2*1 ∧ auditZ.card = 1 := by
  norm_num [auditZ]

  rw [tableTau, Fin.sum_univ_two]

example : tableEstimator auditW auditZ = 1 := by
  norm_num [tableEstimator, auditW, auditZ, bit, Fin.sum_univ_two]
example : tableEstimator auditW auditZᶜ = -1 := by
  norm_num [tableEstimator, auditW, auditZ, bit, Fin.sum_univ_two, Finset.card_compl]
