lemma auditCovered : 0 ∈ waldInterval auditCritical 25 12 := by
  have hz : (1 : ℝ) ≤ auditCritical := by norm_num [auditCritical]
  have hs : (1/25 : ℝ) ≤ Real.sqrt (13/625) := by
    apply (Real.le_sqrt (by norm_num) (by norm_num)).2
    norm_num
  have hmul : (1/25 : ℝ) ≤ auditCritical*Real.sqrt (13/625) := by
    exact hs.trans (le_mul_of_one_le_left (Real.sqrt_nonneg _) hz)
  change (0 : ℝ) ∈ Set.Icc
    ((waldEstimateQ 25 12 : ℝ)-auditCritical*Real.sqrt (waldVarianceQ 25 12))
    ((waldEstimateQ 25 12 : ℝ)+auditCritical*Real.sqrt (waldVarianceQ 25 12))
  have ht : (waldEstimateQ 25 12 : ℝ) = -1/25 := by norm_num [waldEstimateQ]
  have hv : (waldVarianceQ 25 12 : ℝ) = 13/625 := by norm_num [waldVarianceQ]
  rw [ht, hv]
  constructor <;> linarith

-- The same singleton satisfies every finite-run antecedent, including >=0.925.
example : let runs : List (Fin 26) := [12]

      runs.length ≥ 0.925 ∧
    runs.length < 2*(runs.filter fun k => decide (0 ∈ waldInterval auditCritical 25 k.val)).length := by
  classical
  norm_num [auditCovered]
