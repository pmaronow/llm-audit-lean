lemma accepted_member (o v : Counts) (α : ℝ)
    (hv : v ∈ possibleTables o) (hp : α ≤ (pValue v o : ℝ)) :
    tau v ∈ accepted α o := by
  classical
  apply Finset.mem_image.mpr
  refine ⟨v, ?_, rfl⟩
  simpa only [List.mem_toFinset, List.mem_filter, decide_eq_true_eq] using And.intro hv hp

lemma nonempty_from_rational_filter (o : Counts) (α : ℚ)
    (h : ((possibleTables o).filter (fun v => decide (α ≤ pValue v o))) ≠ []) :
    (accepted (α : ℝ) o).Nonempty := by
  obtain ⟨v, hv⟩ := List.exists_mem_of_ne_nil _ h
  obtain ⟨hv, hp⟩ := List.mem_filter.mp hv
  have hp' : α ≤ pValue v o := of_decide_eq_true hp
  exact ⟨tau v, accepted_member o v (α : ℝ) hv (by exact_mod_cast hp')⟩

    badMask q ∈ missingMasks (observed badPopulation q) := by
  native_decide

theorem t26_finite_filters : ∀ q ∈ randomStates badPopulation 3,
    ((possibleTables (minusCompletion (observed badPopulation q) (badMask q))).filter
      (fun v => decide ((109/500 : ℚ) ≤
        pValue v (minusCompletion (observed badPopulation q) (badMask q))))) ≠ [] ∧
    ((possibleTables (plusCompletion (observed badPopulation q) (badMask q))).filter
      (fun v => decide ((109/500 : ℚ) ≤
        pValue v (plusCompletion (observed badPopulation q) (badMask q))))) ≠ [] := by
  native_decide

example : ∀ q ∈ randomStates badPopulation 3,
    (accepted (109/500) (minusCompletion (observed badPopulation q) (badMask q))).Nonempty ∧
    (accepted (109/500) (plusCompletion (observed badPopulation q) (badMask q))).Nonempty := by
  intro q hq
  obtain ⟨hm, hp⟩ := t26_finite_filters q hq
  have hcast : ((109/500 : ℚ) : ℝ) = 109/500 := by norm_num
  simpa only [hcast] using And.intro (nonempty_from_rational_filter _ _ hm)
    (nonempty_from_rational_filter _ _ hp)

lemma collapsed_execution (fuel : ℕ) :
    printedExecution fuel 1 2 (fun _ => true) 1 1 = none := by
  induction fuel with
  | zero => rfl
  | succ t ih => simp [printedExecution, searchStep, ih]

example (fuel : ℕ) : printedExecution fuel 1 2 (fun _ => true) 1 2 = none := by
  cases fuel with
  | zero => rfl
  | succ t => simp [printedExecution, searchStep, collapsed_execution]

-- A rational computation of T26's complete finite event. The endpoint
-- lists use the exact pValue comparison, without any real sInf/sSup.
def rationalExtrema (α : ℚ) (o : Counts) : Option (ℚ × ℚ) :=
  let vals := ((possibleTables o).filter (fun v => decide (α ≤ pValue v o))).map tauQ
  match vals with
  | [] => none
  | t :: ts => some (ts.foldl min t, ts.foldl max t)

def rationalMissingInterval (o r : Counts) : Option (ℚ × ℚ) :=
  let lo := minusCompletion o r
  let hi := plusCompletion o r
  match rationalExtrema (109/500) lo, rationalExtrema (109/500) hi with
  | some (l, _), some (_, u) => some (min l (estimateQ lo), max u (estimateQ hi))
  | _, _ => none

def t26CoverageNumerator : ℕ :=
  ((randomStates badPopulation 3).map fun q =>
    match rationalMissingInterval (observed badPopulation q) (badMask q) with
    | none => 0
    | some (l, u) => if l ≤ tauQ badPopulation ∧ tauQ badPopulation ≤ u
        then multiplicity badPopulation q else 0).sum

example : t26CoverageNumerator = 891 ∧ binomial badPopulation.total 3 = 1140 ∧
    (t26CoverageNumerator : ℚ) / binomial badPopulation.total 3 = 297/380 ∧
    (297/380 : ℚ) < 1-109/500 := by
  native_decide
