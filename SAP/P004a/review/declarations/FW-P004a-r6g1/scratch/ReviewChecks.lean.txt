lemma review_mle_exists {k : ℕ} (O : ∀ N, Fin N → Fin k → ℝ)
    (Θ : Set (Fin k → ℝ)) (hc : IsCompact Θ) (hne : Θ.Nonempty) :
    ∃ θhat : ∀ N, Assignment N → Fin k → ℝ, ∀ N z,
      θhat N z ∈ Θ ∧ ∀ θ ∈ Θ,
      (∑ i, (bit (z i)*dot (O N i) θ-Real.log (1+Real.exp (dot (O N i) θ)))) ≤
      (∑ i, (bit (z i)*dot (O N i) (θhat N z)-
        Real.log (1+Real.exp (dot (O N i) (θhat N z))))) := by
  have h : ∀ N (z : Assignment N), ∃ θ ∈ Θ, ∀ t ∈ Θ,
      (∑ i, (bit (z i)*dot (O N i) t-Real.log (1+Real.exp (dot (O N i) t)))) ≤
      (∑ i, (bit (z i)*dot (O N i) θ-Real.log (1+Real.exp (dot (O N i) θ)))) := by
    intro N z
    have hf : Continuous (fun θ : Fin k → ℝ =>
        ∑ i, (bit (z i)*dot (O N i) θ-Real.log (1+Real.exp (dot (O N i) θ)))) := by
      apply continuous_finsetSum
      intro i hi
      have hd : Continuous (fun θ : Fin k → ℝ => dot (O N i) θ) := by
        unfold dot
        fun_prop
      exact (continuous_const.mul hd).sub
        ((continuous_const.add (Real.continuous_exp.comp hd)).log
          (fun θ => by change 1+Real.exp (dot (O N i) θ) ≠ 0; positivity))
    obtain ⟨θ, hθ, hmax⟩ := hc.exists_isMaxOn hne hf.continuousOn
    exact ⟨θ, hθ, fun t ht => hmax ht⟩
  choose θhat hm hx using h
  exact ⟨θhat, fun N z => ⟨hm N z, hx N z⟩⟩

-- A nonempty, nonsingular, genuinely random finite instance of every finite T2 premise.
example : 0 < (2 : ℕ) ∧ (∀ i, 0 < reviewP i ∧ reviewP i < 1) ∧
    IsUnit (information reviewP reviewO).det ∧
    (∀ i, i ∈ reviewB i) ∧ (∀ i j, j ∈ reviewB i ↔ i ∈ reviewB j) ∧
    (∀ i j, j ∉ reviewB i → ¬ sharesSource reviewR i j) := by
  refine ⟨by norm_num, ?_, ?_, ?_, ?_, ?_⟩
  · intro i; norm_num [reviewP]
  · rw [Matrix.det_fin_one]
    norm_num [information, reviewP, reviewO, Fin.sum_univ_two, isUnit_iff_ne_zero]
  · intro i; simp [reviewB]
  · intro i j; simp [reviewB, eq_comm]
  · intro i j hj
    simp only [reviewB, Finset.mem_singleton] at hj
    rintro ⟨l, hil, hjl⟩
    have hi : i = l := by simpa [interferes, reviewR] using hil
    have hj' : j = l := by simpa [interferes, reviewR] using hjl
    exact hj (hj'.trans hi.symm)

def reviewInterceptTheta : Set (Fin 1 → ℝ) := {θ | ∀ j, -1 ≤ θ j ∧ θ j ≤ 1}

-- An actual MLE sequence satisfying every LogisticConditions clause, with an interior truth.
example : ∃ θhat : ∀ N, Assignment N → Fin 1 → ℝ,
    LogisticConditions (fun N (_ : Fin N) (_ : Fin 1) => 1)
      (fun _ => 0) reviewInterceptTheta θhat := by
  have hc : IsCompact reviewInterceptTheta := isCompact_pi_infinite (fun _ => isCompact_Icc)
  have hzero : (fun _ : Fin 1 => (0 : ℝ)) ∈ reviewInterceptTheta := by
    intro j; norm_num
  obtain ⟨θhat, hmle⟩ := review_mle_exists
    (fun N (_ : Fin N) (_ : Fin 1) => (1 : ℝ)) reviewInterceptTheta hc ⟨_, hzero⟩
  refine ⟨θhat, hc, hzero, ⟨1, ?_⟩, ?_, hmle⟩
  · intro N i j; norm_num
  · refine ⟨1, by norm_num, ?_⟩
    filter_upwards [eventually_ge_atTop (1 : ℕ)] with N hN
    intro v
    have hn : (N : ℝ) ≠ 0 := by exact_mod_cast (by omega : N ≠ 0)
    simp [dot, mul_div_cancel_left₀ _ hn]

example : (fun _ : Fin 1 => (0 : ℝ)) ∈ interior reviewInterceptTheta := by
  have hopen : IsOpen ((fun θ : Fin 1 → ℝ => θ 0) ⁻¹' Set.Ioo (-1 : ℝ) 1) :=
    isOpen_Ioo.preimage (continuous_apply 0)
  have hzero : (fun _ : Fin 1 => (0 : ℝ)) ∈
      ((fun θ : Fin 1 → ℝ => θ 0) ⁻¹' Set.Ioo (-1 : ℝ) 1) := by norm_num
  apply mem_interior_iff_mem_nhds.mpr
  apply Filter.mem_of_superset (hopen.mem_nhds hzero)
  intro θ hθ j
  fin_cases j
  exact ⟨le_of_lt hθ.1, le_of_lt hθ.2⟩
