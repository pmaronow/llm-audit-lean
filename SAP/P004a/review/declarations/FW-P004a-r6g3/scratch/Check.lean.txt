-- No target theorem is used in the independent examples below.
lemma sum_assignment_one {M : Type*} [AddCommMonoid M] (f : Assignment 1 → M) :
    (∑ z, f z) = f (fun _ => true) + f (fun _ => false) := by
  classical
  have hu : (Finset.univ : Finset (Assignment 1)) =
      {fun _ => true, fun _ => false} := by
    ext z
    simp only [Finset.mem_univ, Finset.mem_insert, Finset.mem_singleton, true_iff]
    cases hz : z 0
    · right; funext i; fin_cases i; exact hz
    · left; funext i; fin_cases i; exact hz
  have hne : (fun _ : Fin 1 => true) ≠ (fun _ => false) := by
    intro h
    have hh := congrFun h 0
    cases hh
  have hmem : (fun _ : Fin 1 => true) ∉ ({fun _ => false} : Finset (Assignment 1)) := by
    simpa only [Finset.mem_singleton] using hne
  rw [hu, Finset.sum_insert hmem, Finset.sum_singleton]
