-- Every own-coin table has precisely the forced diagonal interference graph.
lemma own_interferes {N : ℕ} (f : Fin N → Bool → ℝ) (i j : Fin N) :
    interferes (fun k z => f k (z k)) i j ↔ i = j := by
  constructor
  · rintro (h | ⟨z, z', hz, hne⟩)
    · exact h
    · by_contra hij
      exact hne (congrArg (f i) (hz i hij))
  · exact Or.inl

lemma own_no_shared {N : ℕ} (f : Fin N → Bool → ℝ) (i j : Fin N)
    (hij : i ≠ j) : ¬ sharesSource (fun k z => f k (z k)) i j := by
  rintro ⟨l, hi, hj⟩
  exact hij ((own_interferes f i l).mp hi |>.trans ((own_interferes f j l).mp hj).symm)
