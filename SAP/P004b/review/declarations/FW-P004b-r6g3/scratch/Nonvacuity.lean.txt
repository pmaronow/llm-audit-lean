lemma sum_assignments_two (f : Assignment 2 → ℝ) :
    (∑ z, f z) = f ![false,false] + f ![false,true] +
      f ![true,false] + f ![true,true] := by
  classical
  rw [← (finTwoArrowEquiv Bool).symm.sum_comp f]
  simp [Fintype.sum_prod_type]
  ring

-- Positive counts in a complete randomized design, including the supports.
example :
    (completeSupport 4 2).card = 6 ∧
    ((completeSupport 4 2).filter (fun z => z 0 = true)).card = 3 ∧
    (remainderSupport 2 (0 : Fin 4) true).card = 3 ∧
    (remainderSupport 2 (0 : Fin 4) false).card = 3 ∧
    Disjoint (remainderSupport 2 (0 : Fin 4) false)
      (remainderSupport 2 (0 : Fin 4) true) := by
  decide

-- Endpoint S3 marginalization does not turn the missing arm into zero.
example :
    forcedEffect (fun _ : Fin 2 => 0) (fun i z => bit (z i)) 0 = 1 ∧
    forcedAME (fun _ : Fin 2 => 0) (fun i z => bit (z i)) = 1 ∧
    (∀ i : Fin 2, forcedCentered (fun _ => 0) (fun i z => bit (z i)) i = 0) := by
  norm_num [forcedEffect, forcedMarginal, forcedAME, forcedCentered, expect,
    sum_assignments_two, designWeight, Fin.prod_univ_succ, Fin.sum_univ_succ,
    bit, coin]

-- A balanced deterministic table has positive linearized variance.
example :
    variance (fun _ : Fin 2 => 1/2)
      (linearHajek (fun _ => 1/2) (fun i _ => if i = 0 then 1 else -1)) = 2 := by
  norm_num [variance, cov, linearHajek, augmented, barMu, marginal, expect,
    sum_assignments_two, designWeight, Fin.prod_univ_succ, Fin.sum_univ_succ,
    bit, coin]

-- Genuine defined intervals and nonzero residuals for the HAC algebra.
example :
    let z : Assignment 4 := ![true,true,false,false]
    let R : CircleTable 4 := fun i _ => if i = 2 then 1 else if i = 3 then -1 else 0
    ValidInterval R (1 : Matrix (Fin 4) (Fin 4) ℝ) z ∧
      kernelVariance z (residual R z) (1 : Matrix (Fin 4) (Fin 4) ℝ) = 1/2 := by
  dsimp
  have hcount (a : Bool) : armCount (![true,true,false,false] : Assignment 4) a = 2 := by
    cases a <;>
      simp only [armCount, Fin.sum_univ_succ, Finset.sum_empty, Finset.sum_singleton,
        Matrix.cons_val_zero, Matrix.cons_val_succ, Fin.isValue]
      <;> norm_num
  norm_num [ValidInterval, ValidArms, kernelVariance, lambda,
    SAP.P004b.residual, armMean, Fin.sum_univ_succ, bit, Matrix.one_apply, hcount]

-- The kernel hypotheses admit the intended uniform kernel.
example : KernelConditions uniformKernel := by
  refine ⟨by norm_num [uniformKernel], ?_, ⟨2, ?_⟩, ?_⟩
  · intro t ht
    simp [uniformKernel, not_le.mpr ht]
  · intro t ht
    by_cases h : t ≤ 1 <;> simp [uniformKernel, ht, h]
  · refine ⟨1/2, 1, by norm_num, by norm_num, by norm_num, ?_⟩
    intro t ht hlt
    have ht1 : t ≤ 1 := by linarith
    simp [uniformKernel, ht, ht1]
