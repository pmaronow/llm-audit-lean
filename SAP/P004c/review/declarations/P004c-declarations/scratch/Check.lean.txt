    (forall t : Real, 0 <= t -> abs (if t <= 1 then (1 : Real) else 0) <= 1)
    (forall t : Real, 1 < t -> (if t <= 1 then (1 : Real) else 0) = 0) := by
  constructor
  case left =>
    intro t ht
    split_ifs <;> norm_num
  case right =>
    intro t ht
    simp [not_le.mpr ht]
