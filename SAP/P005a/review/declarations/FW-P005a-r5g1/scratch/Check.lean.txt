-- These examples use definitions and Mathlib, never the five audited theorems.
example : 4 ≤ (6 : ℕ) ∧ 0 < (4 : ℕ) ∧ (4 : ℕ) < 6 ∧
    (design 6 4).card = 15 := by
  norm_num [design, Finset.card_powersetCard]
  decide

example : popMean witnessA = -(3/2 : ℝ) ∧
    popMean witnessB = (5/6 : ℝ) ∧
    popCov witnessA witnessB = -(23/12 : ℝ) := by
  norm_num [popMean, mean, popCov, sampleCov, witnessA, witnessB,
    Fin.sum_univ_succ]

example : standardized signPopulation ∧
    popMean (fun i => signPopulation.yT i^2) ≤ 28 ∧
    popMean (fun i => signPopulation.yC i^2) ≤ 28 ∧
    (1/4 : ℝ) < pT 6 2 ∧ pT 6 2 < (3/4 : ℝ) := by
  norm_num [standardized, xMean, SAP.P005a.gram, popMean, mean, signPopulation, signX,
    pT, Fin.sum_univ_succ, Matrix.one_apply, funext_iff]
  ext k l
  fin_cases k
  fin_cases l
  norm_num [SAP.P005a.gram, popMean, mean, signPopulation, signX,
    Fin.sum_univ_succ, Matrix.one_apply]

example : Function.Injective signX := by

-- The universal T2 hypotheses have a positive, nonconstant instance.
example : ∃ (n m : ℕ) (a b : Fin n → ℝ), 4 ≤ n ∧ 0 < m ∧ m < n ∧
    popMean a ≠ 0 ∧ popMean b ≠ 0 := by
  refine ⟨6, 4, witnessA, witnessB, by norm_num, by norm_num, by norm_num, ?_, ?_⟩ <;>
    norm_num [popMean, mean, witnessA, witnessB, Fin.sum_univ_succ]

-- A finite, positive-dimensional T4 instance, with an actual moment bound.
def checkPopulation : Population 4 1 where
  x := fun i _ => (![-1, -1, 1, 1] : Fin 4 → ℝ) i
  yT := fun i => (i.val : ℝ)
  yC := fun i => -(i.val : ℝ)

example : 4 ≤ (4 : ℕ) ∧ 0 < (2 : ℕ) ∧ (2 : ℕ) < 4 ∧
    standardized checkPopulation ∧
    popMean (fun i => checkPopulation.yT i^2) ≤ 4 ∧
    popMean (fun i => checkPopulation.yC i^2) ≤ 4 := by
  norm_num [standardized, xMean, SAP.P005a.gram, popMean, mean, checkPopulation,
    Fin.sum_univ_succ, Matrix.one_apply, funext_iff]
  ext k l
  fin_cases k
  fin_cases l
  norm_num [SAP.P005a.gram, popMean, mean, checkPopulation,
    Fin.sum_univ_succ, Matrix.one_apply]

example : standardized leveragePopulation := by
  constructor
  · ext k
    fin_cases k
    norm_num [xMean, mean, leveragePopulation, scalarPopulation, leverageA,
      Fin.sum_univ_succ]
  · ext k l
    fin_cases k
    fin_cases l
    change popMean (fun i =>
      (Real.sqrt (8/71) * leverageA i) * (Real.sqrt (8/71) * leverageA i)) = 1
    generalize hs : Real.sqrt (8/71 : ℝ) = c
    have hsq : c^2 = (8/71 : ℝ) := by
      rw [← hs]
      exact Real.sq_sqrt (by norm_num)
    norm_num [popMean, mean, leverageA, Fin.sum_univ_succ]
    nlinarith [hsq]

-- An entire sequence satisfying every T4 sequence hypothesis; zero dimension
-- is explicitly in scope, while the finite example above has dimension one.
def checkZeroSequence (n : ℕ) : Population n 0 where
  x := fun _ _ => 0
  yT := fun _ => 0
  yC := fun _ => 0

example : W1SequenceAssumptions (fun _ => 0) (fun n => n/2)
    checkZeroSequence 1 (1/4) (3/4) := by
  refine ⟨by norm_num, by norm_num, by norm_num, by norm_num, ?_, ?_⟩
  · intro n hn
    dsimp only
    have hnR : (0 : ℝ) < n := by exact_mod_cast (show 0 < n by omega)
    have hlo : n < 4*(n/2) := by omega
    have hhi : 4*(n/2) < 3*n := by omega
    refine ⟨by omega, by omega, by omega, ?_, ?_, ?_, ?_, ?_⟩
    · norm_num [popMean, mean, checkZeroSequence]
    · norm_num [popMean, mean, checkZeroSequence]
    · intro k
      exact Fin.elim0 k
    · rw [pT, lt_div_iff₀ hnR]
      have h : (n : ℝ) < 4*(n/2 : ℕ) := by exact_mod_cast hlo
      linarith
    · rw [pT, div_lt_iff₀ hnR]
      have h : 4*(n/2 : ℕ) < 3*(n : ℝ) := by exact_mod_cast hhi
      linarith
  · exact Filter.Eventually.of_forall (fun n => by
      constructor
      · ext k
        exact Fin.elim0 k
      · ext k l
        exact Fin.elim0 k)
