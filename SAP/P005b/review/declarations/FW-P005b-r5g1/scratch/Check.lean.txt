-- T15: both covariance terms and both third-moment coefficients are nonzero.
def p15 : Population 5 1 where
  x := fun i _ => ![1, 0, 0, 0, 2] i
  yT := ![2, 0, 0, 0, 0]
  yC := ![0, 0, 1, 2, 3]

example : 2 ≤ (2 : ℕ) ∧ 2 ≤ (3 : ℕ) ∧
    ({0, 1} : Finset (Fin 5)) ∈ design 5 2 := by decide

example : sampleCov {0, 1} (leverage p15) p15.yT = 1 / 2 := by
  norm_num [sampleCov, mean, leverage, dot, p15, Fin.sum_univ_succ]

example : sampleCov (control {0, 1}) (leverage p15) p15.yC = 4 / 3 := by
  rw [show SAP.P005b.control ({0, 1} : Finset (Fin 5)) = {2, 3, 4} from by decide]
  norm_num [sampleCov, mean, leverage, dot, p15, Fin.sum_univ_succ]

example : covarianceCombination 2 3 (1/2) (4/3) = 5/18 ∧
    controlCoefficient 2 3 = 1/3 ∧ treatedCoefficient 2 3 = -1/3 := by
  norm_num [covarianceCombination, controlCoefficient, treatedCoefficient, NTTT, NTTC]

-- T17 and T18: positive dimension, a genuine standardized Gram matrix,
-- nonzero residual differences, and nonzero actual residual star.
def p17 : Population 4 1 where
  x := fun i _ => ![1, -1, 1, -1] i
  yT := ![1, 0, 0, 0]
  yC := 0

lemma p17_standardized : standardized p17 := by
  constructor
  · funext k
    fin_cases k
    norm_num [xMean, mean, p17, Fin.sum_univ_succ]
  · ext k l
    fin_cases k
    fin_cases l
    norm_num [gram, popMean, mean, p17, Fin.sum_univ_succ]

example : 2 ≤ (4 : ℕ) ∧ 0 < (1 : ℕ) ∧ (1 : ℕ) < 4 ∧
    standardized p17 ∧ ({0} : Finset (Fin 4)) ∈ design 4 1 ∧
    popMean (fun i => p17.yT i ^ 2) ≤ 1 ∧
    popMean (fun i => p17.yC i ^ 2) ≤ 1 := by
  refine ⟨by norm_num, by norm_num, by norm_num, p17_standardized,
    by decide, ?_, ?_⟩
  all_goals norm_num [popMean, mean, p17, Fin.sum_univ_succ]

example : wholeW1Star p17 1 {0} = 1/6 := by

    observedResidual, commonResidual, centered, popMean, N, pT, pC, p17,

  funext k
  fin_cases k
  norm_num [populationSlope, commonResidual, centered, popMean, mean,
    dot, N, pT, pC, p17, Fin.sum_univ_succ]

-- T19 printed reading: all fixed uncentered vectors; n>=4; nonempty arms;
-- uniform complete randomization; either arm; both the mean identity and
-- the RAW-product variance bound asserted. This witness checks every condition.
-- The product a_i*b_i is the constant 20, so its population variance is zero.
def a19 : Fin 6 → ℝ := ![20, 1, 1, 1, 1, 1]
def b19 : Fin 6 → ℝ := ![1, 20, 20, 20, 20, 20]

lemma designSixFive : design 6 5 =
    {{0, 1, 2, 3, 4}, {0, 1, 2, 3, 5}, {0, 1, 2, 4, 5},
     {0, 1, 3, 4, 5}, {0, 2, 3, 4, 5}, {1, 2, 3, 4, 5}} := by decide

example : 4 ≤ (6 : ℕ) ∧ 0 < (5 : ℕ) ∧ (5 : ℕ) < 6 ∧
    (design 6 5).card = 6 ∧ armSize 6 5 .treated = 5 := by
  decide

example : ∀ i, a19 i * b19 i = 20 := by
  intro i
  fin_cases i <;> norm_num [a19, b19]

lemma t19_mean : expectation 5 (fun S => sampleCov S a19 b19) = -722/15 := by
  simp (disch := decide) only [expectation, designSixFive, Finset.sum_insert,
    Finset.sum_singleton, Finset.card_insert_of_notMem, Finset.card_singleton]
  norm_num [sampleCov, mean, a19, b19]

lemma t19_variance : variance 5 (fun S => sampleCov S a19 b19) = 521284/1125 := by
  unfold variance
  rw [t19_mean]
  simp (disch := decide) only [expectation, designSixFive, Finset.sum_insert,
    Finset.sum_singleton, Finset.card_insert_of_notMem, Finset.card_singleton]
  norm_num [sampleCov, mean, a19, b19]

lemma t19_bound : covarianceBound 5 a19 b19 = 54027/125 := by
  norm_num [covarianceBound, popMean, mean, a19, b19, Fin.sum_univ_succ]

example : expectation 5 (fun S => sampleCov (armSet .treated S) a19 b19) =
    ((armSize 6 5 .treated : ℝ)-1)*6/((armSize 6 5 .treated : ℝ)*(6-1))*popCov a19 b19 := by
  simp only [armSet]
  rw [t19_mean]
  norm_num [armSize, popCov, sampleCov, popMean, mean, a19, b19, Fin.sum_univ_succ]

example : variance 5 (fun S => sampleCov (armSet .treated S) a19 b19) >
    covarianceBound (armSize 6 5 .treated) a19 b19 := by
  simp only [armSet, armSize]
  rw [t19_variance, t19_bound]
  norm_num

end
