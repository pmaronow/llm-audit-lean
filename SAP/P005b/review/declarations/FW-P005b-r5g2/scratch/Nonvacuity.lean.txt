import SAP.P005b.Defs.WholeW1

open scoped BigOperators Matrix
open Finset
open SAP.P005b

namespace Review

noncomputable section

def xs : Fin 5 → ℝ := ![-3/2, -1/2, 0, 1/2, 3/2]
def P : Population 5 1 :=
  { x := fun i _ => xs i, yT := fun i => (xs i)^2, yC := fun _ => 0 }

theorem standardP : standardized P := by
  constructor
  · ext k
    fin_cases k
    norm_num [xMean, popMean, mean, P, xs, Fin.sum_univ_succ]
  · ext k l
    fin_cases k
    fin_cases l
    norm_num [gram, popMean, mean, P, xs, Fin.sum_univ_succ]

theorem interactedP : interactedRegular P 2 := by
  classical
  intro S hS
  fin_cases S <;> simp [design] at hS <;>
    norm_num [SAP.P005b.control, Finset.card_sdiff, Finset.eq_univ_iff_forall,
      Fin.forall_fin_succ, armL, xMean, mean, P, xs,
      Matrix.det_fin_one, Fin.sum_univ_succ]

theorem noninteractedP : noninteractedRegular P 2 := by
  classical
  intro S hS
  fin_cases S <;> simp [design] at hS <;>
    norm_num [SAP.P005b.control, Finset.card_sdiff, Lhat, pT, pC, xMean, mean, P, xs,
      Matrix.det_fin_one, Matrix.vecMulVec, Fin.sum_univ_succ]

example : 3 ≤ (5 : ℕ) ∧ 0 < (2 : ℕ) ∧ (2 : ℕ) < 5 ∧
    standardized P ∧ interactedRegular P 2 ∧ noninteractedRegular P 2 ∧
    2 ≤ (2 : ℕ) ∧ 2 ≤ (5 - 2 : ℕ) := by
  exact ⟨by norm_num, by norm_num, by norm_num, standardP, interactedP,
    noninteractedP, by norm_num, by norm_num⟩

def assignments : Finset (Finset (Fin 5)) :=
  {{0,1}, {0,2}, {0,3}, {0,4}, {1,2}, {1,3}, {1,4}, {2,3}, {2,4}, {3,4}}

theorem design5 : design 5 2 = assignments := by decide

theorem sum_assignments (f : Finset (Fin 5) → ℝ) :
    ∑ S ∈ assignments, f S =
      f {0,1} + f {0,2} + f {0,3} + f {0,4} + f {1,2} +
      f {1,3} + f {1,4} + f {2,3} + f {2,4} + f {3,4} := by
  classical
  simp [assignments, Finset.sum_insert, Finset.ext_iff, Fin.forall_fin_succ, add_assoc]

theorem card_assignments : assignments.card = 10 := by decide

example : (design 5 2).card = 10 := by decide

example : leverageCov P P.yT = 21/20 := by
  norm_num [leverageCov, popCov, sampleCov, popMean, mean, leverage,
    dot, P, xs, Fin.sum_univ_succ]

theorem actualThird : expectation 2 (fun S => dot (xMean P (SAP.P005b.control S))
      (armNu3 P (SAP.P005b.control S) P.yC)) -
    expectation 2 (fun S => dot (xMean P S) (armNu3 P S P.yT)) = 21/320 := by
  norm_num [expectation, design5, sum_assignments, card_assignments,
    SAP.P005b.control, Finset.card_sdiff, dot,
    xMean, armNu3, centered, popMean, mean, P, xs, Fin.sum_univ_succ]

theorem printedThird : NCCC 5 2 * leverageCov P P.yC - NTTT 5 2 * leverageCov P P.yT = -21/320 := by
  norm_num [NCCC, NTTT, leverageCov, popCov, sampleCov, popMean, mean,
    leverage, dot, P, xs, Fin.sum_univ_succ]

def weakP : Population 5 1 := { P with x := fun i _ => 2 * xs i }

example : xMean weakP univ = 0 := by
  ext k
  fin_cases k
  norm_num [xMean, mean, weakP, xs, Fin.sum_univ_succ]

example : ¬ standardized weakP := by
  intro h
  have h00 := congrFun (congrFun h.2 (0 : Fin 1)) (0 : Fin 1)
  norm_num [gram, popMean, mean, weakP, xs, Fin.sum_univ_succ] at h00

-- This is a witness to the exact negated finite assertion in T20.
example : ∃ Q : Population 5 1, standardized Q ∧ interactedRegular Q 2 ∧
    expectation 2 (fun S => dot (xMean Q (SAP.P005b.control S))
      (armNu3 Q (SAP.P005b.control S) Q.yC)) -
    expectation 2 (fun S => dot (xMean Q S) (armNu3 Q S Q.yT)) ≠
      NCCC 5 2 * leverageCov Q Q.yC - NTTT 5 2 * leverageCov Q Q.yT := by
  refine ⟨P, standardP, interactedP, ?_⟩
  rw [actualThird, printedThird]
  norm_num

-- The noninteracted main hypotheses admit a singleton treatment arm.
example : noninteractedRegular P 1 := by
  classical
  intro S hS
  fin_cases S <;> simp [design] at hS <;>
    norm_num [SAP.P005b.control, Finset.card_sdiff, Lhat, pT, pC,
      xMean, mean, P, xs, Matrix.det_fin_one, Matrix.vecMulVec, Fin.sum_univ_succ]

-- A positive-dimensional balanced, additive-effect model exercises both
-- local antecedents of Remark 1 in T23.
def xb : Fin 4 → ℝ := ![-7/5, -1/5, 1/5, 7/5]
def balancedP : Population 4 1 :=
  { x := fun i _ => xb i,
    yT := fun i => (xb i)^2 + 7, yC := fun i => (xb i)^2 }

example : standardized balancedP ∧ noninteractedRegular balancedP 2 ∧
    (2 : ℕ) = 4 - 2 ∧ (∃ c : ℝ, ∀ i, balancedP.yT i - balancedP.yC i = c) := by
  refine ⟨?_, ?_, by norm_num, 7, ?_⟩
  · constructor
    · ext k
      fin_cases k
      norm_num [xMean, mean, balancedP, xb, Fin.sum_univ_succ]
    · ext k l
      fin_cases k
      fin_cases l
      norm_num [gram, popMean, mean, balancedP, xb, Fin.sum_univ_succ]
  · classical
    intro S hS
    fin_cases S <;> simp [design] at hS <;>
      norm_num [SAP.P005b.control, Finset.card_sdiff, Lhat, pT, pC,
        xMean, mean, balancedP, xb, Matrix.det_fin_one, Matrix.vecMulVec,
        Fin.sum_univ_succ]
  · intro i
    simp [balancedP]

-- The n=2 edge of T24 is a legitimate no-covariate model.
def emptyP : Population 2 0 :=
  { x := fun _ k => Fin.elim0 k, yT := ![1, 3], yC := ![0, 1] }

example : standardized emptyP ∧ interactedRegular emptyP 1 := by
  constructor
  · constructor
    · ext k
      exact Fin.elim0 k
    · ext k
      exact Fin.elim0 k
  · intro S hS
    simp [interactedRegular, Matrix.det_isEmpty]

end
end Review
