import SAP.P005b.Challenge

set_option autoImplicit false
set_option relaxedAutoImplicit false

open scoped BigOperators Matrix
open Finset Filter
open SAP.P005b

namespace Audit

noncomputable section

-- Rational, standardized, distinct covariates; nonconstant potential outcomes.
def finiteP : Population 4 1 where
  x := fun i _ => ![-(7/5 : ℝ), -(1/5 : ℝ), (1/5 : ℝ), (7/5 : ℝ)] i
  yT := fun i => (i.val : ℝ)^2
  yC := fun i => (i.val : ℝ)

example : standardized finiteP := by
  constructor
  · ext k
    fin_cases k
    norm_num [xMean, mean, finiteP, Fin.sum_univ_succ]
  · ext k l
    fin_cases k
    fin_cases l
    norm_num [gram, popMean, mean, finiteP, Fin.sum_univ_succ]

example : noninteractedRegular finiteP 2 := by
  intro S hS
  have hu : (univ : Finset (Fin 4)) = {0,1,2,3} := by decide
  have hd : design 4 2 = {{0,1}, {0,2}, {0,3}, {1,2}, {1,3}, {2,3}} := by decide
  rw [hd] at hS
  simp only [Finset.mem_insert, Finset.mem_singleton] at hS
  rcases hS with rfl | rfl | rfl | rfl | rfl | rfl <;>
    norm_num [Lhat, pT, pC, xMean, mean, SAP.P005b.control, finiteP,
      Matrix.det_fin_one, Matrix.vecMulVec, hu, Fin.sum_univ_succ,
      Finset.sdiff_eq_filter, Finset.filter_insert, Finset.filter_singleton]

example : interactedRegular finiteP 2 := by
  intro S hS
  have hu : (univ : Finset (Fin 4)) = {0,1,2,3} := by decide
  have hd : design 4 2 = {{0,1}, {0,2}, {0,3}, {1,2}, {1,3}, {2,3}} := by decide
  rw [hd] at hS
  simp only [Finset.mem_insert, Finset.mem_singleton] at hS
  rcases hS with rfl | rfl | rfl | rfl | rfl | rfl <;>
    norm_num [armL, xMean, mean, SAP.P005b.control, finiteP,
      Matrix.det_fin_one, hu, Fin.sum_univ_succ,
      Finset.sdiff_eq_filter, Finset.filter_insert, Finset.filter_singleton]

example : 3 ≤ (4 : ℕ) ∧ 1 ≤ (1 : ℕ) ∧ (1 : ℕ) ≤ 4 ∧
    0 < (2 : ℕ) ∧ (2 : ℕ) < 4 ∧
    (1/4 : ℝ) < pT 4 2 ∧ pT 4 2 < (3/4 : ℝ) ∧
    popMean (fun i => (finiteP.yT i)^2) ≤ 100 ∧
    popMean (fun i => (finiteP.yC i)^2) ≤ 100 ∧
    (∀ k, popMean (fun i => (finiteP.x i k)^2) ≤ 100) := by
  norm_num [pT, popMean, mean, finiteP, Fin.sum_univ_succ]

end
end Audit
