-- T10: one-unit ATO integrates π(c)y1, pointwise even before moment assumptions.
lemma review_oneATO {C : Type} [MeasurableSpace C]
    (D : OutcomeLaw C) (π : C → ℝ) :
    oneATOExpectation D π = ∫ z, π z.1 * z.2.2 ∂D := by
  simp [oneATOExpectation, wATO, treatedCount, coinMass, bit]

-- T10: full treated-data invariance; no finite-support restriction.
example {C : Type} [MeasurableSpace C]
    (D D' : OutcomeLaw C) (π : C → ℝ) (hπ : Admissible π)
    (hD : Measure.map treatedProjection D = Measure.map treatedProjection D') :
    oneATOExpectation D π = oneATOExpectation D' π := by
  let f : C × ℝ → ℝ := fun z => π z.1 * z.2
  have hp : Measurable (treatedProjection : C × (ℝ × ℝ) → C × ℝ) := by
    unfold treatedProjection
    fun_prop
  have hf : StronglyMeasurable f := by
    apply Measurable.stronglyMeasurable
    exact (hπ.1.comp measurable_fst).mul measurable_snd
  rw [review_oneATO, review_oneATO]
  change (∫ z, f (treatedProjection z) ∂D) = ∫ z, f (treatedProjection z) ∂D'
  rw [← integral_map_of_stronglyMeasurable hp hf,
    ← integral_map_of_stronglyMeasurable hp hf, hD]

-- T10: every witness condition, with arbitrary B>0, and no junk supremum.
example (B : ℝ) (hB : 0 < B) :
    let D : OutcomeLaw Unit := Measure.dirac ((), (1+B, 1))
    let F : Set (Unit → ℝ) := {fun _ => 1}
    FiniteOutcomeLaw D ∧ Admissible (fun _ : Unit => (1 : ℝ)) ∧
    IsBest F (oneATOExpectation D) (fun _ => 1) ∧
    (∫ z, (1 : ℝ) ∂D) = 1 ∧
    outcomeWelfare D (fun _ => 1) = -B ∧
    bestValue F (outcomeWelfare D) = -B ∧
    (∀ V : (Unit → ℝ) → ℝ, regret F V (fun _ => 1) = 0) ∧
    (fun _ : Unit => (0 : ℝ)) ∉ F ∧ ¬ (0 ≥ B) := by
  have hD : FiniteOutcomeLaw (Measure.dirac ((), (1+B, (1 : ℝ)))) :=
    ⟨inferInstance, integrable_dirac (by finiteness), integrable_dirac (by finiteness)⟩
  simp [hD, Admissible, IsBest, outcomeWelfare, bestValue,
    regret, not_le.mpr hB]
  intro h
  have h01 := congrFun h ()
  norm_num at h01
