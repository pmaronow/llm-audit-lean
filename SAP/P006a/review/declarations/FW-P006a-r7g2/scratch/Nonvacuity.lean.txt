def propensity {C : Type} (c : C) (π : C → ℝ) : unitInterval :=
  if h : 0 ≤ π c ∧ π c ≤ 1 then ⟨π c, h⟩ else 0

lemma propensity_eq {C : Type} [MeasurableSpace C] (c : C) (π : C → ℝ)
    (hπ : Admissible π) : (propensity c π : ℝ) = π c := by
  simp [propensity, hπ.2 c]

/-- One genuine Bernoulli treatment draw, constant positive treatment effect,
    and a deterministic oracle baseline. The law is defined even off the
    admissible-policy domain, where it is not used. -/
def experiment {C : Type} [MeasurableSpace C] (c : C) : Experiment Bool C 1 where
  hn := by decide
  PX := Measure.dirac c
  μ0 := fun _ => 0
  μ1 := fun _ => 1
  law := fun π => bernoulliMeasure true false (propensity c π)
  X := fun _ _ => c
  Y0 := fun _ _ => 0
  Y1 := fun _ _ => 1
  T := fun _ _ ω => ω
  μhat := fun _ _ _ => 0

lemma design {C : Type} [MeasurableSpace C] [MeasurableSingletonClass C]
    (c : C) (π : C → ℝ) (hπ : Admissible π) : PolicyDesign (experiment c) π := by
  refine {
    px_probability := by dsimp [experiment]; infer_instance
    means_measurable := ⟨measurable_const, measurable_const⟩
    means_integrable := ⟨by simp [experiment], by simp [experiment]⟩
    law_probability := by dsimp [experiment]; infer_instance
    measurable_units := fun _ => ⟨measurable_const, measurable_const,
      measurable_const, measurable_id⟩
    integrable_units := fun _ => ⟨by apply integrable_bernoulliMeasure,
      by apply integrable_bernoulliMeasure⟩
    covariate_law := ?_
    conditional_means := ?_
    bernoulli := ?_
    ignorability := ?_ }
  · intro i
    refine ⟨measurable_const.aemeasurable, measurable_id.aemeasurable, ?_⟩
    simp [experiment, Measure.map_const]
  · intro i
    simp [experiment, Experiment.covariateSigma, condExp_const]
  · intro i
    apply Filter.Eventually.of_forall
    intro ω
    simp [experiment, Experiment.covariateSigma, condExp_bot,
      integral_bernoulliMeasure, bit, propensity_eq c π hπ]
  · intro i B hB
    apply Filter.Eventually.of_forall
    intro ω
    simp [experiment, Experiment.covariateSigma, condExp_bot,
      integral_bernoulliMeasure, bit, propensity_eq c π hπ]

lemma standing {C : Type} [MeasurableSpace C] [MeasurableSingletonClass C]
    (c : C) : Standing (experiment c) := by
  refine {
    px_probability := by dsimp [experiment]; infer_instance
    means_measurable := ⟨measurable_const, measurable_const⟩
    means_integrable := ⟨by simp [experiment], by simp [experiment]⟩
    law_probability := fun π hπ => (design c π hπ).law_probability
    measurable_units := fun π hπ => (design c π hπ).measurable_units
    integrable_units := ?_
    covariate_law := fun π hπ => (design c π hπ).covariate_law
    potential_law_fixed := ?_
    iid_independent := ?_
    iid_identical := ?_
    conditional_means := fun π hπ => (design c π hπ).conditional_means
    bernoulli := fun π hπ => (design c π hπ).bernoulli
    ignorability := fun π hπ => (design c π hπ).ignorability }
  · intro π hπ i
    exact ⟨(design c π hπ).integrable_units i |>.1,
      (design c π hπ).integrable_units i |>.2, by simp [experiment]⟩
  · intro π π' hπ hπ' i
    dsimp [experiment]
    refine ⟨measurable_const.aemeasurable, measurable_const.aemeasurable, ?_⟩
    simp [experiment, Measure.map_const]
  · intro π hπ
    dsimp [experiment]
    exact iIndepFun.of_subsingleton
  · intro π hπ i j
    have hij : i = j := Subsingleton.elim i j
    subst j
    apply IdentDistrib.refl
    fun_prop

lemma residual_integrable {C : Type} [MeasurableSpace C] (c : C)
    (π : C → ℝ) (i : Fin 1) :
    Integrable (fun ω => bit ((experiment c).T π i ω) *
      ((experiment c).μhat π ω ((experiment c).X i ω) -
        (experiment c).μ0 ((experiment c).X i ω))) ((experiment c).law π) := by
  simp [experiment]

lemma treated_unbiased {C : Type} [MeasurableSpace C] (c : C)
    (F : Set (C → ℝ)) : TreatedUnbiased (experiment c) (experiment c).μ0 F := by
  intro π hπ i
  simp [experiment]

lemma pointwise_unbiased {C : Type} [MeasurableSpace C] (c : C)

lemma independent_estimator {C : Type} [MeasurableSpace C] (c : C)
    (F : Set (C → ℝ)) : IndependentEstimator (experiment c) F := by
  refine ⟨Unit, inferInstance, (fun _ => ()), (fun _ _ _ => 0),
    measurable_const, ?_⟩
  intro π hπ
  refine ⟨measurable_const, (fun _ _ => rfl), ?_⟩
  dsimp [experiment]
  exact indepFun_const_left _ _

-- A nonempty feasible class with both an inferior and a superior policy.
example : ∃ E : Experiment Bool Unit 1,
    let F := {π | Admissible π}
    (fun _ => (0 : ℝ)) ∈ F ∧ (fun _ => (1 : ℝ)) ∈ F ∧
    (∀ π ∈ F, PolicyDesign E π) ∧
    (∀ π ∈ F, ∀ i, Integrable (fun ω => bit (E.T π i ω) *
      (E.μhat π ω (E.X i ω) - E.μ0 (E.X i ω))) (E.law π)) ∧
    TreatedUnbiased E E.μ0 F ∧ PointwiseUnbiased E F ∧
    IndependentEstimator E F := by
  refine ⟨experiment (), ?_⟩
  exact ⟨⟨measurable_const, fun _ => by norm_num⟩,
    ⟨measurable_const, fun _ => by norm_num⟩,
    fun π hπ => design () π hπ,
    fun π _ i => residual_integrable () π i,
    treated_unbiased () _, pointwise_unbiased () _, independent_estimator () _⟩

-- Every hypothesis of T7, including design and moments for every admissible π.
example : ∃ (E : Experiment Bool Unit 1) (g : Unit → ℝ),
    Standing ({E with μhat := fun _ _ _ => 0} : Experiment Bool Unit 1) ∧
    Measurable g ∧ (∀ x, 0 < g x) ∧
    (∀ π, Admissible π → ∀ i, Integrable (fun ω => bit (E.T π i ω) *
      (E.μhat π ω (E.X i ω) - E.μ0 (E.X i ω))) (E.law π)) ∧
    TreatedUnbiased E E.μ0 {π | Admissible π} ∧
    Integrable (fun x => g x * E.τ x) E.PX ∧
    (∀ π, Admissible π → ∀ i, Integrable (fun ω =>
      (E.observed π ω i - E.μhat π ω (E.X i ω)) * bit (E.T π i ω) *
        g (E.X i ω)) (E.law π)) := by
  refine ⟨experiment (), fun _ => 1, ?_, measurable_const,
    (fun _ => by norm_num), (fun π _ i => residual_integrable () π i),
    treated_unbiased () _, ?_, ?_⟩
  · simpa [experiment] using standing ()
  · simp [experiment, Experiment.τ]
  · intro π hπ i
    apply integrable_bernoulliMeasure

-- All hypotheses of T8's main information bound, with a nonempty feasible set.
example : ∃ (E : Experiment Bool (Unit × Unit) 1) (b0 : Unit → ℝ)
    (F : Set ((Unit × Unit) → ℝ)),

      (TreatedResidualW2 E (fun c => b0 c.1) π i) (E.law π)) ∧

    (∃ b s, IsBest F E.expectedTT b ∧ IsBest F E.V s) := by

    (∀ π, Admissible π → PolicyDesign E π ∧ CoinMomentW2 E π) ∧
    (∀ π, Admissible π → ∀ i,

        id (E.law π) D) ∧
    Integrable (fun z => |z.2 - E.μ0 z.1|) D := by
  let c : Unit × Unit := ((), ())
  refine ⟨experiment c, Measure.dirac (c, (0 : ℝ)), inferInstance, ?_, ?_, ?_⟩
  · intro π hπ

  · intro π hπ i

-- A nonzero selected measure, with every hypothesis of SelectedEquivalence
-- and CoinFactorization (including optional full integrability and conditional
-- independence). Bool is a standard Borel space.
example : ∃ (μ : Measure Bool) (hμ : IsProbabilityMeasure μ),
    letI : IsProbabilityMeasure μ := hμ
    (μ {true} > 0) ∧
    Measurable (id : Bool → Bool) ∧
    Measurable (fun _ : Bool => (0 : ℝ)) ∧
    Measurable[⊥] (fun _ : Bool => (0 : ℝ)) ∧
    Integrable (fun _ : Bool => (0 : ℝ)) μ ∧
    Measurable[⊥] (fun _ : Bool => (1/2 : ℝ)) ∧
    (∀ _ : Bool, 0 ≤ (1/2 : ℝ) ∧ (1/2 : ℝ) ≤ 1) ∧
    (μ[(fun ω => bit ω) | ⊥] =ᵐ[μ] (fun _ => (1/2 : ℝ))) ∧
    Integrable (fun ω => bit ω * ((0 : ℝ) - 0)) μ ∧
    Integrable (fun ω => bit ω * (0 : ℝ)) μ ∧
    Integrable (fun _ : Bool => (1/2 : ℝ) * 0) μ ∧
    CondIndepFun ⊥ bot_le (id : Bool → Bool) (fun _ => (0 : ℝ)) μ := by
  let p : unitInterval := ⟨1/2, by constructor <;> norm_num⟩
  let μ := bernoulliMeasure true false p
  refine ⟨μ, inferInstance, ?_, measurable_id, measurable_const,
    measurable_const, by simp, measurable_const, (fun _ => by norm_num), ?_,
    by simp, by simp, by simp, ?_⟩
  · norm_num [μ, bernoulliMeasure_apply, p, unitInterval.toNNReal]
    apply ENNReal.coe_pos.mpr
    change (0 : ℝ) < 1/2
    norm_num
  · apply Filter.Eventually.of_forall
    intro ω
    simp [μ, p, condExp_bot, integral_bernoulliMeasure, bit]
  · exact condIndepFun_const_right _ _
