lemma card_rows_lt (n k : ℕ) (hk : k < n) :
    (Finset.univ.filter (fun i : Fin n => i.val < k)).card = k := by
  have hf : Finset.univ.filter (fun i : Fin n => i.val < k) =
      Finset.Iio (⟨k, hk⟩ : Fin n) := by
    ext i
    simp [Fin.lt_def]
  rw [hf, Fin.card_Iio]

lemma card_rows_not_lt (n k : ℕ) (hk : k < n) :
    (Finset.univ.filter (fun i : Fin n => ¬i.val < k)).card = n - k := by
  have hh := Finset.card_filter_add_card_filter_not (s := (Finset.univ : Finset (Fin n)))
    (fun i => i.val < k)
  rw [card_rows_lt n k hk, Finset.card_univ, Fintype.card_fin] at hh
  omega

noncomputable def countArray : ObservedArray 8580 1205 :=
  fun i => Finset.Iio (if i.val < 7633 then (406 : Fin 1205) else 405)

lemma countArray_card (i : Fin 8580) : (countArray i).card = if i.val < 7633 then 406 else 405 := by
  simp only [countArray, Fin.card_Iio]
  split <;> rfl

noncomputable def countMask : Fin 8580 → (Fin 1205 ⊕ Fin 19) → Bool :=
  fun i j => match j with
    | Sum.inl j => decide (j ∉ countArray i)
    | Sum.inr _ => true

lemma missingAverage_count {n : ℕ} {J : Type*} [Fintype J]
    (m : Fin n → J → Bool) :
    missingAverage m = (missingCount m : ℝ) / ((n : ℝ) * Fintype.card J) := by
  have hc : (missingCount m : ℝ) = ∑ j, ∑ i, if m i j then (1 : ℝ) else 0 := by
    unfold missingCount
    rw [Nat.cast_sum]
    apply Finset.sum_congr rfl
    intro j _
    rw [Nat.cast_sum]
    apply Finset.sum_congr rfl
    intro i _
    split <;> norm_num
  unfold missingAverage
  rw [← Finset.sum_div, div_div, hc]

lemma missing_from_set (K : ℕ) (s : Finset (Fin K)) :
    (∑ j : Fin K, if decide (j ∉ s) then (1:ℕ) else 0) = K - s.card := by
  simp only [decide_eq_true_eq]
  rw [Finset.sum_boole]
  have he : Finset.univ.filter (fun j => j ∉ s) = Finset.univ \ s := by
    ext j
    simp
  rw [he, Finset.card_sdiff]
  simp only [Finset.inter_univ, Finset.card_univ, Fintype.card_fin, Nat.cast_id]

lemma retained_missing_per_row (i : Fin 8580) :
    (∑ j : Fin 1205, if countMask i (Sum.inl j) then 1 else 0 : ℕ) =
      if i.val < 7633 then 799 else 800 := by
  change (∑ j : Fin 1205, if decide (j ∉ countArray i) then (1:ℕ) else 0) = _
  rw [missing_from_set, countArray_card]
  split <;> norm_num

lemma sum_rows_if (n t a b : ℕ) (ht : t < n) :
    (∑ i : Fin n, if i.val < t then a else b) = t*a+(n-t)*b := by
  rw [Finset.sum_ite]
  simp only [Finset.sum_const, card_rows_lt n t ht, card_rows_not_lt n t ht,
    nsmul_eq_mul, Nat.cast_id]

lemma counts_from_rows {n K d : ℕ} (m : Fin n → (Fin K ⊕ Fin d) → Bool)
    (t a b : ℕ) (ht : t < n)
    (hr : ∀ i, (∑ j : Fin K, if m i (Sum.inl j) then (1:ℕ) else 0) =
      if i.val < t then a else b)
    (hd : ∀ i j, m i (Sum.inr j) = true) :
    missingCount m = t*a+(n-t)*b+d*n := by
  unfold missingCount
  rw [Fintype.sum_sum_type]
  rw [Finset.sum_comm (f := fun (j : Fin K) (i : Fin n) => if m i (Sum.inl j) then (1:ℕ) else 0)]
  simp_rw [hr]
  rw [sum_rows_if n t a b ht]
  simp only [hd, ite_true, Finset.sum_const, Finset.card_univ, Fintype.card_fin,
    nsmul_eq_mul, mul_one, Nat.cast_id]

lemma countMask_missing : missingCount countMask = 7019387 := by
  have he := counts_from_rows countMask 7633 799 800 (by norm_num)
    retained_missing_per_row (by intro i j; rfl)
  exact he.trans (by norm_num)

example : ∃ m : Fin 8580 → (Fin 1205 ⊕ Fin 19) → Bool,
    (∀ i j, m i (Sum.inr j) = true) ∧
    |missingAverage m - (0.668390827582004 : ℝ)| ≤ 0.000000001 := by
  refine ⟨countMask, by intro i j; rfl, ?_⟩
  rw [missingAverage_count, countMask_missing]
  simp only [Fintype.card_sum, Fintype.card_fin]
  norm_num
