lemma sumPiProd {ι κ : Type*} [Fintype ι] [DecidableEq ι] [Fintype κ] (g : ι → κ → ℝ) :
    (∑ f : ι → κ, ∏ i, g i (f i)) = ∏ i, ∑ b : κ, g i b := by
  classical
  simpa using (Finset.prod_univ_sum (fun _ : ι => (Finset.univ : Finset κ)) g).symm

noncomputable def productLaw (n h : ℕ) (a : Fin n → Fin h → ℝ)
    (ha : ∀ i j, 0 ≤ a i j) (ha' : ∀ i j, a i j ≤ 1) : Law (Array n h) where
  weight := cellWeights a
  nonneg := by
    intro x
    unfold cellWeights
    apply Finset.prod_nonneg
    intro i hi
    apply Finset.prod_nonneg
    intro j hj
    split_ifs <;> linarith [ha i j, ha' i j]
  total := by
    unfold cellWeights
    rw [sumPiProd (fun i (r : Row h) => ∏ j, if r j then 1-a i j else a i j)]
    have hr (i : Fin n) :
        (∑ r : Row h, ∏ j, if r j then 1-a i j else a i j) = 1 := by
      rw [sumPiProd (fun j (b : Bool) => if b then 1-a i j else a i j)]
      simp [Fintype.sum_bool]
    simp [hr]

noncomputable def oneByTwo (a b : ℝ) (ha : 0 ≤ a) (ha' : a ≤ 1)
    (hb : 0 ≤ b) (hb' : b ≤ 1) : Law (Array 1 2) :=
  productLaw 1 2 (fun _ => ![a,b]) (by intro i j; fin_cases j <;> simp_all)
    (by intro i j; fin_cases j <;> simp_all)

lemma sumFinOneFun {β : Type*} [Fintype β] (f : (Fin 1 → β) → ℝ) :
    (∑ x, f x) = ∑ b : β, f (fun _ => b) := by
  let e : (Fin 1 → β) ≃ β := {
    toFun := fun x => x 0
    invFun := fun b _ => b
    left_inv := by
      intro x
      funext i
      exact congrArg x (Subsingleton.elim (0 : Fin 1) i)
    right_inv := by intro b; rfl }
  apply Fintype.sum_equiv e
  intro x
  apply congrArg f
  funext i
  exact congrArg x (Subsingleton.elim i (0 : Fin 1))

lemma sumFinTwoFun {β : Type*} [Fintype β] (f : (Fin 2 → β) → ℝ) :
    (∑ x, f x) = ∑ a : β, ∑ b : β, f ![a,b] := by
  calc
    _ = ∑ p : β × β, f ![p.1,p.2] := by
      apply Fintype.sum_equiv (finTwoArrowEquiv β)
      intro x
      congr 1
      funext i
      fin_cases i <;> rfl
    _ = _ := Fintype.sum_prod_type _

lemma singletonRowsIndependent (h : ℕ) (L : Law (Array 1 h)) : rowIndependent L := by
  intro E
  simp only [Fin.prod_univ_one]
  congr 1
  funext x
  simp [Fin.forall_fin_one]

lemma oneByTwoSurvival (a b : ℝ) (ha : 0 ≤ a) (ha' : a ≤ 1)
    (hb : 0 ≤ b) (hb' : b ≤ 1) :
    survival (oneByTwo a b ha ha' hb hb') 0 1 = a ∧
    survival (oneByTwo a b ha ha' hb hb') 0 2 = a*b := by
  norm_num [survival, probability, oneByTwo, productLaw, cellWeights,
    sumFinOneFun, sumFinTwoFun, Fintype.sum_bool, Fin.prod_univ_succ,
    prefixObserved, Fin.forall_fin_succ]
  <;> ring

lemma oneByTwoDeleted (a b : ℝ) (ha : 0 ≤ a) (ha' : a ≤ 1)
    (hb : 0 ≤ b) (hb' : b ≤ 1) :
    allDeleted (oneByTwo a b ha ha' hb hb') = 1-a*b := by
  norm_num [allDeleted, probability, oneByTwo, productLaw, cellWeights,
    sumFinOneFun, sumFinTwoFun, Fintype.sum_bool, Fin.prod_univ_succ,
    prefixObserved, Fin.forall_fin_succ]
  <;> ring

lemma rowWeightTotal {n h : ℕ} (a : Fin n → Fin h → ℝ) (i : Fin n) :
    (∑ r : Row h, ∏ j, if r j then 1-a i j else a i j) = 1 := by
  rw [sumPiProd (fun j (b : Bool) => if b then 1-a i j else a i j)]
  simp

lemma productEvent {n h : ℕ} (a : Fin n → Fin h → ℝ)
    (ha : ∀ i j, 0 ≤ a i j) (ha' : ∀ i j, a i j ≤ 1)
    (E : Fin n → Row h → Prop) :
    probability (productLaw n h a ha ha') (fun x => ∀ i, E i (x i)) =
      ∏ i, ∑ r : Row h, if E i r then (∏ j, if r j then 1-a i j else a i j) else 0 := by
  classical
  unfold probability productLaw
  dsimp only
  unfold cellWeights
  calc
    _ = ∑ x : Array n h, ∏ i,
        if E i (x i) then (∏ j, if x i j then 1-a i j else a i j) else 0 := by
      apply Finset.sum_congr rfl
      intro x hx
      simp only [Finset.prod_ite_zero, Finset.mem_univ, true_implies]
      split_ifs <;> rfl
    _ = _ := sumPiProd (fun i (r : Row h) =>
      if E i r then (∏ j, if r j then 1-a i j else a i j) else 0)

lemma productMarginal {n h : ℕ} (a : Fin n → Fin h → ℝ)
    (ha : ∀ i j, 0 ≤ a i j) (ha' : ∀ i j, a i j ≤ 1)
    (i : Fin n) (E : Row h → Prop) :
    probability (productLaw n h a ha ha') (fun x => E (x i)) =
      ∑ r : Row h, if E r then (∏ j, if r j then 1-a i j else a i j) else 0 := by
  classical
  have he : (fun x : Array n h => E (x i)) =
      (fun x => ∀ j, j = i → E (x j)) := by
    funext x
    simp
  rw [he, productEvent a ha ha' (fun j r => j = i → E r)]
  calc
    _ = ∏ j, if j = i then
        (∑ r : Row h, if E r then (∏ t, if r t then 1-a j t else a j t) else 0)
        else 1 := by
      apply Finset.prod_congr rfl
      intro j hj
      by_cases hi : j = i
      · simp [hi]
      · simp [hi, rowWeightTotal]
    _ = _ := by simp

lemma productRowsIndependent {n h : ℕ} (a : Fin n → Fin h → ℝ)
    (ha : ∀ i j, 0 ≤ a i j) (ha' : ∀ i j, a i j ≤ 1) :
    rowIndependent (productLaw n h a ha ha') := by
  intro E
  rw [productEvent]
  apply Finset.prod_congr rfl
  intro i hi
  exact (productMarginal a ha ha' i (E i)).symm
