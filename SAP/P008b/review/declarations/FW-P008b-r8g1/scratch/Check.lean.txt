def axis (a : ℝ) : Plane := WithLp.toLp 2 ![a, 0]

lemma axis_dist (a b : ℝ) : dist (axis a) (axis b) = |a - b| := by
  simp [EuclideanSpace.dist_eq, axis, Fin.sum_univ_two, Real.dist_eq,
    Real.sqrt_sq_eq_abs]

def field15 : MetricSpatialModel (Fin 2) 1 where
  coord x := axis x.val
  coord_injective := by
    intro x y h
    have h0 := congrArg (fun p : Plane => p 0) h
    simp only [axis, PiLp.toLp_apply, Matrix.cons_val_zero] at h0
    exact Fin.ext (Nat.cast_injective h0)
  node _ := 0
  metric := (inferInstance : NormedAddCommGroup Plane).toMetricSpace
  ζ := Measure.count
  Y _ z := 2 + bit (z 0)

lemma distance15 (x : Fin 2) : field15.distance (field15.node 0) x = x.val := by
  unfold MetricSpatialModel.distance field15
  simp only [Fin.val_zero, Nat.cast_zero]
  rw [axis_dist]
  simp

lemma band15 : field15.band 0 1 0 = {1} := by
  ext x
  fin_cases x <;> norm_num [MetricSpatialModel.band, distance15]

lemma circle15 : field15.circle 0 1 = {1} := by
  ext x
  fin_cases x <;> norm_num [MetricSpatialModel.circle, distance15]

-- A single instance satisfies all T15 local hypotheses at once, including
-- positive inner radius and the node-exclusion separation premise.
example : (0 : ℝ) ≤ 1 ∧ (0 : ℝ) ≤ 0 ∧
    (∀ z, field15.validRegion (field15.band 0 1 0) (fun x => field15.Y x z)) ∧
    (∀ z, field15.validRegion (field15.circle 0 1) (fun x => field15.Y x z)) ∧
    (0 : ℝ) < 1 - 0 ∧
    (∀ j : Fin 1, j ≠ 0 → 1 + 0 < field15.distance (field15.node 0) (field15.node j)) := by
  have hv : ∀ z, field15.validRegion ({1} : Set (Fin 2)) (fun x => field15.Y x z) := by
    intro z
    refine ⟨MeasurableSet.singleton 1, ?_, ?_, ?_⟩
    · norm_num [field15]
    · simp [field15]
    · change Integrable (fun _ : Fin 2 => 2 + bit (z 0)) (Measure.count.restrict {1})
      exact integrable_const _
  refine ⟨by norm_num, by norm_num, ?_, ?_, by norm_num, ?_⟩
  · simpa only [band15] using hv
  · simpa only [circle15] using hv
  · intro j hj
    exact False.elim (hj (Subsingleton.elim j 0))

example (z : Assignment 1) :

-- T17: nonempty first buffer and a strictly growing second buffer for an
-- actual nonempty closed planar set, using a zero-radius first grid entry.
def centroid17 (c : Fin 2) : Plane := axis c.val
def grid17 (j : ℕ) : ℝ := if j = 0 then 0 else 1

lemma buffer17zero :
    polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) 0 = {0} := by
  ext c
  fin_cases c <;> norm_num [polygonFullBuffer, distanceBuffer, centroid17,
    Metric.infDist_singleton, axis_dist]

lemma buffer17one :
    polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) 1 = Finset.univ := by
  ext c
  fin_cases c <;> norm_num [polygonFullBuffer, distanceBuffer, centroid17,
    Metric.infDist_singleton, axis_dist]

example : IsClosed ({axis 0} : Set Plane) ∧ ({axis 0} : Set Plane).Nonempty ∧
    grid17 0 = 0 ∧
    (polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) (grid17 0)).Nonempty ∧
    0 < (1 : ℕ) ∧ 0 ≤ grid17 (1 - 1) ∧ grid17 (1 - 1) < grid17 1 ∧
    (polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) (grid17 (1 - 1))).card <
      (polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) (grid17 1)).card := by
  simp [grid17, buffer17zero, buffer17one]

example :
    archivedPolygonSummary .donut
      (fun j => polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) (grid17 j))
      (fun c => (2 : ℝ) + c.val) 0 = 2 ∧
    archivedPolygonSummary .donut
      (fun j => polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) (grid17 j))
      (fun c => (2 : ℝ) + c.val) 1 = 3 ∧
    archivedPolygonSummary .disk
      (fun j => polygonFullBuffer Finset.univ centroid17 ({axis 0} : Set Plane) (grid17 j))
      (fun c => (2 : ℝ) + c.val) 1 = 5/2 := by
  norm_num [archivedPolygonSummary, grid17, buffer17zero, buffer17one, countMean,
    Fin.sum_univ_two]

-- Relate the index representation to the actual half-integer cell centers
-- and the two filled geometric squares in the canonical fixture.
example (i : Fin 2) (c : PolygonRasterCell) :
    c ∈ rasterPolygonCells i ↔
      (rasterPolygonLeft i : ℝ) ≤ (c.1.val : ℝ) + 1/2 ∧
      (c.1.val : ℝ) + 1/2 ≤ (rasterPolygonLeft i : ℝ) + 5 ∧
      (2 : ℝ) ≤ (c.2.val : ℝ) + 1/2 ∧ (c.2.val : ℝ) + 1/2 ≤ 7 := by
  simp only [rasterPolygonCells, Finset.mem_filter, Finset.mem_univ, true_and]
  constructor
  · rintro ⟨h1, h2, h3, h4⟩
    have h1' : (rasterPolygonLeft i : ℝ) ≤ c.1.val := by exact_mod_cast h1
    have h2' : (c.1.val : ℝ) + 1 ≤ (rasterPolygonLeft i : ℝ) + 5 := by
      exact_mod_cast h2
    have h3' : (2 : ℝ) ≤ c.2.val := by exact_mod_cast h3
    have h4' : (c.2.val : ℝ) + 1 ≤ 7 := by exact_mod_cast h4
    exact ⟨by linarith, by linarith, by linarith, by linarith⟩
  · rintro ⟨h1, h2, h3, h4⟩
    have h1' : rasterPolygonLeft i < c.1.val + 1 := by
      exact_mod_cast (by linarith : (rasterPolygonLeft i : ℝ) < c.1.val + 1)
    have h2' : c.1.val < rasterPolygonLeft i + 5 := by
      exact_mod_cast (by linarith : (c.1.val : ℝ) < (rasterPolygonLeft i : ℝ) + 5)
    have h3' : 2 < c.2.val + 1 := by
      exact_mod_cast (by linarith : (2 : ℝ) < c.2.val + 1)
    have h4' : c.2.val < 7 := by
      exact_mod_cast (by linarith : (c.2.val : ℝ) < 7)
    exact ⟨by omega, h2', by omega, h4'⟩
