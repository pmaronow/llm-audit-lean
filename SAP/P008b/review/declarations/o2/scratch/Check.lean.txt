-- T20 (c): the nonempty-frontier hypothesis is needed (frontier univ = ∅, infDist to ∅ is 0)

-- T20 (a) nonvacuity: a 3x3 square raster has nonempty raster boundary
example : (rasterBoundary ((Finset.range 3 ×ˢ Finset.range 3).image
    (fun p : ℕ × ℕ => ((p.1 : ℤ), (p.2 : ℤ))))).Nonempty := by
  refine ⟨(0, 0), ?_⟩
  unfold rasterBoundary
  simp only [Finset.mem_filter]
  refine ⟨by decide, ⟨0, 0, by decide, by decide⟩⟩

-- T20 (c) nonvacuity: closed unit ball has nonempty frontier
example : (frontier (Metric.closedBall (0 : Plane) 1)).Nonempty := by
  have : frontier (Metric.closedBall (0 : Plane) 1) = Metric.sphere 0 1 :=
    frontier_closedBall (0 : Plane) one_ne_zero
  rw [this]
  exact (NormedSpace.sphere_nonempty).2 zero_le_one

-- T23 nonvacuity: a genuine (nonzero covariance) instance on Fin 2
noncomputable def lawF2 : FiniteLaw (Fin 2) := uniformLaw (Fin 2)
example : lawF2.var (fun z => if z = 0 then 1 else 0) = 1 / 4 := by
  simp [lawF2, FiniteLaw.var, FiniteLaw.cov, FiniteLaw.expect, uniformLaw, Fin.sum_univ_two]
  norm_num
