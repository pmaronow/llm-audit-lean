-- An admissible real-valued reflexive binary array and involution for T20.

    (∀ i : Fin 1, (1 : ℝ) = 1) ∧
    Function.Involutive (id : Fin 1 → Fin 1) := by

-- The zero-outcome fair experiment makes the no-interference hypothesis real.
example : NoInterference (fairExperiment 1 (fun _ _ => 0)) := by
  simp [NoInterference, fairExperiment]

-- These are actual three-valued laws on which the finite CDF clauses act.
example : 0 < (1 / 2 : ℝ) ∧ (1 / 2 : ℝ) < 1 := by norm_num
example : (∀ i : Fin 3, sharpValue i = -1 ∨ sharpValue i = 0 ∨ sharpValue i = 1) := by
  intro i
  fin_cases i <;> norm_num [sharpValue]
