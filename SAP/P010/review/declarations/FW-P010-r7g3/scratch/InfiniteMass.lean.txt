import SAP.P010.Defs.WholeW1

open MeasureTheory ProbabilityTheory Set Filter
open SAP.P010
open scoped ENNReal NNReal Classical Topology BigOperators

namespace ReviewInfinite

noncomputable def R (n : ℕ) : ℝ := (1/2) * (1/2)^n
noncomputable def O (n : ℕ) : ℝ := (3/4) * (1/4)^n

lemma R_pos (n : ℕ) : 0 < R n := by unfold R; positivity
lemma O_pos (n : ℕ) : 0 < O n := by unfold O; positivity

lemma R_hasSum : HasSum R 1 := by
  have h := (hasSum_geometric_of_lt_one (by norm_num : (0 : ℝ) ≤ 1/2)
    (by norm_num : (1/2 : ℝ) < 1)).mul_left (1/2)
  norm_num at h
  exact h

lemma O_hasSum : HasSum O 1 := by
  have h := (hasSum_geometric_of_lt_one (by norm_num : (0 : ℝ) ≤ 1/4)
    (by norm_num : (1/4 : ℝ) < 1)).mul_left (3/4)
  norm_num at h
  exact h

lemma R_integrable : Integrable R Measure.count := by
  apply integrable_count_iff.mpr
  simpa only [Real.norm_eq_abs, abs_of_pos (R_pos _)] using R_hasSum.summable

lemma O_integrable : Integrable O Measure.count := by
  apply integrable_count_iff.mpr
  simpa only [Real.norm_eq_abs, abs_of_pos (O_pos _)] using O_hasSum.summable

lemma integral_R : (∫ n, R n ∂Measure.count) = 1 := by
  rw [integral_countable R_integrable]
  simpa using R_hasSum.tsum_eq

lemma integral_O : (∫ n, O n ∂Measure.count) = 1 := by
  rw [integral_countable O_integrable]
  simpa using O_hasSum.tsum_eq

lemma ratio (n : ℕ) : R n ^ 2 / O n = 1/3 := by
  unfold R O
  have h : ((1/2 : ℝ)^n)^2 = (1/4 : ℝ)^n := by
    rw [← pow_mul, mul_comm n 2, pow_mul]
    norm_num
  rw [mul_pow, h]
  field_simp
  <;> ring

-- The infinite alternative is not an impossible premise.
example :
    (∀ᵐ n ∂(Measure.count : Measure ℕ), 0 < R n) ∧
    (∀ᵐ n ∂(Measure.count : Measure ℕ), 0 < O n) ∧
    Integrable R Measure.count ∧ Integrable O Measure.count ∧
    (∫ n, R n ∂Measure.count) = 1 ∧ (∫ n, O n ∂Measure.count) = 1 ∧
    squareRatioMass Measure.count R O = ⊤ := by
  refine ⟨Filter.Eventually.of_forall R_pos, Filter.Eventually.of_forall O_pos,
    R_integrable, O_integrable, integral_R, integral_O, ?_⟩
  simp only [squareRatioMass, lintegral_count, ratio]
  exact ENNReal.tsum_const_eq_top_of_ne_zero (by positivity)

end ReviewInfinite
