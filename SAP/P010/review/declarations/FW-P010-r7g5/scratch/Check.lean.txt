-- Three equally likely subject types, independently assigned to either arm.
-- The atom table in Defs stores (Y0,Y1),(C0,C1),Z,X:
-- (1,1),(false,true); (0,0),(true,false); (0,0),(true,true).
abbrev atoms := SAP.P010.selectionPopulationAtoms

noncomputable def mu : Measure (SAP.P010.Population 2 1) :=
  SAP.P010.finitePopulation atoms (fun _ => (1/6 : ℝ≥0∞))

lemma mu_apply (E : Set (SAP.P010.Population 2 1)) :
    mu E = ∑ i : Fin 6, if atoms i ∈ E then (1/6 : ℝ≥0∞) else 0 := by
  classical
  simp [mu, SAP.P010.finitePopulation, Measure.sum_apply_of_countable,
    Measure.smul_apply, Measure.dirac_apply, Set.indicator_apply, tsum_fintype]

instance mu_probability : IsProbabilityMeasure mu := by
  constructor
  norm_num [mu_apply, Fin.sum_univ_succ]
  apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp
  norm_num [ENNReal.toReal_add, ENNReal.toReal_inv]

lemma mu_integrable (f : SAP.P010.Population 2 1 → ℝ) : Integrable f mu := by
  apply integrable_sum_dirac
  · intro i
    norm_num
  · exact hasSum_fintype _ |>.summable

lemma mu_covariate_random : SAP.P010.covariateRandomAssignment mu := by
  constructor
  · intro a
    fin_cases a <;>
      rw [indepFun_iff_measure_inter_preimage_eq_mul] <;>
      intro B D hB hD <;>
      simp only [mu_apply, Fin.sum_univ_succ] <;>
      norm_num [atoms, SAP.P010.selectionPopulationAtoms,
        SAP.P010.potentialOutcome, SAP.P010.potentialCheck, SAP.P010.features, SAP.P010.arm,
        Set.indicator_apply] <;>
      (by_cases h0 : (0 : Fin 2) ∈ D) <;>
      (by_cases h1 : (1 : Fin 2) ∈ D) <;>
      simp_all only [and_true, and_false, ite_true, ite_false, add_zero, zero_add, mul_zero, zero_mul] <;>
      split_ifs <;> simp_all <;>
      apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp <;>
      norm_num [ENNReal.toReal_add, ENNReal.toReal_mul, ENNReal.toReal_inv]
  · intro a
    fin_cases a <;>
      norm_num [mu_apply, Fin.sum_univ_succ, atoms, SAP.P010.selectionPopulationAtoms,
        SAP.P010.assigned, SAP.P010.arm]

lemma mu_random : SAP.P010.randomAssignment mu := by
  constructor
  · intro a
    exact (mu_covariate_random.1 a).comp
      (measurable_fst.prodMk measurable_snd.fst) measurable_id
  · exact mu_covariate_random.2

lemma mu_rate (a : Fin 2) : SAP.P010.passRate mu a = 2/3 := by
  fin_cases a <;>
    norm_num [SAP.P010.passRate, mu_apply, Fin.sum_univ_succ, atoms,
      SAP.P010.selectionPopulationAtoms, SAP.P010.assigned, SAP.P010.passed,
      SAP.P010.check, SAP.P010.arm, ENNReal.toReal_add, ENNReal.toReal_inv]

lemma mu_population : SAP.P010.populationAssumptions mu := by
  refine ⟨inferInstance, mu_random, ?_⟩
  intro a
  rw [mu_rate]
  norm_num

lemma mu_check_independent : IndepFun SAP.P010.check SAP.P010.arm mu := by
  rw [indepFun_iff_measure_inter_preimage_eq_mul]
  intro B D hB hD
  simp only [mu_apply, Fin.sum_univ_succ]
  norm_num [atoms, SAP.P010.selectionPopulationAtoms,
    SAP.P010.check, SAP.P010.arm, Set.indicator_apply]
  by_cases h0 : (0 : Fin 2) ∈ D <;>
    by_cases h1 : (1 : Fin 2) ∈ D <;>
    by_cases hf : false ∈ B <;>
    by_cases ht : true ∈ B <;>
    simp_all <;>
    apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp <;>
    norm_num [ENNReal.toReal_add, ENNReal.toReal_mul, ENNReal.toReal_inv]

lemma mu_restrict (E : Set (SAP.P010.Population 2 1)) :
    mu.restrict E = Measure.sum (fun i : Fin 6 =>
      (if atoms i ∈ E then (1/6 : ℝ≥0∞) else 0) • Measure.dirac (atoms i)) := by
  classical
  unfold mu SAP.P010.finitePopulation
  rw [Measure.restrict_sum_of_countable]
  congr 1
  funext i
  rw [Measure.restrict_smul, restrict_dirac]
  split_ifs <;> simp

lemma mu_restrict_apply (E D : Set (SAP.P010.Population 2 1)) :
    mu.restrict E D = ∑ i : Fin 6,
      if atoms i ∈ E ∧ atoms i ∈ D then (1/6 : ℝ≥0∞) else 0 := by
  classical
  rw [mu_restrict]
  simp only [Measure.sum_apply_of_countable, Measure.smul_apply, Measure.dirac_apply,
    Set.indicator_apply, Pi.one_apply, smul_eq_mul, tsum_fintype]
  apply Finset.sum_congr rfl
  intro i hi
  split_ifs <;> simp_all

lemma mu_integralOn (f : SAP.P010.Population 2 1 → ℝ)
    (E : Set (SAP.P010.Population 2 1)) :
    (∫ x in E, f x ∂mu) = ∑ i : Fin 6,
      if atoms i ∈ E then (1/6 : ℝ) * f (atoms i) else 0 := by
  classical
  rw [mu_restrict, integral_sum_dirac (by intro i; split_ifs <;> norm_num), tsum_fintype]
  apply Finset.sum_congr rfl
  intro i hi
  split_ifs <;> norm_num [ENNReal.toReal_inv]

lemma mu_balance : SAP.P010.covariateBalance mu := by
  intro b hb
  rw [indepFun_iff_measure_inter_preimage_eq_mul]
  intro B D hB hD
  cases b <;>
    norm_num [Measure.smul_apply, smul_eq_mul, mu_restrict_apply, mu_apply,
      Fin.sum_univ_succ, atoms, SAP.P010.selectionPopulationAtoms,
      SAP.P010.check, SAP.P010.arm, SAP.P010.features]
  all_goals
    by_cases h0 : (0 : Fin 2) ∈ D <;>
      by_cases h1 : (1 : Fin 2) ∈ D <;>
      by_cases hx : (fun _ : Fin 1 => (0 : ℝ)) ∈ B <;>
      simp_all <;>
      apply (ENNReal.toReal_eq_toReal_iff'
        (by simp [ENNReal.mul_eq_top]) (by simp [ENNReal.mul_eq_top])).mp <;>
      norm_num [ENNReal.toReal_add, ENNReal.toReal_mul, ENNReal.toReal_inv]

lemma mu_allPass_mass : mu SAP.P010.allPass = 1/3 := by
  norm_num [mu_apply, Fin.sum_univ_succ, atoms, SAP.P010.selectionPopulationAtoms,
    SAP.P010.allPass, Fin.forall_fin_two]
  apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp
  norm_num [ENNReal.toReal_add, ENNReal.toReal_inv]

lemma mu_selected_mass (a : Fin 2) :
    mu (SAP.P010.passed ∩ SAP.P010.assigned a) = 1/3 := by
  fin_cases a <;>
    norm_num [mu_apply, Fin.sum_univ_succ, atoms, SAP.P010.selectionPopulationAtoms,
      SAP.P010.passed, SAP.P010.assigned, SAP.P010.check, SAP.P010.arm] <;>
    apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp <;>
    norm_num [ENNReal.toReal_add, ENNReal.toReal_inv]

lemma mu_means :
    SAP.P010.meanOn mu SAP.P010.response (SAP.P010.passed ∩ SAP.P010.assigned 1) = 1/2 ∧
    SAP.P010.meanOn mu SAP.P010.response (SAP.P010.passed ∩ SAP.P010.assigned 0) = 0 ∧
    SAP.P010.meanOn mu (fun w => SAP.P010.potentialOutcome 1 w - SAP.P010.potentialOutcome 0 w)
      SAP.P010.allPass = 0 := by
  simp only [SAP.P010.meanOn, mu_integralOn, mu_selected_mass, mu_allPass_mass]
  norm_num [
    Fin.sum_univ_succ, atoms, SAP.P010.selectionPopulationAtoms,
    SAP.P010.response, SAP.P010.passed, SAP.P010.assigned, SAP.P010.check,
    SAP.P010.arm, SAP.P010.potentialOutcome, SAP.P010.allPass, Fin.forall_fin_two,
    ENNReal.toReal_inv]

-- Full witness for the entire existential statement, independent of Challenge proofs.
example : SAP.P010.T21Statement := by
  refine ⟨mu, mu_population, mu_check_independent, ?_, ?_, ?_,
    (mu_integrable _).restrict, (mu_integrable _).restrict,
    (mu_integrable _).restrict, ?_⟩
  · rw [mu_allPass_mass]; norm_num
  · rw [mu_selected_mass]; norm_num
  · rw [mu_selected_mass]; norm_num
  · rw [mu_means.1, mu_means.2.1, mu_means.2.2]; norm_num

lemma atoms_zero (i : Fin 6) :
    SAP.P010.features (atoms i) = (fun _ : Fin 1 => (0 : ℝ)) := by
  fin_cases i <;> rfl

lemma mu_zero_mass (E : Set (SAP.P010.Population 2 1)) :
    mu (E ∩ {w | SAP.P010.features w = fun _ => 0}) = mu E := by
  simp only [mu_apply, Set.mem_inter_iff, Set.mem_ofPred_eq, atoms_zero, and_true]

lemma mu_zero_mean (f : SAP.P010.Population 2 1 → ℝ)
    (E : Set (SAP.P010.Population 2 1)) :
    SAP.P010.meanOn mu f (E ∩ {w | SAP.P010.features w = fun _ => 0}) =
      SAP.P010.meanOn mu f E := by
  simp only [SAP.P010.meanOn, mu_zero_mass, mu_integralOn, Set.mem_inter_iff,
    Set.mem_ofPred_eq, atoms_zero, and_true]

-- Every premise and the strict discrepancy in T23 are checked for the explicit law.
example : SAP.P010.T23Statement := by

    ?_, ?_, ?_, (mu_integrable _).restrict, (mu_integrable _).restrict,

  · rw [mu_zero_mass, mu_allPass_mass]; norm_num
  · rw [mu_zero_mass, mu_selected_mass]; norm_num
  · rw [mu_zero_mass, mu_selected_mass]; norm_num
  · simp only [mu_zero_mean]
    rw [mu_means.1, mu_means.2.1, mu_means.2.2]
    norm_num

-- A separate nonvacuity model for T22 has two arms, nonconstant covariates,
-- and positive passing AND failing strata. The check is common across arms.
def commonAtoms : Fin 4 → SAP.P010.Population 2 1 :=
  ![(fun _ => 0, fun _ => false, 0, fun _ => 0),
    (fun _ => 0, fun _ => true, 0, fun _ => 1),
    (fun _ => 0, fun _ => false, 1, fun _ => 0),
    (fun _ => 0, fun _ => true, 1, fun _ => 1)]

noncomputable def commonMu : Measure (SAP.P010.Population 2 1) :=
  SAP.P010.finitePopulation commonAtoms (fun _ => (1/4 : ℝ≥0∞))

lemma commonMu_apply (E : Set (SAP.P010.Population 2 1)) :
    commonMu E = ∑ i : Fin 4, if commonAtoms i ∈ E then (1/4 : ℝ≥0∞) else 0 := by
  classical
  simp [commonMu, SAP.P010.finitePopulation, Measure.sum_apply_of_countable,
    Measure.smul_apply, Measure.dirac_apply, Set.indicator_apply, tsum_fintype]

instance commonMu_probability : IsProbabilityMeasure commonMu := by
  constructor
  norm_num [commonMu_apply, Fin.sum_univ_succ]
  apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp
  norm_num [ENNReal.toReal_mul, ENNReal.toReal_inv]

lemma commonMu_covariate_random : SAP.P010.covariateRandomAssignment commonMu := by
  constructor
  · intro a
    rw [indepFun_iff_measure_inter_preimage_eq_mul]
    intro B D hB hD
    norm_num [commonMu_apply, Fin.sum_univ_succ, commonAtoms,
      SAP.P010.potentialOutcome, SAP.P010.potentialCheck, SAP.P010.features, SAP.P010.arm]
    by_cases h0 : (0 : Fin 2) ∈ D <;>
      by_cases h1 : (1 : Fin 2) ∈ D <;>
      by_cases hf : ((0 : ℝ), false, fun _ : Fin 1 => (0 : ℝ)) ∈ B <;>
      by_cases ht : ((0 : ℝ), true, fun _ : Fin 1 => (1 : ℝ)) ∈ B <;>
      simp_all <;>
      apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp <;>
      norm_num [ENNReal.toReal_add, ENNReal.toReal_mul, ENNReal.toReal_inv]
  · intro a
    fin_cases a <;>
      norm_num [commonMu_apply, Fin.sum_univ_succ, commonAtoms,
        SAP.P010.assigned, SAP.P010.arm]

lemma commonMu_same : commonMu SAP.P010.sameCheck = 1 := by
  norm_num [commonMu_apply, Fin.sum_univ_succ, commonAtoms, SAP.P010.sameCheck]
  apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp
  norm_num [ENNReal.toReal_add, ENNReal.toReal_mul, ENNReal.toReal_inv]

example : ∃ ν : Measure (SAP.P010.Population 2 1),
    IsProbabilityMeasure ν ∧ SAP.P010.covariateRandomAssignment ν ∧
      ν SAP.P010.sameCheck = 1 ∧
      0 < ν {w | SAP.P010.check w = false} ∧
      0 < ν {w | SAP.P010.check w = true} := by
  refine ⟨commonMu, inferInstance, commonMu_covariate_random, commonMu_same, ?_, ?_⟩ <;>
    norm_num [commonMu_apply, Fin.sum_univ_succ, commonAtoms, SAP.P010.check, SAP.P010.arm]
