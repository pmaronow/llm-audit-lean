    pathMean (fun u v => (∑ i, pathBlock t u v i)^2) = 29/5 ∧

      8*t^2+8836/625 := by

    norm_num [pathMean, pathBlock, quadratic, oracleMatrix, adjacency, pathGraph,
      Matrix.one_apply, Fin.sum_univ_succ]

example (t : ℝ) (ht : t = 1 ∨ t = 2) :
    let D : Matrix (Fin 3) (Fin 2) ℝ :=
      fun i j => if j = 0 then 1 else (![t,-2*t,t] : Fin 3 → ℝ) i
    Matrix.det (Dᵀ * D) ≠ 0 := by
  intro D
  have hD : Dᵀ * D = (!![3,0;0,6*t^2] : Matrix (Fin 2) (Fin 2) ℝ) := by
    ext i j
    change (∑ r : Fin 3, D r i * D r j) = _
    fin_cases i <;> fin_cases j <;>
      simp [D, Fin.sum_univ_succ] <;> ring
  rw [hD, Matrix.det_fin_two]
  rcases ht with rfl | rfl <;> norm_num

-- Asymptotic premise instances: use the certified concrete counterexample,
-- not any assumed scaling or the conclusion of the result being instantiated.
theorem cycleScaling : Scaling (cycleFamily (3/2)) (7/2) 2 4 2 :=
  T1.1.1.1

theorem cycleValid : Valid (cycleFamily (3/2)) :=
  T1.1.1.2.1

theorem cycleRow : RowScaling (cycleFamily (3/2)) (7/2) 2 4 2 := by
  exact cycleScaling.2

theorem cycleWhole : WholeScaling_W3 (cycleFamily (3/2)) (7/2) 2 4 2 := by
  exact cycleScaling.2

example : ∃ F : Family, RowScaling F (7/2) 2 4 2 :=
  ⟨cycleFamily (3/2), cycleRow⟩

example : ∃ (F : Family)
    (M : ∀ k, Matrix (Fin (F.size k)) (Fin (F.size k)) ℝ),
    WholeScaling_W3 F (7/2) 2 4 2 ∧
    (∀ k i j, M k i j = M k j i) ∧
    (∀ k, operatorNorm (M k) ≤ 0) := by
  refine ⟨cycleFamily (3/2), fun _ => 0, cycleWhole, ?_, ?_⟩
  · simp
  · intro k
    simp [operatorNorm]

example : ∃ (F : Family)
    (A : ∀ k, (Fin (F.size k) → ℝ) → Graph (F.size k)),
    WholeScaling_W3 F (7/2) 2 4 2 ∧ Valid F ∧
    (∀ k y, Maximizes (F.model k).R (F.model k).d
      (fun B => V1plus (F.model k).d B (weighted (F.model k) y)) (A k y)) := by
  refine ⟨cycleFamily (3/2), fun k _ => ((cycleFamily (3/2)).model k).G,
    cycleWhole, cycleValid, ?_⟩
  intro k y
  refine ⟨cycleValid k, ?_⟩
  intro B hB
  have hs := (T1.1.1.2.2.2.1 k).2.2.2.1 B
  have heq := hs.mp hB
  subst B
  exact le_rfl
