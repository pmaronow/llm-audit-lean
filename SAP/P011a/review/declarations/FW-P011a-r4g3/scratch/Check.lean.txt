lemma edge_sum {n : ℕ} {M : Type*} [AddCommMonoid M] (f : Edge n → M) :
    (∑ e, f e) = ∑ i : Fin n, ∑ j : Fin n,
      if h : i < j then f ⟨(i,j),h⟩ else 0 := by
  classical
  calc
    (∑ e, f e) = ∑ p : Fin n × Fin n,
        if h : p.1 < p.2 then f ⟨p,h⟩ else 0 := by
      symm
      apply Finset.sum_congr_set (s := {p : Fin n × Fin n | p.1 < p.2})
      · intro p hp
        exact dif_pos hp
      · intro p hp
        exact dif_neg hp
    _ = _ := by rw [Fintype.sum_prod_type]

-- Every finite gadget actually has a maximum matching: the maximum-matching
-- antecedent of MatchingReduction is inhabited independently of T13.
example {n : ℕ} (P : DegreeProblem n) :
    ∃ J : Finset (GadgetEdge P), IsMatching J ∧
      ∀ J' : Finset (GadgetEdge P), IsMatching J' →
        (∑ e ∈ J', gadgetWeight e) ≤ ∑ e ∈ J, gadgetWeight e := by
  classical
  let choices : Finset (Finset (GadgetEdge P)) := Finset.univ.filter IsMatching
  have hne : choices.Nonempty := by
    refine ⟨∅, ?_⟩
    simp [choices, IsMatching]
  obtain ⟨J, hJ, hmax⟩ := Finset.exists_max_image choices
    (fun J => ∑ e ∈ J, gadgetWeight e) hne
  refine ⟨J, (Finset.mem_filter.mp hJ).2, ?_⟩
  intro J' hJ'
  exact hmax J' (by simp [choices, hJ'])

-- Independent fractional feasibility for the triangle.
example : RelaxedFeasible (emptyGraph 3) (fun _ => 1) (fun _ => 1/2) := by
  constructor
  · intro e
    norm_num
  constructor
  · simp [toUpper, emptyGraph]
  · intro i
    rw [edge_sum]
    fin_cases i <;> norm_num [Fin.sum_univ_succ]

-- A full instance of the general RelaxationUpperBound antecedents.
-- The certificate 3 is intentionally loose; attainment is not assumed.
example : (∀ a : Edge 3 → ℝ,
    RelaxedFeasible (emptyGraph 3) (fun _ => 1) a →
      relaxedObjective a (fun _ => 1) ≤ 3) ∧
    ∃ b : Edge 3 → Bool, Compatible (emptyGraph 3) (fun _ => 1) (fromUpper b) := by
  constructor
  · intro a ha
    change (∑ e, 1 * a e) ≤ 3
    calc
      (∑ e : Edge 3, 1 * a e) ≤ ∑ _e : Edge 3, (1 : ℝ) :=
        Finset.sum_le_sum (fun e _ => by simpa using (ha.1 e).2)
      _ = 3 := by
        rw [edge_sum]
        simp only [Fin.sum_univ_succ, Fin.sum_univ_zero, add_zero]
        norm_num
  · refine ⟨fun _ => false, ?_⟩
    constructor
    · simp [emptyGraph]
    · intro i
      simp [degree, fromUpper]
