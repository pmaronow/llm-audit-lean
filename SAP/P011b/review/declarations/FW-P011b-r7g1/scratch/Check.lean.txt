lemma uniform_ae {Ω : Type} [Fintype Ω] [Nonempty Ω] {n : ℕ}
    (f : Ω → (Fin n → ℝ)) (p : (Fin n → ℝ) → Prop)
    (hp : MeasurableSet {y | p y}) (h : ∀ u, p (f u)) :
    ∀ᵐ y ∂uniformLaw f, p y := by
  classical
  letI : MeasurableSpace Ω := ⊤
  have hf : Measurable f := Measurable.of_discrete
  rw [uniformLaw, ← PMF.toMeasure_map f (PMF.uniformOfFintype Ω) hf]
  exact (ae_map_iff hf.aemeasurable hp).2 (ae_of_all _ h)

lemma uniform_integrable {Ω : Type} [Fintype Ω] [Nonempty Ω] {n : ℕ}
    (f : Ω → (Fin n → ℝ)) (g : (Fin n → ℝ) → ℝ) (hg : Measurable g) :
    Integrable g (uniformLaw f) := by
  classical
  letI : MeasurableSpace Ω := ⊤
  have hf : Measurable f := Measurable.of_discrete
  rw [uniformLaw, ← PMF.toMeasure_map f (PMF.uniformOfFintype Ω) hf]
  exact (integrable_map_measure hg.aestronglyMeasurable hf.aemeasurable).2 Integrable.of_finite

-- T17: all five integrable moment domains hold simultaneously. Indices may
-- repeat, so this also checks that finite subsets refer to positions.
example (i j k l : Fin 4) :
    let m := constantWitness
    let B := betaHat m
    let X : Fin 4 → (Fin 4 → ℝ) → ℝ := fun a y => weighted m y a
    let v : Fin 4 → Fin 4 := ![i,j,k,l]
    let Y : Fin 4 → (Fin 4 → ℝ) → ℝ := fun a y => y (v a)
    (Integrable (fun y => (B y)^2) m.law ∧
      ∀ p ∈ ({(i,j),(k,l)} : Finset (Fin 4 × Fin 4)),
        Integrable (fun y => X p.1 y*X p.2 y) m.law ∧
        Integrable (fun y => X p.1 y*B y) m.law ∧
        Integrable (fun y => X p.2 y*B y) m.law) ∧
    (∀ s : Finset (Fin 4),
      Integrable (fun y => (∏ a ∈ s, X (v a) y)*(B y)^(4-s.card)) m.law) ∧
    (∀ s : Finset (Fin 4), s.Nonempty →
      Integrable (fun y => ∏ a ∈ s, X (v a) y) m.law) ∧
    (Integrable (fun y => (B y)^2) m.law ∧
      ∀ p ∈ ({(i,j),(k,l)} : Finset (Fin 4 × Fin 4)),
        Integrable (fun y => y p.1*y p.2) m.law ∧
        Integrable (fun y => y p.1*B y) m.law ∧
        Integrable (fun y => y p.2*B y) m.law) ∧
    (∀ s : Finset (Fin 4),
      Integrable (fun y => (∏ a ∈ s, Y a y)*(B y)^(4-s.card)) m.law) := by
  simp [cw_integrable]
