-- T21: one explicit model satisfies every outer hypothesis, including L².
example : (0 < (3 : ℕ)) ∧ Dependency zeroModel ∧ Homoskedastic zeroModel ∧
    (∀ i, MemLp (fun y => weighted zeroModel y i) 2 zeroModel.law) := by
  refine ⟨by decide, ?_, ?_, ?_⟩
  · intro S T _ _
    change IndepFun (fun y : Fin 3 → ℝ => fun i : {i // i ∈ S} => y i.val)
      (fun y : Fin 3 → ℝ => fun j : {j // j ∈ T} => y j.val)
      (Measure.dirac (fun _ => (0 : ℝ)))
    have h : IndepFun
        (fun _ : Fin 3 → ℝ => fun _ : {i // i ∈ S} => (0 : ℝ))
        (fun _ : Fin 3 → ℝ => fun _ : {j // j ∈ T} => (0 : ℝ))
        (Measure.dirac (fun _ : Fin 3 => (0 : ℝ))) := indepFun_const_left _ _
    apply h.congr <;> simp [Filter.EventuallyEq, MeasureTheory.ae_dirac_eq]
  · intro i j
    simp [SAP.P011b.covariance, expectation, weighted, zeroModel]
  · intro i
    change eLpNorm (fun y => weighted zeroModel y i) 2
      (Measure.dirac (fun _ => (0 : ℝ))) < ∞
    simp [eLpNorm_dirac, weighted, zeroModel]

-- The estimator's correlation term really vanishes on the empty true graph.
example (n : ℕ) (hn : 0 < n) (y : Fin n → ℝ) :
    (n : ℝ) * V2star (signModel n) y = sampleVariance y := by
  have hnr : (n : ℝ) ≠ 0 := by exact_mod_cast (Nat.ne_of_gt hn)
  have hx : weighted (signModel n) y = y := by
    ext i
    simp [weighted, signModel]
  unfold V2star
  rw [hx]
  simp [signModel, emptyGraph, adjacency]
  field_simp

def goodSignCount (n : ℕ) : ℕ :=
  ∑ k ∈ Finset.range (n+1),
    if 2 * (2*(k : ℤ) - (n : ℤ))^2 < (n : ℤ)^2 then Nat.choose n k else 0

-- Independent integer enumeration of the strict event at epsilon = 1/2.
example : goodSignCount 2 = 2 := by decide
example : goodSignCount 4 = 14 := by decide
example : goodSignCount 16 = 65262 := by decide
example : (2 : ℚ)/2^2 = 1/2 ∧ (14 : ℚ)/2^4 = 7/8 ∧
    (65262 : ℚ)/2^16 = 32631/32768 := by norm_num
