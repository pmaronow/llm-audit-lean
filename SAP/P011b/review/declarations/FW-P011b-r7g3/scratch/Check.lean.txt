-- T26: the explicit source draw has the claimed positive mass under the actual law.
example : probability (threePointModel 3)
    {y | y = (![0, 1, 2] : Fin 3 → ℝ)} = 1 / 27 := by
  classical
  let f : (Fin 3 → Fin 3) → (Fin 3 → ℝ) := fun u i => ((u i).val : ℝ)
  let a : Fin 3 → Fin 3 := ![0, 1, 2]
  have hf : Measurable f := measurable_of_countable _
  have hinj : Function.Injective f := by
    intro u v huv
    funext i
    apply Fin.ext
    have hi : ((u i).val : ℝ) = ((v i).val : ℝ) := congrFun huv i
    exact_mod_cast hi
  have ha : f a = (![0, 1, 2] : Fin 3 → ℝ) := by
    funext i
    fin_cases i <;> norm_num [f, a]
  have hpre : f ⁻¹' {y | y = (![0, 1, 2] : Fin 3 → ℝ)} = {a} := by
    ext u
    change f u = _ ↔ u = a
    rw [← ha]
    exact hinj.eq_iff
  change (((PMF.uniformOfFintype (Fin 3 → Fin 3)).map f).toMeasure
    {y | y = (![0, 1, 2] : Fin 3 → ℝ)}).toReal = _
  rw [PMF.toMeasure_map_apply f _ _ hf (by measurability), hpre]
  norm_num [PMF.toMeasure_apply_singleton, PMF.uniformOfFintype_apply]

def onePositiveCap : Fin 3 → ℕ := ![2, 0, 0]

lemma only_empty (B : Graph 3)
    (hB : Compatible (emptyGraph 3) onePositiveCap B) : B = emptyGraph 3 := by
  have h1 : degree B 1 ≤ 0 := by simpa [onePositiveCap] using hB.2 1
  have h2 : degree B 2 ≤ 0 := by simpa [onePositiveCap] using hB.2 2
  have h10 : B.val 1 0 = false := by
    cases h : B.val 1 0
    · rfl
    · simp [degree] at h1
      simpa [h] using h1 (x := 0)
  have h12 : B.val 1 2 = false := by
    cases h : B.val 1 2
    · rfl
    · simp [degree] at h1
      simpa [h] using h1 (x := 2)
  have h20 : B.val 2 0 = false := by
    cases h : B.val 2 0
    · rfl
    · simp [degree] at h2
      simpa [h] using h2 (x := 0)
  apply Subtype.ext
  funext i j
  have h01 : B.val 0 1 = false := (B.property.1 0 1).trans h10
  have h02 : B.val 0 2 = false := (B.property.1 0 2).trans h20
  have h21 : B.val 2 1 = false := (B.property.1 2 1).trans h12
  fin_cases i <;> fin_cases j <;>
    simp [emptyGraph, B.property.2, h10, h12, h20, h01, h02, h21]

lemma empty_maximizes : Maximizes (emptyGraph 3) onePositiveCap
    (fun B => V2 B (![0, 1, 2] : Fin 3 → ℝ)) (emptyGraph 3) := by
  constructor
  · constructor
    · simp [emptyGraph]
    · intro i; simp [degree, emptyGraph]
  · intro B hB
    rw [only_empty B hB]
