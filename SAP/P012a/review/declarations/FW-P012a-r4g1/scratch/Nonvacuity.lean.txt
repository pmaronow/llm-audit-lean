example : preferential proposalData proposalStart = some (-1/2) := by
  have hJ : nonseeds proposalData = {2} := by decide
  have hU : susceptible proposalData proposalStart 1 2 = {s 2, u 3 0} := by decide
  have hM : sameSusceptible proposalData proposalStart 1 2 = {u 3 0} := by decide
  have hdef : PreferentialDefined proposalData proposalStart := by decide
  simp only [preferential, if_pos hdef]
  congr 1
  unfold pValue pSum
  rw [hJ]
  simp only [Finset.card_singleton, Finset.sum_singleton, Nat.cast_one, div_one]
  change (0 : ℝ) - (sameSusceptible proposalData proposalStart 1 2).card /
    ((susceptible proposalData proposalStart 1 2).card : ℝ) = -1/2
  rw [hU, hM]
  norm_num [s, u, Finset.card_insert_eq_ite]
