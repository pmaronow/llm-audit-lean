-- Independent evaluation of the two programs on the same nondegenerate datum.
example : w2Finish (w2Scan proofStepData [3,1,2] [3,2]) =
    ((-5 / 7 : ℚ), (23 / 35 : ℚ)) := by
  norm_num [w2Finish, w2Scan, w2ScanVertices, w2ScanEdges,
    w2VertexStep, w2EdgeStep, proofStepData]

example : let st := w2Scan proofStepData [3,1,2] [3,2]
    st.d0 = 5 ∧ st.d1 = 3 ∧ st.ones = 0 ∧ st.zeros = 1 ∧
    st.mixed = 1 ∧ st.visits = 5 ∧ st.additions = 5 := by
  norm_num [w2Scan, w2ScanVertices, w2ScanEdges,
    w2VertexStep, w2EdgeStep, proofStepData]

example : pWholeOutput (pWholeRun proofStepData [1,2,3]) =
    ((0 : ℚ), (1 / 3 : ℚ)) := by
  norm_num [pWholeOutput, pWholeRun, pWholeRecord, pWholeStep,
    pWholeInitial, proofStepData, List.foldl, Function.update]
