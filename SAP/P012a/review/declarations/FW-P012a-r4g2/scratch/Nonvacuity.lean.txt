private theorem pair_eq_iff (a b c d : Vertex) :
    ({a,b} : Finset Vertex) = {c,d} ↔
      (a=c ∧ b=d) ∨ (a=d ∧ b=c) := by
  rw [← Finset.coe_inj, Finset.coe_pair, Finset.coe_pair]
  exact Set.pair_eq_pair_iff

attribute [local simp] pair_eq_iff
