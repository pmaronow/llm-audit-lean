-- T11: a nonempty complete-randomization instance, including the numerical premise.
example : ∃ (U B : Finset ℕ) (i n : ℕ),
    i ∈ U ∧ B ⊆ U ∧ i ∉ B ∧ B.card = 39 ∧ 1 ≤ n ∧
    n ≤ U.card - 39 ∧ U.card ≤ 10*n+5 := by
  refine ⟨range 45, range 39, 39, 4, ?_⟩
  norm_num [range_subset]
  intro x hx

-- T12: use the explicitly specified four-atom design, with outcomes 2 and -1.
-- These atoms encode the independent unit exposures with marginals 1/3 and 2/3.
abbrev signedF := signedReplicationPopulation 2

lemma signed_marginals : SAP.P013a.pi signedF 0 1 = 1/3 ∧ SAP.P013a.pi signedF 1 1 = 2/3 := by
  norm_num [SAP.P013a.pi, expectation, indicator, signedF, signedReplicationPopulation,
    signedReplicationDesign, Fin.sum_univ_succ]

lemma signed_positive : Positivity signedF 1 := by
  intro i
  fin_cases i <;> norm_num [SAP.P013a.pi, expectation, indicator, signedF,
    signedReplicationPopulation, signedReplicationDesign, Fin.sum_univ_succ]

lemma signed_minimum : IsMinimumPi signedF 1 (1/3) := by
  refine ⟨⟨0, signed_marginals.1⟩, ?_⟩
  intro i
  fin_cases i <;>
    norm_num [SAP.P013a.pi, expectation, indicator, signedF,
      signedReplicationPopulation, signedReplicationDesign, Fin.sum_univ_succ]

lemma signed_total : potentialTotal signedF 1 = 1 := by
  norm_num [potentialTotal, signedF, signedReplicationPopulation, Fin.sum_univ_succ]

lemma sum_one_function (f : (Fin 1 → Fin 4) → ℝ) :
    ∑ z, f z = ∑ z : Fin 4, f (fun _ => z) := by
  apply Fintype.sum_equiv (Equiv.funUnique (Fin 1) (Fin 4))
  intro z
  congr 1
  ext i
  have : i = 0 := Subsingleton.elim _ _
  subst i
  rfl

lemma signed_expectation :
    expectation (independentProduct signedReplicationDesign
      (replicateDesign signedReplicationDesign 1)) (simulatedHTTotal signedF 1 1) = 2/9 := by
  unfold expectation independentProduct
  rw [Fintype.sum_prod_type]
  simp_rw [sum_one_function]
  norm_num [simulatedHTTotal, simulatedPi, replicateCount, observed, indicator,
    replicateDesign, signedF, signedReplicationPopulation, signedReplicationDesign,
    Fin.sum_univ_succ, Fin.prod_univ_succ]

-- All assumptions hold, and the actual bias is 7/9, exceeding 4/9.
example : (0 < (2 : ℕ)) ∧ Positivity signedF 1 ∧
    IsMinimumPi signedF 1 (1/3) ∧ potentialTotal signedF 1 ≠ 0 ∧
    |(expectation (independentProduct signedReplicationDesign
      (replicateDesign signedReplicationDesign 1)) (simulatedHTTotal signedF 1 1) -
      potentialTotal signedF 1) / potentialTotal signedF 1| > (1-(1/3 : ℝ))^(1+1) := by
  refine ⟨by norm_num, signed_positive, signed_minimum, ?_, ?_⟩
  · rw [signed_total]; norm_num
  · rw [signed_expectation, signed_total]; norm_num

-- b=2/3 is a strictly positive target below one. Its logarithmic threshold is zero.
example : (0 < (2/3 : ℝ)) ∧
    (1 : ℝ) ≥ Real.log (2/3) / Real.log (1-(1/3 : ℝ)) - 1 ∧
    |(expectation (independentProduct signedReplicationDesign
      (replicateDesign signedReplicationDesign 1)) (simulatedHTTotal signedF 1 1) -
      potentialTotal signedF 1) / potentialTotal signedF 1| > (2/3 : ℝ) := by
  have hl : Real.log (2/3 : ℝ) ≠ 0 :=
    Real.log_ne_zero_of_pos_of_ne_one (by norm_num) (by norm_num)
  refine ⟨by norm_num, ?_, ?_⟩
  · norm_num [show (1-(1/3 : ℝ)) = 2/3 by norm_num, div_self hl]
  · rw [signed_expectation, signed_total]; norm_num

-- Directly instantiate the exact negated universal assertions, independently of T12.
example : ¬ PrintedBiasBound := by
  intro h
  have hn : potentialTotal signedF 1 ≠ 0 := by rw [signed_total]; norm_num
  have hb := h signedF (by norm_num) 1 signed_positive 1 (1/3) signed_minimum hn
  rw [signed_expectation, signed_total] at hb
  norm_num at hb

example : ¬ PrintedReplicateRule := by
  intro h
  have hn : potentialTotal signedF 1 ≠ 0 := by rw [signed_total]; norm_num
  have hl : Real.log (2/3 : ℝ) ≠ 0 :=
    Real.log_ne_zero_of_pos_of_ne_one (by norm_num) (by norm_num)
  have hr : (1 : ℝ) ≥ Real.log (2/3) / Real.log (1-(1/3 : ℝ)) - 1 := by
    norm_num [show (1-(1/3 : ℝ)) = 2/3 by norm_num, div_self hl]
  have hb := h signedF (by norm_num) 1 signed_positive 1 (1/3) signed_minimum hn
    (2/3) (by norm_num) (by simpa only [Nat.cast_one] using hr)
  rw [signed_expectation, signed_total] at hb
  norm_num at hb

-- T13 and the exposure branch of T14: both labels have positive probability,
-- units are dependent, and their within-label joint probability is zero.
def exposureF : Population sharedDesign 2 2 where
  exposure z i := if i = z then 1 else 0
  outcome i d := if d = 1 then (if i = 0 then 2 else -1)
    else if i = 0 then 1 else 3

lemma exposure_positive (d : Fin 2) : Positivity exposureF d := by
  intro i
  fin_cases i <;> fin_cases d <;>
    norm_num [SAP.P013a.pi, expectation, indicator, exposureF, sharedDesign, Fin.sum_univ_succ]

example : (0 < (2 : ℕ)) ∧ (1 : Fin 2) ≠ 0 ∧
    Positivity exposureF 1 ∧ Positivity exposureF 0 := by
  exact ⟨by norm_num, by decide, exposure_positive 1, exposure_positive 0⟩

example : expectation sharedDesign (htTotal exposureF 1) = 1 ∧
    variance sharedDesign (htTotal exposureF 1) = 9 ∧
    expectation sharedDesign (htEffect exposureF 1 0) = -3/2 ∧
    variance sharedDesign (htEffect exposureF 1 0) = 1/4 ∧
    covariance sharedDesign (htTotal exposureF 1) (htTotal exposureF 0) = 6 := by
  norm_num [variance, covariance, expectation, htEffect, htTotal, SAP.P013a.pi, indicator,
    exposureF, sharedDesign, Fin.sum_univ_succ]

example : jointPi exposureF 0 1 1 1 = 0 ∧ missingA exposureF 1 = -4 ∧
    expectation sharedDesign (varianceHT exposureF 1) = 5 ∧
    expectation sharedDesign (fun z => varianceHT exposureF 1 z + correctionA2 exposureF 1 z) = 10 := by
  norm_num [jointPi, missingA, expectation, varianceHT, pairTerm, correctionA2,
    SAP.P013a.pi, indicator, exposureF, sharedDesign, Fin.sum_univ_succ,
    Finset.univ_fin2]

-- A separate instance has all same-label joint probabilities strictly positive.
def fullSupportF : Population signedReplicationDesign 2 2 where
  exposure := (signedReplicationPopulation 2).exposure
  outcome := (signedReplicationPopulation 2).outcome

example : (0 < (2 : ℕ)) ∧ Positivity fullSupportF 1 ∧
    (∀ i j, 0 < jointPi fullSupportF i j 1 1) := by
  refine ⟨by norm_num, signed_positive, ?_⟩
  intro i j
  fin_cases i <;> fin_cases j <;>
    norm_num [jointPi, expectation, indicator, fullSupportF,
      signedReplicationPopulation, signedReplicationDesign, Fin.sum_univ_succ]

-- T14 general sampling: one certainty unit, two mutually exclusive units,
-- and a zero-weight atom at which every unit is selected.
def sampleD : WholeSamplingDesign (Fin 3) (Fin 3) where
  weight := ![1/2, 1/2, 0]
  nonneg z := by fin_cases z <;> norm_num
  total := by norm_num [Fin.sum_univ_succ]
  selected z i := if i = 0 then true else if z = 2 then true
    else decide (i.val = z.val+1)

def sampleY : Fin 3 → ℝ := ![5, 2, -1]

lemma sample_positive : ∀ i, 0 < samplingPi sampleD i := by
  intro i
  fin_cases i <;>
    norm_num [samplingPi, samplingExpect, samplingIndicator, sampleD, Fin.sum_univ_succ]

example : samplingPi sampleD 0 = 1 ∧ samplingPi sampleD 1 = 1/2 ∧
    samplingPi sampleD 2 = 1/2 ∧ samplingJoint sampleD 1 2 = 0 ∧
    sampleD.weight 2 = 0 ∧ sampleD.selected 2 1 = true ∧
    sampleD.selected 2 2 = true := by
  norm_num [samplingPi, samplingJoint, samplingExpect, samplingIndicator,
    sampleD, Fin.sum_univ_succ]

example : (∀ i, 0 < samplingPi sampleD i) ∧
    (∀ i j : Fin 3, i ≠ j ∧ samplingJoint sampleD i j = 0 →
      0 < (3 : ℝ) ∧ 0 < (3/2 : ℝ) ∧ 1/(3 : ℝ)+1/(3/2 : ℝ)=1) := by
  refine ⟨sample_positive, ?_⟩
  intro i j _
  norm_num

example : samplingVariance sampleD (samplingTotal sampleD sampleY) = 9 ∧
    samplingMissingA sampleD sampleY = -4 ∧
    samplingExpect sampleD (samplingVarianceHT sampleD sampleY) = 5 ∧
    samplingExpect sampleD (fun z => samplingVarianceHT sampleD sampleY z +
      samplingQuadraticCorrection sampleD sampleY z) = 10 := by
  norm_num [samplingVariance, samplingTotal, samplingMissingA, samplingVarianceHT,
    samplingQuadraticCorrection, samplingPi, samplingJoint, samplingExpect,
    samplingIndicator, sampleD, sampleY, Fin.sum_univ_succ]
