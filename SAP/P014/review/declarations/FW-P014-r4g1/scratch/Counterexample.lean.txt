  intro σ
  have h01 : σ 0 ≠ σ 1 := σ.injective.ne (by decide)
  have h02 : σ 0 ≠ σ 2 := σ.injective.ne (by decide)
  have h12 : σ 1 ≠ σ 2 := σ.injective.ne (by decide)
  generalize h0 : σ 0 = j0 at h01 h02
  generalize h1 : σ 1 = j1 at h01 h12
  generalize h2 : σ 2 = j2 at h02 h12
  fin_cases j0 <;> fin_cases j1 <;> fin_cases j2 <;>
    simp_all [probability, uniformSix, univ6, scores1, Fin.forall_fin_succ] <;> norm_num
