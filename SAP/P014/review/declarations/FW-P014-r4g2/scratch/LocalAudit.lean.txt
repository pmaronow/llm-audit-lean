-- Expand a three-subject CDF by finite-sum identities before simplifying indicators.
private theorem cdf_three (x : Fin 4 → ℝ) (t : ℝ) :
    windowCDF localThree x t =
      ((if x 0 ≤ t then (1:ℝ) else 0) +
       (if x 1 ≤ t then (1:ℝ) else 0) +
       (if x 2 ≤ t then (1:ℝ) else 0)) / 3 := by
  classical
  unfold windowCDF localThree
  rw [Finset.sum_insert (by norm_num : (0:Fin 4) ∉ ({1,2} : Finset (Fin 4))),
    Finset.sum_insert (by norm_num : (1:Fin 4) ∉ ({2} : Finset (Fin 4))), Finset.sum_singleton]
  norm_num
  ring

-- Independent verification of every printed-local hypothesis for each witness.
example : PrintedLocalConditions uniformSix scores1 2 localThree := by
  classical
  refine ⟨?_, ?_, ?_, ?_, ?_⟩
  · intro ω hω
    fin_cases ω <;>
      norm_num [windowSubjects, insideWindow, localThree, scores1, Fin.univ_succ,
        Finset.filter_insert, Finset.filter_singleton, Finset.filter_empty]
  · refine ⟨fun t => ((if (-2:ℝ) ≤ t then (1:ℝ) else 0) +
        (if (-1:ℝ) ≤ t then (1:ℝ) else 0) +
        (if (1:ℝ) ≤ t then (1:ℝ) else 0)) / 3, ?_⟩
    intro ω hω t
    fin_cases ω <;>
      by_cases h0 : (-2:ℝ) ≤ t <;>
      by_cases h1 : (-1:ℝ) ≤ t <;>
      by_cases h2 : (1:ℝ) ≤ t <;>
      norm_num [cdf_three, scores1, h0, h1, h2]
  · intro ω hω
    fin_cases ω <;>
      norm_num [probability, uniformSix, scores1, localThree, Fin.sum_univ_succ,
        funext_iff, Fin.forall_fin_succ]
  · intro i hi j hj hne
    fin_cases i <;> fin_cases j <;>
      norm_num [localThree, probability, uniformSix, scores1, Fin.sum_univ_succ] at *
  · intro ω hω
    fin_cases ω <;>
      norm_num [armSize, localThree, thresholdAssignment, scores1,
        Finset.filter_insert, Finset.filter_singleton, Finset.filter_empty]
