import SAP.P014.Challenge
open SAP.P014 MeasureTheory
open scoped ENNReal

noncomputable def μB : Measure Bool := (1/2 : ℝ≥0∞) • (Measure.dirac true + Measure.dirac false)
instance : IsProbabilityMeasure μB := ⟨by simp [μB]; exact ENNReal.inv_two_add_inv_two⟩
open Classical in
lemma μB_apply (s : Set Bool) : μB s =
    (1/2 : ℝ≥0∞) * ((if true ∈ s then 1 else 0) + (if false ∈ s then 1 else 0)) := by
  simp only [μB, Measure.smul_apply, Measure.add_apply, smul_eq_mul]
  rw [Measure.dirac_apply' _ (by measurability), Measure.dirac_apply' _ (by measurability)]
  simp [Set.indicator]

-- three subjects, window {0,1}, subject 2 always outside (score -5), so always control
def S3 : Finset (Fin 3) := {0, 1}
noncomputable def X3 : Bool → Fin 3 → ℝ := fun ω => if ω then ![1, -1, -5] else ![-1, 1, -5]
noncomputable def v3 : S3 → ℝ := fun i => if i.1 = 0 then 1 else -1
def t0 : S3 := ⟨0, by decide⟩
def t1 : S3 := ⟨1, by decide⟩
lemma perm3 : ∀ σ : Equiv.Perm S3, σ = 1 ∨ σ = Equiv.swap t0 t1 := by decide
lemma S3cases (i : S3) : i = t0 ∨ i = t1 := by
  rcases i with ⟨i, hi⟩; simp [S3] at hi; rcases hi with rfl | rfl <;> simp [t0, t1]
lemma S3card : S3.card = 2 := by decide

lemma v3_inj : Function.Injective v3 := by
  intro a b h
  rcases S3cases a with rfl | rfl <;> rcases S3cases b with rfl | rfl <;>
    simp [v3, t0, t1] at h ⊢ <;> norm_num at h
lemma v3_win : ∀ i, insideWindow 2 (v3 i) := by
  intro i; rcases S3cases i with rfl | rfl <;> simp [v3, t0, t1, insideWindow] <;> norm_num
lemma v3_count : (Finset.univ.filter (fun i : S3 => 0 ≤ v3 i)).card = 1 := by
  rw [Finset.card_eq_one]; refine ⟨t0, ?_⟩
  ext i; simp only [Finset.mem_filter, Finset.mem_univ, true_and, Finset.mem_singleton]
  rcases S3cases i with rfl | rfl <;> simp [v3, t0, t1]

lemma X3_CLC : CorrectedLocalConditions μB X3 2 S3 v3 := by
  refine ⟨fun i => measurable_of_countable _, ?_, ?_⟩
  · intro i
    fin_cases i <;> simp [eventProbability, X3, insideWindow, μB_apply, S3] <;> norm_num
  · intro σ
    have hc : ((S3.card.factorial : ℕ) : ℝ) = 2 := by rw [S3card]; simp
    rcases perm3 σ with rfl | rfl
    · have : {ω : Bool | ∀ i : S3, X3 ω i = v3 ((1 : Equiv.Perm S3) i)} = {true} := by
        ext ω; constructor
        · intro h; have h0 := h t0
          cases ω
          · exfalso; simp [X3, v3, t0] at h0; norm_num at h0
          · rfl
        · rintro rfl i; rcases S3cases i with rfl | rfl <;> simp [X3, v3, t0, t1]
      show (μB _).toReal = _
      rw [this, μB_apply, hc]; simp [ENNReal.toReal_inv]
    · have : {ω : Bool | ∀ i : S3, X3 ω i = v3 ((Equiv.swap t0 t1) i)} = {false} := by
        ext ω; constructor
        · intro h; have h0 := h t0
          rw [Equiv.swap_apply_left] at h0
          cases ω
          · rfl
          · exfalso; simp [X3, v3, t0, t1] at h0; norm_num at h0
        · rintro rfl i; rcases S3cases i with rfl | rfl <;> simp [X3, v3, t0, t1]
      show (μB _).toReal = _
      rw [this, μB_apply, hc]; simp [ENNReal.toReal_inv]

def ya : TreatmentVector 3 := ![true, false, false]
def yb : TreatmentVector 3 := ![false, true, false]
lemma yab : ya ≠ yb := by decide
noncomputable def A3 : FiniteDistribution (TreatmentVector 3) := ⟨{ya, yb}, fun _ => 1/2⟩
lemma A3_dist : IsDistribution A3 := by
  refine ⟨by simp [A3], ?_⟩
  simp [A3, Finset.sum_pair yab]; norm_num
lemma thr_t : thresholdAssignment (X3 true) = ya := by
  funext i; fin_cases i <;> simp [thresholdAssignment, X3, ya]
lemma thr_f : thresholdAssignment (X3 false) = yb := by
  funext i; fin_cases i <;> simp [thresholdAssignment, X3, yb]

lemma A3_law : ∀ z : TreatmentVector 3, eventProbability μB (fun ω => thresholdAssignment (X3 ω) = z) =
    if z ∈ A3.atoms then A3.weight z else 0 := by
  intro z
  simp only [eventProbability, μB_apply, Set.mem_ofPred_eq, thr_t, thr_f]
  by_cases ha : ya = z
  · subst ha; simp [A3, yab, yab.symm, ENNReal.toReal_inv]
  · by_cases hb : yb = z
    · subst hb; simp [A3, yab, yab.symm, ENNReal.toReal_inv]
    · have h1 : z ∉ A3.atoms := by simp [A3]; exact ⟨Ne.symm ha, Ne.symm hb⟩
      simp [ha, hb, h1]

lemma outside_fixed : ∀ᵐ ω ∂μB, ∀ i ∉ S3, thresholdAssignment (X3 ω) i = (![false, false, false] : TreatmentVector 3) i := by
  refine Filter.Eventually.of_forall (fun ω i hi => ?_)
  have : i = 2 := by fin_cases i <;> simp_all [S3]
  subst this; cases ω <;> simp [thresholdAssignment, X3]

-- nonvacuity of T3FixedOutsidePart (with a genuine outside subject held at control)
example (y : PotentialOutcomes 3) :=
  (SAP.P014.T3.{0}).2.2.1 μB X3 2 S3 v3 A3 A3_dist ![false, false, false] y 1 (by norm_num) v3_inj v3_win
    (by norm_num) (by rw [S3card]; norm_num) v3_count X3_CLC A3_law outside_fixed
-- the same data also discharge T2's score part and T7's local part, with an outside subject
example := SAP.P014.T2.{0}.2 μB X3 2 S3 v3 1 (by norm_num) v3_inj v3_win (by norm_num)
    (by rw [S3card]; norm_num) v3_count X3_CLC
example := SAP.P014.T7.{0}.2.1 μB X3 2 S3 v3 1 (by norm_num) (by rw [S3card]) v3_inj v3_win (by norm_num)
    (by rw [S3card]; norm_num) v3_count X3_CLC
