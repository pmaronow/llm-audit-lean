-- Conditional independence of a function of X, for the exact package definition.
lemma condIndep_function {A B C : Type*} [Fintype A]
    (p : A → ℝ) (X : A → B) (g : B → C) (D : A → ℝ) :
    ConditionalIndependent p X (fun a => g (X a)) D := by
  classical
  intro x u d
  by_cases hu : g x = u
  · have h₁ : eventMass p (fun a => X a = x ∧ g (X a) = u ∧ D a = d) =
        eventMass p (fun a => X a = x ∧ D a = d) := by
      unfold eventMass
      apply Finset.sum_congr rfl
      intro a _
      by_cases hx : X a = x <;> simp [hx, hu]
    have h₂ : eventMass p (fun a => X a = x ∧ g (X a) = u) =
        eventMass p (fun a => X a = x) := by
      unfold eventMass
      apply Finset.sum_congr rfl
      intro a _
      by_cases hx : X a = x <;> simp [hx, hu]
    rw [h₁, h₂, mul_comm]
  · have h₁ : eventMass p (fun a => X a = x ∧ g (X a) = u ∧ D a = d) = 0 := by
      unfold eventMass
      apply Finset.sum_eq_zero
      intro a _
      by_cases hx : X a = x <;> simp [hx, hu]
    have h₂ : eventMass p (fun a => X a = x ∧ g (X a) = u) = 0 := by
      unfold eventMass
      apply Finset.sum_eq_zero
      intro a _
      by_cases hx : X a = x <;> simp [hx, hu]
    rw [h₁, h₂, zero_mul, zero_mul]

-- T2: a genuine infinite sequence of independent, uniformly distributed draws.
def atomLaw : Measure (Fin 4) := (PMF.uniformOfFintype (Fin 4)).toMeasure

instance : IsProbabilityMeasure atomLaw := by
  unfold atomLaw
  infer_instance

def iidLaw : Measure (ℕ → Fin 4) := Measure.infinitePi (fun _ : ℕ => atomLaw)

instance : IsProbabilityMeasure iidLaw := by
  unfold iidLaw
  infer_instance

example : UniformIndependentSampling iidLaw (fun i s => s i) := by
  refine ⟨fun i => measurable_pi_apply i, ?_, ?_⟩
  · exact iIndepFun_infinitePi (X := fun _ => id) (fun _ => measurable_id)
  · intro i a
    have h := (measurePreserving_eval_infinitePi (fun _ : ℕ => atomLaw) i).measure_preimage
      (measurableSet_singleton a).nullMeasurableSet
    change iidLaw {s | s i = a} = atomLaw {a} at h
    rw [h, atomLaw, PMF.toMeasure_apply_singleton _ _ (measurableSet_singleton a)]
    fin_cases a <;> norm_num [PMF.uniformOfFintype_apply, uniformProbability,
      ENNReal.ofReal_div_of_pos]
