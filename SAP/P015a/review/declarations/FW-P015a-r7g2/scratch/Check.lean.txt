-- T8: nonempty Bernoulli calculus premises with a genuine varying inverse link.
example :
    let W : Matrix (Fin 2) (Fin 2) ℝ := !![1,0;1,1]
    let Y : Fin 2 → ℝ := ![0,1]
    (∀ i, Y i = 0 ∨ Y i = 1) ∧ (∀ i, W i 0 = 1) ∧
    IsOpen (Set.univ : Set ℝ) ∧ DifferentiableOn ℝ inverseLogit Set.univ ∧
    (∀ η ∈ (Set.univ : Set ℝ), 0 < inverseLogit η ∧ inverseLogit η < 1) ∧
    (0 : Fin 2 → ℝ) ∈ admissibleParameters W Set.univ ∧
    inverseLogit 0 = 1/2 ∧
    DifferentiableAt ℝ (deriv inverseLogit) 0 := by
  dsimp
  refine ⟨?_, ?_, isOpen_univ, ?_, ?_, ?_, ?_, ?_⟩
  · intro i; fin_cases i <;> simp
  · intro i; fin_cases i <;> rfl
  · intro η _; exact (Real.hasDerivAt_sigmoid η).differentiableAt.differentiableWithinAt
  · intro η _; exact ⟨Real.sigmoid_pos η, Real.sigmoid_lt_one η⟩
  · simp [admissibleParameters]
  · norm_num [inverseLogit, Real.sigmoid_def]
  · change DifferentiableAt ℝ (deriv Real.sigmoid) 0
    rw [Real.deriv_sigmoid]
    exact (Real.hasDerivAt_sigmoid 0).differentiableAt.mul
      (differentiableAt_const 1 |>.sub (Real.hasDerivAt_sigmoid 0).differentiableAt)

-- Independent protection against the real-sInf default for the normal quantile.
example : Set.Nonempty {a : ℝ | 3/4 ≤ standardNormalCDF a} ∧
    BddBelow {a : ℝ | 3/4 ≤ standardNormalCDF a} := by
  have ht : ∀ᶠ a in atTop, (3/4 : ℝ) < standardNormalCDF a :=
    (tendsto_cdf_atTop (gaussianReal 0 1)).eventually
      (eventually_gt_nhds (by norm_num : (3/4 : ℝ) < 1))
  have hb : ∀ᶠ a in atBot, standardNormalCDF a < (3/4 : ℝ) :=
    (tendsto_cdf_atBot (gaussianReal 0 1)).eventually
      (eventually_lt_nhds (by norm_num : (0 : ℝ) < 3/4))
  constructor
  · obtain ⟨a, ha⟩ := ht.exists
    exact ⟨a, ha.le⟩
  · obtain ⟨b, hb⟩ := hb.exists
    refine ⟨b, ?_⟩
    intro a ha
    by_contra h
    have hab : a < b := lt_of_not_ge h
    have hm := monotone_cdf (gaussianReal 0 1) hab.le
    change standardNormalCDF a ≤ standardNormalCDF b at hm
    change (3/4 : ℝ) ≤ standardNormalCDF a at ha
    linarith
