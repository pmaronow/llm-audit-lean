lemma firstGramInv : (firstGroupDesign.transpose * firstGroupDesign)⁻¹ =
    (!![1, -1; -1, 3/2] : Matrix (Fin 2) (Fin 2) ℝ) := by
  apply Matrix.inv_eq_left_inv
  ext i j; fin_cases i <;> fin_cases j <;>
    norm_num [firstGroupDesign, Matrix.mul_apply, Fin.sum_univ_succ]

lemma secondGramInv : (secondGroupDesign.transpose * secondGroupDesign)⁻¹ =

    norm_num [firstGroupDesign, secondGroupDesign, Matrix.mulVec, dotProduct,

-- An explicit centered, positive-variance, discrete law with a positive atom at 0.
-- It can be used independently for slopes and unit errors in the T9 setting.
def centeredMass : Fin 3 → ℝ := ![3/4, 1/8, 1/8]
def centeredValue : Fin 3 → ℝ := ![0, -2, 2]

example : IsFiniteLaw centeredMass ∧
    finiteMean centeredMass centeredValue = 0 ∧
    finiteMean centeredMass (fun i => centeredValue i ^ 2) = 1 ∧
    0 < centeredMass 0 := by
  refine ⟨⟨?_, ?_⟩, ?_, ?_, ?_⟩
  · intro i; fin_cases i <;> norm_num [centeredMass]
  all_goals norm_num [centeredMass, centeredValue, finiteMean, Fin.sum_univ_succ]

    (∀ _g : Fin 1, (0 : ℝ) < 1) ∧

    (fun (_k : ℕ) (_g : Fin 1) (j : Fin 2) => if j = 1 then (1 : ℝ) else 0) := by
  refine ⟨1, by norm_num, ?_⟩
  intro k g j

  split_ifs <;> norm_num

example : ∀ (g : Fin 1) (j : Fin 2), j ≠ 1 →

  have hz : (1 : Matrix (Fin 2) (Fin 2) ℝ) 1 j = 0 := by

  simpa only [hz] using (tendsto_const_nhds :

    affineCoefficients (0 : Fin 2) (1/2) (1/4) ![-2, 4] = ![0, 1] := by
  refine ⟨?_, ?_, ?_, by norm_num, ?_, ?_⟩

  · intro i; fin_cases i <;> norm_num [design]
  · norm_num [design, Matrix.det_fin_two, Matrix.mul_apply, Fin.sum_univ_succ]
  · ext j; fin_cases j <;>
      norm_num [residualEquation, design, Matrix.mulVec, dotProduct, Fin.sum_univ_succ]
  · ext j; fin_cases j <;> norm_num [affineCoefficients]
