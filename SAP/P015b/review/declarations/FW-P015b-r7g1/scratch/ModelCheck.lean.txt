import SAP.P015b.Challenge

open SAP.P015b MeasureTheory
open scoped ENNReal NNReal

noncomputable section

namespace AuditModel

def fair : Measure (Fin 2) :=
  (1/2 : ℝ≥0) • Measure.dirac 0 + (1/2 : ℝ≥0) • Measure.dirac 1

instance : IsProbabilityMeasure fair := by
  constructor
  change (1/2 : ℝ≥0) • ((Measure.dirac (0 : Fin 2)) Set.univ) +
    (1/2 : ℝ≥0) • ((Measure.dirac (1 : Fin 2)) Set.univ) = 1
  simp only [measure_univ]
  rw [← add_smul]
  norm_num

def binary (s : Fin 2) : ℝ := if s = 1 then 1 else 0

example : CommonLogisticModel fair 0
    (fun _ => binary) (fun _ _ => 0) (fun _ _ _ => 0) := by
  refine ⟨?_, ?_, ?_, ?_, 0, 0, (fun _ => 0), ?_⟩
  · intro i; fun_prop
  · intro i; fun_prop
  · intro i; fun_prop
  · intro i
    exact Filter.Eventually.of_forall (fun s => by fin_cases s <;> norm_num [binary])
  · intro i
    simp only [zero_mul, zero_add, dotProduct_zero, sigmoidMean,
      Real.sigmoid_zero]
    have hm : MeasurableSpace.comap
        (fun (_s : Fin 2) => ((0 : ℝ), (fun _ : Fin 0 => (0 : ℝ))))
        inferInstance = ⊥ := MeasurableSpace.comap_const _
    rw [hm, condExp_bot]
    have hi : (∫ s, (if binary s = 1 then (1 : ℝ) else 0) ∂fair) = 1/2 := by
      rw [fair, integral_add_measure]
      · norm_num [integral_smul_measure, binary, smul_eq_mul]
        norm_num [NNReal.smul_def]
      · exact Integrable.of_finite
      · exact Integrable.of_finite
    rw [hi]
    norm_num

end AuditModel
