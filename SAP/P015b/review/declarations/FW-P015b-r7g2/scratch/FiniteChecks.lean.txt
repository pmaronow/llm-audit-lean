set_option linter.unusedSimpArgs false

-- Expand the grouped T11 data into 200 actual binary observations.
abbrev N := Fin 2 × Fin 100
def rawD (i : N) : ℝ := (i.1 : ℕ)
def rawY (i : N) : ℝ :=
  if i.1 = 0 then (if i.2 = 0 then 1 else 0)
  else 1 - (if i.2 = 0 then 1 else 0)
def rawW : Matrix N (Fin 2) ℝ := fun i => ![1, rawD i]

example : Fintype.card N = 200 := by norm_num [N]
example : ∀ i, (rawD i = 0 ∨ rawD i = 1) ∧ (rawY i = 0 ∨ rawY i = 1) := by
  rintro ⟨i,j⟩
  fin_cases i <;> by_cases h : j = 0 <;> simp [rawD, rawY, h]
example : (∑ j : Fin 100, rawY (0,j)) = 1 ∧
    (∑ j : Fin 100, rawY (1,j)) = 99 := by
  norm_num [rawY, Finset.sum_sub_distrib]
example : (∑ i : N, rawY i) / (200 : ℝ) = 1/2 := by
  norm_num [Fintype.sum_prod_type, Fin.sum_univ_two, rawY, Finset.sum_sub_distrib]
example : (rawW.transpose * rawW).det = 10000 := by
  norm_num [Matrix.det_fin_two, Matrix.mul_apply, Matrix.transpose_apply,
    rawW, rawD, Fintype.sum_prod_type, Fin.sum_univ_two]
example (g : ℝ → ℝ) (theta : Fin 2 → ℝ) :
    (∑ i : N, rawW i 1 * (rawY i - g (rawW.mulVec theta i))) =
      99 - 100 * g (theta 0 + theta 1) := by
  simp [Fintype.sum_prod_type, Fin.sum_univ_two, rawW, rawD,
    rawY, Matrix.mulVec, dotProduct, Finset.sum_sub_distrib]
  ring

-- Check the fair-bit witness independently of the certified target theorem.
example : WholeConfoundingWitness := by
  dsimp [WholeConfoundingWitness]
  refine ⟨?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_⟩
  · norm_num [IsFiniteLaw, Fin.sum_univ_two]
  · intros; rfl
  · intro i
    norm_num [finiteCellMean, finiteMass, finiteMean, Fin.sum_univ_two]
  · norm_num [finiteMean, Fin.sum_univ_two]
  · norm_num [finiteMean, Fin.sum_univ_two]
  · intro i
    norm_num [finiteCellMean, finiteMass, finiteMean, Fin.sum_univ_two]
  · intro i
    norm_num [finiteCellMean, finiteMass, finiteMean, Fin.sum_univ_two]
  · intro h
    have bad := h () (0,0) 0
    norm_num [finiteMass, Fin.sum_univ_two] at bad
  · norm_num [populationCoefficients, finiteMean, Fin.sum_univ_two,
      linearOutcome, Matrix.inv_def, Matrix.det_fin_two, Matrix.adjugate_fin_two,
      Matrix.mulVec, dotProduct, Matrix.smul_apply]
  · norm_num [finiteMean, Fin.sum_univ_two]
  · norm_num [finiteMean, Fin.sum_univ_two]
