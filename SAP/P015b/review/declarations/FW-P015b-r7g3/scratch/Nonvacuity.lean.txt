lemma log_pair_bound (p : ℝ) (hp : 0 < p) (hq : p < 1) :
    Real.log p + Real.log (1-p) ≤ Real.log (1/2) + Real.log (1/2) := by
  rw [← Real.log_mul (ne_of_gt hp) (ne_of_gt (sub_pos.mpr hq)),
    ← Real.log_mul (by norm_num : (1/2 : ℝ) ≠ 0) (by norm_num : (1/2 : ℝ) ≠ 0)]
  apply Real.log_le_log (mul_pos hp (sub_pos.mpr hq))
  nlinarith [sq_nonneg (p-1/2)]

def logisticW : Matrix (Fin 4) (Fin 2) ℝ := !![1,0;1,0;1,1;1,1]
def logisticY : Fin 4 → ℝ := ![0,1,0,1]

-- All premises of the universal logistic implication, with a nonempty full-rank sample.
example : (∀ i, logisticY i = 0 ∨ logisticY i = 1) ∧
    (logisticW.transpose * logisticW).det ≠ 0 ∧
    (∀ γ : Fin 2 → ℝ,
      binaryLogLikelihood logisticW (fun _ => 1) logisticY sigmoidMean γ ≤
      binaryLogLikelihood logisticW (fun _ => 1) logisticY sigmoidMean 0) := by
  refine ⟨?_, ?_, ?_⟩
  · intro i; fin_cases i <;> norm_num [logisticY]
  · norm_num [logisticW, Matrix.det_fin_two, Matrix.mul_apply, Fin.sum_univ_succ]
  · intro γ
    have h0 := log_pair_bound (Real.sigmoid (γ 0))
      (Real.sigmoid_pos _) (Real.sigmoid_lt_one _)
    have h1 := log_pair_bound (Real.sigmoid (γ 0 + γ 1))
      (Real.sigmoid_pos _) (Real.sigmoid_lt_one _)
    norm_num [binaryLogLikelihood, binomialLogLikelihoodSAP, logisticW, logisticY,
      sigmoidMean, Matrix.mulVec, dotProduct, Fin.sum_univ_succ, Real.sigmoid_zero] at *
    linarith

-- A literal finite probability space meeting all substantive hypotheses of the positive implication.

  constructor
  norm_num [missingPairPopulation, Measure.smul_apply, Measure.add_apply]
  calc
    _ = (4 : ℝ≥0∞) * 4⁻¹ := by ring
    _ = 1 := ENNReal.mul_inv_cancel (by norm_num) (by norm_num)

  · norm_num [missingPairPopulation, missingPairProfile, Measure.smul_apply,

-- Realize the population objective as normalized finite binary likelihoods along epsilon=1/(16m).
-- The total number of observations is 768m; each row-type count is 64m times cellUnits.
def cellUnits : Fin 6 → ℤ := ![3,1,2,2,1,3]
def baselineUnits : Fin 6 → ℤ := ![5,5,2,2,5,5]
def residualUnits : Fin 6 → ℤ := ![-1,3,-2,2,-3,1]
def rowCount (m : ℕ) (i : Fin 6) : ℤ := 64*(m : ℤ)*cellUnits i
def successes (m : ℕ) (i : Fin 6) : ℤ :=
  cellUnits i * (4*(m : ℤ)*baselineUnits i+residualUnits i)
def flatSuccesses (m : ℕ) (i : Fin 6) : ℤ :=
  cellUnits i * (16*(m : ℤ)+residualUnits i)

example (m : ℕ) (hm : 1 ≤ m) : ∀ i,
    0 ≤ successes m i ∧ successes m i ≤ rowCount m i ∧
    0 ≤ flatSuccesses m i ∧ flatSuccesses m i ≤ rowCount m i ∧
    ((rowCount m i : ℤ) : ℝ) = (768*(m : ℝ))*squarePopulation i ∧
    ((successes m i : ℤ) : ℝ) =
      (768*(m : ℝ))*squarePopulation i*squareProbability (1/16) (1/(16*(m : ℝ))) i ∧
    ((flatSuccesses m i : ℤ) : ℝ) =
      (768*(m : ℝ))*squarePopulation i*squareProbability 0 (1/(16*(m : ℝ))) i := by
  have hmi : (1 : ℤ) ≤ m := by exact_mod_cast hm
  have hmr : (0 : ℝ) < m := by exact_mod_cast (lt_of_lt_of_le (by decide : 0 < 1) hm)
  intro i
  fin_cases i <;>
    norm_num [successes, flatSuccesses, rowCount, cellUnits, baselineUnits, residualUnits,
      squarePopulation, squareProbability, squareBaseline, squareProfile, squareTreatment,
      squarePropensity] <;>
    (repeat' constructor) <;>
    first | omega | (field_simp [ne_of_gt hmr] <;> ring)
