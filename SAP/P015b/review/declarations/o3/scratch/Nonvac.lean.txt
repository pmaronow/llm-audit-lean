/-- T18 / LogisticMLStationarity: the hypotheses (binary Y, θ a global logit-likelihood maximizer)
are jointly satisfiable: N = 2, K = 0, W = [[1,0],[1,0]], Y = (0,1), θ = 0. -/
example : ∃ (N K : ℕ) (W : Matrix (Fin N) (Fin (K+2)) ℝ) (Y : Fin N → ℝ)
    (_ : ∀ i, Y i = 0 ∨ Y i = 1) (θ : Fin (K+2) → ℝ),
    ∀ γ, binaryLogLikelihood W (fun _ => 1) Y sigmoidMean γ ≤
      binaryLogLikelihood W (fun _ => 1) Y sigmoidMean θ := by
  refine ⟨2, 0, !![1,0;1,0], ![0,1], ?_, 0, ?_⟩
  · intro i; fin_cases i <;> simp
  · intro γ
    simp only [binaryLogLikelihood, binomialLogLikelihoodSAP, sigmoidMean, Fin.sum_univ_two,
      Matrix.mulVec, dotProduct, Matrix.of_apply, Matrix.cons_val', Matrix.cons_val_zero,
      Matrix.cons_val_one, Matrix.empty_val', Matrix.cons_val_fin_one, Pi.zero_apply, mul_zero,
      Real.sigmoid_zero]
    set s := Real.sigmoid (γ 0) with hs
    have h0 : 0 < s := Real.sigmoid_pos _
    have h1 : s < 1 := Real.sigmoid_lt_one _
    have key : Real.log s + Real.log (1 - s) ≤ Real.log 2⁻¹ + Real.log (1 - 2⁻¹) := by
      rw [← Real.log_mul h0.ne' (by linarith), ← Real.log_mul (by norm_num) (by norm_num)]
      exact Real.log_le_log (by nlinarith) (by nlinarith [sq_nonneg (s - 1/2)])
    have e2 : Real.sigmoid ((0 : Fin 2 → ℝ) 0 * 1 + (0 : Fin 2 → ℝ) 1 * 0) = 2⁻¹ := by simp
    simp only [Pi.zero_apply, zero_mul, mul_zero, add_zero, Real.sigmoid_zero, mul_one, one_mul, zero_add, sub_zero, sub_self] at *
    norm_num at key ⊢
    have hsum : (∑ x, (![1, 0] : Fin 2 → ℝ) x * γ x) = γ 0 := by simp [Fin.sum_univ_two]
    rw [hsum, ← hs]
    linarith

/-- T17 / DiscretePositivityImplications: hypotheses jointly satisfiable on Ω = Fin 2,
B = Unit, P uniform, X constant, D = identity cast, x = (), d = 0, d' = 1. -/
noncomputable def P2 : Measure (Fin 2) := (2⁻¹ : ℝ≥0∞) • (Measure.dirac 0 + Measure.dirac 1)

instance : IsProbabilityMeasure P2 := ⟨by
  simp [P2]
  rw [← two_mul, ENNReal.mul_inv_cancel] <;> norm_num⟩

example : ∃ (P : Measure (Fin 2)) (_ : IsProbabilityMeasure P) (X : Fin 2 → Unit) (D : Fin 2 → ℝ),
    Measurable X ∧ Measurable D ∧ MemLp D 2 P ∧ 0 < P {s | X s = ()} ∧
    (0 : ℝ) ≠ 1 ∧ 0 < profileProbability P X D () 0 ∧ 0 < profileProbability P X D () 1 := by
  refine ⟨P2, inferInstance, fun _ => (), fun i => (i : ℝ), measurable_const,
    measurable_of_finite _, ?_, ?_, by norm_num, ?_, ?_⟩
  · exact memLp_of_bounded (a := 0) (b := 1) (Filter.Eventually.of_forall (fun i => by
      fin_cases i <;> simp)) (measurable_of_finite _).aestronglyMeasurable 2
  · simp
  · have : {s : Fin 2 | ((s : ℕ) : ℝ) = 0} = {0} := by
      ext s; fin_cases s <;> simp
    have hv : profileProbability P2 (fun _ => ()) (fun i : Fin 2 => ((i : ℕ) : ℝ)) () 0 = 1/2 := by
      simp [profileProbability, profileLaw, this, P2]
      rw [← two_mul, ENNReal.mul_inv_cancel] <;> norm_num
    rw [hv]; norm_num
  · have : {s : Fin 2 | ((s : ℕ) : ℝ) = 1} = {1} := by
      ext s; fin_cases s <;> simp
    have hv : profileProbability P2 (fun _ => ()) (fun i : Fin 2 => ((i : ℕ) : ℝ)) () 1 = 1/2 := by
      simp [profileProbability, profileLaw, this, P2]
      rw [← two_mul, ENNReal.mul_inv_cancel] <;> norm_num
    rw [hv]; norm_num
