private lemma fourVariance :
    sampleVariance (centeredExposure (similarity (fun k => some (scores4 k)))
      matchedBank4 (complete4 1)) = 1/48 := by
  simp only [sampleVariance, centeredExposure, conditionalMean, exposureSum,
    similarity, scores4, matchedBank4, complete4, bit, pooledMean, Fin.sum_univ_succ]
  norm_num

    0 < sampleSD (centeredExposure (similarity (fun k => some (scores4 k)))

  exact Real.sqrt_pos.mpr (by norm_num)

-- T4: a nonempty singleton design, covered bank, symmetric zero-diagonal
-- similarity, positive exposure SD, and nonzero direct and indirect effects.
example : ∃ (S : Fin 4 → Fin 4 → ℝ) (bank : Fin 4 → Assignment 4)
    (Ω : Finset (Assignment 4)) (y0 : Outcome 4)
    (y : Assignment 4 → Outcome 4) (b1 b2 : ℝ),
    2 ≤ (4 : ℕ) ∧ (∀ i j, S i j = S j i) ∧ (∀ i, S i i = 0) ∧
    (∀ i a, ∃ b, bank b i = a) ∧ Ω.Nonempty ∧
    (∀ z ∈ Ω, 0 < sampleSD (centeredExposure S bank z)) ∧
    (∀ z ∈ Ω, ∀ i, y z i = y0 i + bit (z i)*b1 + b2*codedExposure S bank z i) ∧
    b1 ≠ 0 ∧ b2 ≠ 0 := by
  classical
  let S := similarity (fun k => some (scores4 k))
  refine ⟨S, matchedBank4, {complete4 1}, u4,
    (fun z i => u4 i + bit (z i)*2 + 3*codedExposure S matchedBank4 z i),
    2, 3, by decide, ?_, ?_, by decide, ?_, ?_, ?_, by norm_num, by norm_num⟩
  · intro i j
    fin_cases i <;> fin_cases j <;> norm_num [S, similarity, scores4]
  · intro i
    simp [S, similarity]
  · exact ⟨complete4 1, Finset.mem_singleton_self _⟩
  · intro z hz
    simp only [Finset.mem_singleton] at hz
    subst z
    exact fourSD
  · intro z hz i
    rfl
