-- No use of the Challenge theorems: infima are justified by a lower bound
-- on every coefficient vector and an explicitly attaining vector.
theorem inf_at {n k : ℕ} (u : Outcome n) (X : Fin n → Fin k → ℝ)
    (v : Fin k → ℝ) (c : ℝ)
    (hbound : ∀ b, c ≤ lsObjective u X b)
    (hatt : lsObjective u X v = c) : lsRSS u X = c := by
  have hne : (Set.range (lsObjective u X)).Nonempty := ⟨_, ⟨v, rfl⟩⟩
  have hbd : BddBelow (Set.range (lsObjective u X)) := by
    exact ⟨c, by rintro _ ⟨b, rfl⟩; exact hbound b⟩
  unfold lsRSS
  apply le_antisymm
  · calc
      sInf (Set.range (lsObjective u X)) ≤ lsObjective u X v := csInf_le hbd ⟨v, rfl⟩
      _ = c := hatt
  · exact le_csInf hne (by rintro _ ⟨b, rfl⟩; exact hbound b)
