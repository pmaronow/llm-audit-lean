-- (c)(i) claim edge case m = n: the claim's raw identity, which Lean's (c) does not assert for m = n
example : let s : Outcome 2 := ![0,1]
    ∀ i, exposureSum (similarity (fun k => some (s k))) (fun _ => true) i =
      (2:ℝ) - bit true - (1/2) * exposureSum (distance s) (fun _ => true) i := by
  intro s i; fin_cases i <;> simp [s, exposureSum, similarity, distance, bit, Fin.sum_univ_two] <;> norm_num
-- with m = 2 = n no bank satisfies hbank ∧ hm
example (N : ℕ) (bank : Fin N → Assignment 2) (hm : ∀ b, groupSize (bank b) true = 2) :
    ¬ ∀ i a, ∃ b, bank b i = a := by
  intro h
  obtain ⟨b, hb⟩ := h 0 false
  have := hm b
  simp only [groupSize, Fin.sum_univ_two, hb] at this
  split_ifs at this <;> simp_all
