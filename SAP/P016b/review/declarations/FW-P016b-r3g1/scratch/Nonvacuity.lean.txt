-- T12: a genuine assignment with two nonempty groups, directional outcomes,
-- and a strict drop in both KS statistics.
example :
    0 < groupSize (complete4 0) false ∧
    0 < groupSize (complete4 0) true ∧
    (∀ i : Fin 4, if complete4 0 i then
      u4 i ≤ (![2,3,2,3] : Outcome 4) i
      else (![2,3,2,3] : Outcome 4) i ≤ u4 i) ∧
    ks (![2,3,2,3] : Outcome 4) (complete4 0) < ks u4 (complete4 0) ∧
    signedKS (![2,3,2,3] : Outcome 4) (complete4 0) <
      signedKS u4 (complete4 0) := by
  simp only [groupSize, complete4, u4, ks, signedKS, ecdf,
    Fin.sum_univ_succ, List.ofFn_succ]
  norm_num [Fin.forall_fin_succ]

-- Strictly positive normalized weights used by T12 and T14(e).
example : ∃ P : FiniteLaw (Fin 6), (∀ i, 0 < P.weight i) ∧
    ∑ i, P.weight i = 1 := by
  refine ⟨uniformLaw (Fin 6), ?_, (uniformLaw (Fin 6)).total⟩
  intro i
  norm_num [uniformLaw]

-- T13: an actual balanced assignment and its distinct balanced complement.
def firstEight : Assignment 16 := fun i => decide (i.val < 8)
def lastEight : Assignment 16 := fun i => !(firstEight i)

example : groupSize firstEight true = 8 ∧
    groupSize firstEight false = 8 ∧
    groupSize lastEight true = 8 ∧ firstEight ≠ lastEight := by
  constructor
  · simp only [groupSize, firstEight, Fin.sum_univ_succ]; norm_num
  constructor
  · simp only [groupSize, firstEight, Fin.sum_univ_succ]; norm_num
  constructor
  · simp only [groupSize, lastEight, firstEight, Fin.sum_univ_succ]; norm_num
  intro h
  have h0 := congrFun h (0 : Fin 16)
  norm_num [firstEight, lastEight] at h0

example : Nonempty (Balanced 16 8) := by
  refine ⟨⟨firstEight, ?_⟩⟩
  simp only [groupSize, firstEight, Fin.sum_univ_succ]
  norm_num

-- T14(d) explicitly allows zero-weight atoms without losing normalization.
def zeroAtomLaw : FiniteLaw (Fin 2) where
  weight := ![1,0]
  nonneg := by intro i; fin_cases i <;> norm_num
  total := by norm_num [Fin.sum_univ_succ]

example : zeroAtomLaw.weight 1 = 0 ∧ ∑ i, zeroAtomLaw.weight i = 1 := by
  constructor
  · rfl
  · exact zeroAtomLaw.total

-- T15: independently verify the complete concrete counterexample proposition.
example : T15Statement := by

      groupSize, Fin.sum_univ_succ]

      additiveAdjust, multiplicativeAdjust, bit]

  · simp only [sampleECDF, Fin.sum_univ_succ]; norm_num
  constructor
  · intro t b

  · rintro ⟨t0, h1, h2⟩

-- T16: both groups nonempty, treated response nonzero, and all parameter
-- differences and denominator hypotheses hold simultaneously.
example : ∃ (Y : Outcome 2) (z : Assignment 2) (i : Fin 2)
    (t t' a a' : ℝ),
    0 < groupSize z false ∧ 0 < groupSize z true ∧
    z i = true ∧ Y i ≠ 0 ∧ a ≠ 0 ∧ a' ≠ 0 ∧ t ≠ t' ∧ a ≠ a' ∧
    additiveAdjust Y z t i ≠ additiveAdjust Y z t' i ∧
    multiplicativeAdjust Y z a i ≠ multiplicativeAdjust Y z a' i ∧
    ecdf (additiveAdjust Y z t) z false 9 = ecdf Y z false 9 ∧
    ecdf (multiplicativeAdjust Y z a) z false 9 = ecdf Y z false 9 := by
  refine ⟨![9,10], ![false,true], 1, 0, 1, 1, 2, ?_⟩
  simp only [groupSize, additiveAdjust, multiplicativeAdjust, bit, ecdf,
    Fin.sum_univ_succ]
  norm_num

example : sampleECDF (![1,3] : Outcome 2) 2 = 1/2 ∧
    (∃ t0 b0 : ℝ, (1/2 : ℝ) = t0 ∧ (2 : ℝ) = b0) ∧
    ¬ (∃ t0 : ℝ, (1/2 : ℝ) = t0 ∧ (2 : ℝ) = t0) := by
  constructor
  · simp only [sampleECDF, Fin.sum_univ_succ]; norm_num
  constructor
  · exact ⟨1/2, 2, rfl, rfl⟩
  · rintro ⟨t0, h1, h2⟩
    linarith
