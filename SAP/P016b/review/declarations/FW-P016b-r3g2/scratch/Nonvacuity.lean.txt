-- T19/T21: assignments with m treated units are in bijection with m-subsets.
def auditBalancedEquiv (n m : ℕ) :
    Balanced n m ≃ {s : Finset (Fin n) // s.card = m} := by
  classical
  refine
    { toFun := fun z => ⟨Finset.univ.filter (fun i => z.val i = true), ?_⟩
      invFun := fun s => ⟨fun i => decide (i ∈ s.val), ?_⟩
      left_inv := ?_
      right_inv := ?_ }
  · simpa only [Finset.card_filter, groupSize] using z.property
  · simpa [groupSize] using s.property
  · intro z
    apply Subtype.ext
    funext i
    simp
  · intro s
    apply Subtype.ext
    ext i
    simp

theorem auditBalancedCard (n m : ℕ) :
    Fintype.card (Balanced n m) = Nat.choose n m := by
  classical
  calc
    Fintype.card (Balanced n m) =
        Fintype.card {s : Finset (Fin n) // s.card = m} :=
      Fintype.card_congr (auditBalancedEquiv n m)
    _ = Nat.choose n m := by simp

-- T19: the uniform expectation never uses an empty type for m > 0.
example (m : ℕ) (hm : 0 < m) : Nonempty (Balanced (2*m) m) := by
  classical
  apply Fintype.card_pos_iff.mp
  rw [auditBalancedCard]
  exact Nat.choose_pos (by omega)

-- T21: construct a normalized law satisfying the *exact* binomial weight,
-- together with a nonzero symmetric binary network and all other inputs.
example : ∃ (P : FiniteLaw (Balanced 256 128))
    (S : Fin 256 → Fin 256 → ℝ)
    (y0 : Outcome 256) (T : Outcome 256 → Assignment 256 → ℝ),
    (∀ z, P.weight z = 1/(Nat.choose 256 128 : ℝ)) ∧
    (∀ i j, S i j = S j i) ∧
    (∀ i j, S i j = 0 ∨ S i j = 1) ∧
    S 0 1 = 1 ∧ y0 0 = 1 ∧ T = meanDifference := by
  classical
  have hpos : 0 < Fintype.card (Balanced 256 128) := by
    rw [auditBalancedCard]
    exact Nat.choose_pos (by decide)
  letI : Nonempty (Balanced 256 128) := Fintype.card_pos_iff.mp hpos
  refine ⟨uniformLaw (Balanced 256 128),
    (fun i j => if i = j then 0 else 1),
    (fun i => (i.val : ℝ)+1), meanDifference, ?_⟩
  refine ⟨?_, ?_, ?_, ?_, ?_, rfl⟩
  · intro z
    simp only [uniformLaw, auditBalancedCard]
  · intro i j
    simp [eq_comm]
  · intro i j
    change (if i = j then (0 : ℝ) else 1) = 0 ∨
      (if i = j then (0 : ℝ) else 1) = 1
    split_ifs <;> simp
  · norm_num
  · norm_num
