def G : Measure ℝ := fairPair (-1) 1
instance : IsProbabilityMeasure G := fairPair_probability (-1) 1

lemma tau_integrable : Integrable (fun x : ℝ => x) G := by
  apply Integrable.add_measure
  · exact (integrable_dirac (by simp)).smul_measure (by norm_num)
  · exact (integrable_dirac (by simp)).smul_measure (by norm_num)

lemma mu_zero : populationMean G (fun x : ℝ => x) = 0 := by
  have ha : Integrable (fun x : ℝ => x)
      ((1 / 2 : ℝ≥0∞) • Measure.dirac (-1)) :=
    (integrable_dirac (by simp)).smul_measure (by norm_num)
  have hb : Integrable (fun x : ℝ => x)
      ((1 / 2 : ℝ≥0∞) • Measure.dirac 1) :=
    (integrable_dirac (by simp)).smul_measure (by norm_num)
  unfold populationMean G fairPair
  rw [integral_add_measure ha hb, integral_smul_measure, integral_smul_measure]
  norm_num

lemma support_pair : effectSupport G (fun x : ℝ => x) = {-1, 1} := by
  rw [effectSupport, Measure.map_id']
  apply Set.Subset.antisymm
  · apply Measure.support_subset_of_isClosed (isClosed_singleton.union isClosed_singleton)
    change ∀ᵐ x ∂G, x ∈ ({-1, 1} : Set ℝ)
    rw [ae_iff]
    simp [G, fairPair, Measure.add_apply, Measure.smul_apply]
  · intro x hx
    simp only [mem_insert_iff, mem_singleton_iff] at hx
    rcases hx with rfl | rfl
    · rw [Measure.mem_support_iff_forall]
      intro U hU
      have hu : (-1 : ℝ) ∈ U := mem_of_mem_nhds hU
      simp only [G, fairPair, Measure.add_apply, Measure.smul_apply,
        Measure.dirac_apply_of_mem hu, smul_eq_mul, mul_one]
      exact lt_of_lt_of_le (by norm_num : (0 : ℝ≥0∞) < 1 / 2) le_self_add
    · rw [Measure.mem_support_iff_forall]
      intro U hU
      have hu : (1 : ℝ) ∈ U := mem_of_mem_nhds hU
      simp only [G, fairPair, Measure.add_apply, Measure.smul_apply,
        Measure.dirac_apply_of_mem hu, smul_eq_mul, mul_one]
      exact lt_of_lt_of_le (by norm_num : (0 : ℝ≥0∞) < 1 / 2) le_add_self

lemma radius_one : effectRadius G (fun x : ℝ => x) = 1 := by
  simp only [effectRadius, support_pair, mu_zero, supportSup, supportInf,
    Set.image_insert_eq, Set.image_singleton, sSup_insert, sSup_singleton,
    sInf_insert, sInf_singleton]
  have h : ((-1 : ℝ) : EReal) ≤ ((1 : ℝ) : EReal) := by
    norm_cast
  rw [max_eq_right h, min_eq_left h]
  change (min (((1 : ℝ) : EReal) - ((0 : ℝ) : EReal))
    (((0 : ℝ) : EReal) - ((-1 : ℝ) : EReal))).toENNReal = 1
  rw [← EReal.coe_sub, ← EReal.coe_sub]
  norm_num
  change (((1 : ℝ) : EReal).toENNReal) = 1
  rw [EReal.real_coe_toENNReal]
  norm_num

def P : Measure ℝ := gaussianReal 0 1
instance : IsProbabilityMeasure P := inferInstanceAs (IsProbabilityMeasure (gaussianReal 0 1))
def hat (n : ℕ) (z : ℝ) : ℝ := z / sqrtN n
def Z (_ : ℕ) (z : ℝ) : ℝ := z
def eps (_ : ℕ) (_ : ℝ) : ℝ := 0

lemma scaled_hat (n : ℕ) (hn : 1 ≤ n) (z : ℝ) :
    sqrtN n * (hat n z - 0) = z := by
  have hs : sqrtN n ≠ 0 := ne_of_gt (Real.sqrt_pos.mpr (by exact_mod_cast hn))
  dsimp [hat]
  field_simp
  ring

lemma root_normal : RootNNormal P hat 0 := by
  refine ⟨1, by norm_num, ?_⟩
  refine { forall_aemeasurable := ?_, tendsto := ?_ }
  · intro n
    dsimp [hat]
    fun_prop
  · let Q : ProbabilityMeasure ℝ := (gaussianReal 0 1).toProbabilityMeasure
    have hconst : Tendsto (β := ProbabilityMeasure ℝ) (fun _ : ℕ => Q) atTop (𝓝 Q) :=
      tendsto_const_nhds
    simp only [Measure.map_id']
    apply hconst.congr'
    filter_upwards [eventually_ge_atTop 1] with n hn
    apply Subtype.ext
    change gaussianReal 0 1 = P.map (fun z => sqrtN n * (hat n z - 0))
    have hf : (fun z => sqrtN n * (hat n z - 0)) = (fun z : ℝ => z) :=
      funext (scaled_hat n hn)
    rw [hf]
    exact Measure.map_id'.symm

lemma normal_remainder : NormalRemainder P hat Z eps 0 1 := by
  refine ⟨?_, ?_, ?_⟩
  · intro n hn
    constructor
    · exact measurable_id
    · exact Measure.map_id'
  · intro n hn
    exact ⟨measurable_const, by simp [eps]⟩
  · intro n hn
    filter_upwards [] with z
    simpa [Z, eps] using scaled_hat n hn z

lemma small_remainder : SmallRemainderMS P eps := by
  simpa [SmallRemainderMS, eps] using
    (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (𝓝 0))

-- This also supplies every population and estimator hypothesis of T13 and T14.
example : Measurable (fun x : ℝ => x) ∧ Integrable (fun x : ℝ => x) G ∧

-- All hypotheses of T15, including the remainder moment rate, are satisfied.

def p (n : ℕ) : ℝ := 1 / ((n : ℝ) + 1)
lemma p_valid : (∀ n : ℕ, 1 ≤ n → 0 < p n ∧ p n ≤ 1) ∧
    Tendsto p atTop (𝓝 0) := by
  constructor
  · intro n hn
    have hn0 : (0 : ℝ) ≤ n := Nat.cast_nonneg n
    constructor <;> dsimp [p]
    · positivity
    · exact (div_le_one (by positivity)).mpr (by linarith)
  · have htn : Tendsto (fun n : ℕ => (n : ℝ)) atTop atTop := tendsto_natCast_atTop_atTop
    have ht : Tendsto (fun n : ℕ => (n : ℝ) + 1) atTop atTop :=
      tendsto_atTop_mono (fun n => by linarith) htn
    change Tendsto (fun n : ℕ => 1 / ((n : ℝ) + 1)) atTop (𝓝 0)
    simpa only [one_div, Function.comp_def] using tendsto_inv_atTop_zero.comp ht

-- The p and r inputs to the rare-outlier and no-universal-rate clauses are genuine.
example : (∀ n : ℕ, 1 ≤ n → 0 < p n ∧ p n ≤ 1) ∧ Tendsto p atTop (𝓝 0) := p_valid
example : (∀ n : ℕ, 1 ≤ n → 0 < p n) ∧ Tendsto p atTop (𝓝 0) :=
  ⟨fun n hn => (p_valid.1 n hn).1, p_valid.2⟩
