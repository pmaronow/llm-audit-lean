-- Independent, satisfiable input fixture for the universal single-record claims.
def checkKernel : Kernel Unit 1 :=
  ⟨Measure.dirac (), fun _ _ => ![1], fun _ _ => 0⟩

example : Standing checkKernel := by
  refine ⟨by dsimp [checkKernel]; infer_instance,
    measurable_const, measurable_const, ?_, ?_, ?_, ?_⟩
  · intro u v
    exact ⟨rfl, rfl⟩
  · refine ⟨1, by norm_num, Filter.Eventually.of_forall ?_⟩
    intro uv
    norm_num [checkKernel, pi_norm_le_iff_of_nonneg (by norm_num : (0:ℝ) ≤ 1)]
  · exact ⟨0, fun _ _ => rfl⟩
  · norm_num [populationGram, meanMat, checkKernel, outer, Matrix.det_fin_one]

-- Bounded, positive-count repeated data at an actual nonempty sample size.
example : RepeatedAtSize (singleRecord checkKernel) 2 := by
  refine ⟨by dsimp [singleRecord, checkKernel]; infer_instance,
    ?_, ?_, ?_⟩
  · intro d
    norm_num [singleRecord]
  · intro d t
    exact ⟨measurable_const, measurable_const⟩
  · refine ⟨1, by norm_num, ?_⟩

    change ‖(![1] : Vec 1)‖ ≤ 1 ∧ |(0 : ℝ)| ≤ 1
    norm_num [pi_norm_le_iff_of_nonneg (by norm_num : (0:ℝ) ≤ 1)]

-- The finite N=3 covariance-mismatch witness is a nonzero matrix entry.
example (t1 t2 s : ℝ) :
    sharedCov3 1 1 0 2 = 1 ∧ twoRoleCov3 t1 t2 s 0 2 = 0 := by
  norm_num [sharedCov3, twoRoleCov3, roleZ1, roleZ2, Matrix.mul_apply,
    Fin.sum_univ_succ, Matrix.vecMul, dotProduct]

-- An actual nonsingular pair of rows in the fixed uniform construction.
example : Matrix.det (!![1, 2+(0:ℝ)+1/2; 1, 2+0-1/2] : Mat 2) = -1 := by
  norm_num [Matrix.det_fin_two]

-- All hypotheses of the translation-equivariant variance bound are inhabited.
def checkA (_ : Fin 1) (ω : ℝ) : ℝ := ω
example : iIndepFun checkA (gaussianReal 0 1) := by
  rw [iIndepFun_iff]
  intro s f hf
  have hs : s = ∅ ∨ s = {0} := by
    by_cases h : (0 : Fin 1) ∈ s
    · right
      ext i
      fin_cases i
      simp [h]
    · left
      ext i
      fin_cases i
      simp [h]
  rcases hs with rfl | rfl <;> simp

example (i : Fin 1) : HasLaw (checkA i) (gaussianReal 0 1) (gaussianReal 0 1) :=
  HasLaw.id

example : (∀ (y : Vec 1) (c : ℝ), (y + c • (fun _ : Fin 1 => (1 : ℝ))) 0 = y 0 + c) ∧
    IndepFun (fun _ : ℝ => (0 : Vec 1))
      (fun ω => (∑ i : Fin 1, checkA i ω) / (1 : ℝ)) (gaussianReal 0 1) ∧
    Measurable (fun y : Vec 1 => y 0) ∧
    MemLp (fun _ : ℝ => (0 : Vec 1) 0) 2 (gaussianReal 0 1) ∧
    MemLp (fun ω : ℝ => ((0 : Vec 1) +
      (2 * (∑ i : Fin 1, checkA i ω) / (1 : ℝ)) • (fun _ : Fin 1 => (1 : ℝ))) 0) 2
      (gaussianReal 0 1) := by
  refine ⟨?_, indepFun_const_left _ _, measurable_pi_apply 0, memLp_const 0, ?_⟩
  · intro y c
    simp
  · simpa [checkA] using
      (memLp_id_gaussianReal (μ := (0 : ℝ)) (v := 1) 2).const_mul 2

lemma checkRepeatedGram (ω : Fin 2 → Unit) :
    repeatedGram (singleRecord checkKernel) ω = (fun _ _ => (1 : ℝ)) := by
  classical
  ext a b
  fin_cases a
  fin_cases b
  simp only [repeatedGram, repeatedX, singleRecord, checkKernel, outer,
    Matrix.sum_apply]
  norm_num [singleRecord, checkKernel, Finset.card_univ, Fintype.card_fin,
    Finset.card_attach, dyads, Finset.card_powersetCard]
  have hc (c : Dyad 2) :
      @Finset.card (Fin 1) (@Finset.univ (Fin 1)
        (Fin.fintype ((singleRecord checkKernel).T 2 c))) = 1 := by
    exact Fintype.card_fin 1
  simp only [hc, Finset.sum_const, Finset.card_attach]
  norm_num [dyads, Finset.card_powersetCard]

lemma checkRepeatedL : repeatedL (singleRecord checkKernel) 2 = 1 := by
  classical
  norm_num [repeatedL, singleRecord, Fintype.card_coe, dyads,
    Finset.card_powersetCard]

example : (repeatedH (singleRecord checkKernel) 2).det ≠ 0 := by
  unfold repeatedH
  rw [checkRepeatedL]
  simp only [inv_one, one_smul]
  rw [Matrix.det_fin_one]
  change (∫ ω, repeatedGram (singleRecord checkKernel) ω 0 0
    ∂repeatedLaw (singleRecord checkKernel) 2) ≠ 0
  simp_rw [checkRepeatedGram]
  norm_num [repeatedLaw, singleRecord, checkKernel]

example : (recordWeightedH (singleRecord checkKernel)
    (fun (_ : Dyad 2) _ => (1 : ℝ))).det ≠ 0 := by
  have hD : Dreal 2 = 1 := by norm_num [Dreal]
  unfold recordWeightedH
  rw [hD]
  simp only [inv_one, one_smul]
  rw [Matrix.det_fin_one]
  change (∫ ω, recordWeightedGram (singleRecord checkKernel)
    (fun (_ : Dyad 2) _ => (1 : ℝ)) ω 0 0
    ∂repeatedLaw (singleRecord checkKernel) 2) ≠ 0
  have hG (ω : Fin 2 → Unit) :
      recordWeightedGram (singleRecord checkKernel)
        (fun (_ : Dyad 2) _ => (1 : ℝ)) ω =
        repeatedGram (singleRecord checkKernel) ω := by
    simp [recordWeightedGram, repeatedGram]
  simp_rw [hG, checkRepeatedGram]
  norm_num [repeatedLaw, singleRecord, checkKernel]
