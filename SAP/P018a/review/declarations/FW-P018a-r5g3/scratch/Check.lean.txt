-- Direct checks of data; none uses a Challenge theorem as a premise.
local instance : IsProbabilityMeasure fairLaw := by

example : populationGram signKernel = (1 : Mat 1) := by
  ext a b
  fin_cases a
  fin_cases b
  simp [populationGram, meanMat, signKernel, outer]

theorem signStanding : Standing signKernel := by
  refine ⟨?_, ?_, ?_, ?_, ?_, ?_, ?_⟩
  · unfold signKernel fairLaw
    infer_instance
  · exact measurable_const
  · exact measurable_of_finite _
  · intro u v
    simp [signKernel, add_comm]
  · refine ⟨2, by norm_num, ?_⟩
    filter_upwards [] with uv
    rcases uv with ⟨u,v⟩
    cases u <;> cases v <;> norm_num [signKernel, signValue, Pi.norm_def]
  · exact ⟨0, by intros; rfl⟩
  · have hQ : populationGram signKernel = (1 : Mat 1) := by
      ext a b
      fin_cases a
      fin_cases b
      simp [populationGram, meanMat, signKernel, outer]
    rw [hQ]
    norm_num

-- T12: one concrete, positive-dimensional instance of all finite contract premises.
private def R : RepeatedKernel Bool 1 := singleRecord signKernel

example : WholeRecordRegular R 3 ∧ 2 ≤ (3 : ℕ) ∧
    0 < repeatedL R 3 ∧ (repeatedH R 3).det ≠ 0 ∧
    (recordWeightedH (N := 3) R (fun _ _ => 1)).det ≠ 0 ∧
    (meanMat (repeatedLaw R 3) (repeatedGram R)).det ≠ 0 ∧ Standing signKernel := by
  have hL : repeatedL R 3 = 3 := by
    simp [repeatedL, R, singleRecord,
      show (dyads 3).card = 3 from by decide]
  have hG : ∀ ω : Fin 3 → Bool, repeatedGram R ω = (!![3] : Mat 1) := by
    intro ω
    change (∑ d : Dyad 3, ∑ _ : Fin 1, outer (![1] : Vec 1) ![1]) = (!![3] : Mat 1)
    simp only [Fin.sum_univ_one]
    simpa only [sampleGram, design, signKernel] using signGram ω
  have hGW : ∀ ω : Fin 3 → Bool,
      recordWeightedGram R (fun _ _ => 1) ω = (!![3] : Mat 1) := by
    intro ω
    simpa [recordWeightedGram, repeatedGram] using hG ω
  have hMean : meanMat (repeatedLaw R 3) (repeatedGram R) = (!![3] : Mat 1) := by
    unfold meanMat
    simp_rw [hG]
    ext a b
    simp [repeatedLaw, R, singleRecord, signKernel]
  have hMeanW : meanMat (repeatedLaw R 3)
      (recordWeightedGram R (fun _ _ => 1)) = (!![3] : Mat 1) := by
    unfold meanMat
    simp_rw [hGW]
    ext a b
    simp [repeatedLaw, R, singleRecord, signKernel]
  have hRegular : WholeRecordRegular R 3 := by
    refine ⟨?_, ?_, ?_⟩
    · change IsProbabilityMeasure fairLaw
      infer_instance
    · intro d t
      exact ⟨measurable_const, measurable_of_finite _⟩
    · refine ⟨2, by norm_num, ?_⟩
      intro d t
      filter_upwards [] with uv
      change ‖(![1] : Vec 1)‖ ≤ 2 ∧ |signValue uv.1 + signValue uv.2| ≤ 2
      rcases uv with ⟨u,v⟩
      cases u <;> cases v <;> norm_num [signValue, Pi.norm_def]
  refine ⟨hRegular, by norm_num, ?_, ?_, ?_, ?_, signStanding⟩
  · rw [hL]
    norm_num
  · rw [repeatedH, hL, hMean]
    norm_num [Matrix.det_fin_one]
  · rw [recordWeightedH, hMeanW]
    norm_num [Dreal, Matrix.det_fin_one]
  · rw [hMean]
    norm_num [Matrix.det_fin_one]

-- T14: all hypotheses hold together at the explicit T13 witness.
