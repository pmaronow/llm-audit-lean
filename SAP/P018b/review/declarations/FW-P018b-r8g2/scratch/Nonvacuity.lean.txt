@[simp] lemma dyads_card (N : ℕ) : (dyads N).card = N.choose 2 := by
  simp [dyads]

lemma dyad_card (N : ℕ) : Fintype.card (Dyad N) = N.choose 2 := by
  change Fintype.card ↥(Finset.univ.powersetCard 2 : Finset (Finset (Fin N))) = _
  rw [Fintype.card_coe, Finset.card_powersetCard]
  simp

lemma R_L (N : ℕ) : repeatedL R N = (N.choose 2 : ℝ) := by
  simp [repeatedL, R, dyads_card]

lemma R_L_pos (N : ℕ) (hN : 2 ≤ N) : 0 < repeatedL R N := by
  have hcard : repeatedL R N = (N.choose 2 : ℝ) := by
    simp [repeatedL, R, dyads_card]
  rw [hcard]
  exact_mod_cast Nat.choose_pos hN

lemma R_H (N : ℕ) (hN : 2 ≤ N) : wholeRepeatedH R N = (1 : Mat 1) := by

  fin_cases b

lemma R_beta_zero (N : ℕ) : repeatedBeta R N = 0 := by
  unfold repeatedBeta
  have hxy : meanVec (repeatedLaw R N) (fun ω => ∑ d : Dyad N, ∑ t,
      score (repeatedX R ω d t) (repeatedY R ω d t)) = 0 := by
    ext a
    simp [meanVec, repeatedX, repeatedY, R, score]
  rw [hxy, Matrix.mulVec_zero]

lemma R_score_zero (N : ℕ) (d : Dyad N) :
    meanVec (repeatedLaw R N) (fun ω => repeatedBlockScore R (repeatedBeta R N) ω d) = 0 := by
  rw [R_beta_zero]
  ext a
  simp [meanVec, repeatedBlockScore, repeatedX, repeatedY, dot, R, score]

-- All UI and weak-limit hypotheses for a concrete one-dimensional zero error.
example :
    let μ : ∀ _ : ℕ, Measure Unit := fun _ => Measure.dirac ()
    let e : ∀ _ : ℕ, Unit → Vec 1 := fun _ _ => 0
    let ξ : ∀ _ : ℕ, Unit → Vec 1 := fun N ω => Real.sqrt N • e N ω
    let ν : Measure (Vec 1) := Measure.dirac 0
    (∀ N a, MemLp (fun ω => e N ω a) 2 (μ N)) ∧
    weakLimit μ ξ ν ∧ uiSquares μ ξ ∧
    (∀ a, MemLp (fun x : Vec 1 => x a) 2 ν) ∧ meanVec ν id = 0 := by
  dsimp
  refine ⟨?_, ?_, ?_, ?_, ?_⟩
  · intro N a; simpa using (memLp_const (0 : ℝ) (μ := Measure.dirac ()))
  · refine ⟨fun _ => (by fun_prop), inferInstance, ?_⟩
    intro g
    simpa using (tendsto_const_nhds : Tendsto (fun _ : ℕ => g 0) atTop (𝓝 (g 0)))
  · refine ⟨?_, ?_⟩
    · intro N a; simpa using (memLp_const (0 : ℝ) (μ := Measure.dirac ()))
    · intro ε hε
      refine ⟨1, by norm_num, ?_⟩
      intro N
      simpa using hε
  · intro a
    apply MemLp.of_bound (measurable_pi_apply a).aestronglyMeasurable 0
    simp
  · ext a
    simp [meanVec]

-- Every directed consistency premise is simultaneously satisfiable.
example :
    RepeatedStanding R ∧ (∀ N d, R.T N d ≤ 2) ∧
    (∃ (X : Unit → Unit → Vec 1) (Y : Unit → Unit → ℝ)
      (dir : ∀ N, (d : Dyad N) → Fin (R.T N d) → Bool),
      (∀ N d, Function.Injective (dir N d)) ∧
      (∀ N d t u v, R.X N d t u v = (if dir N d t then X v u else X u v) ∧
        R.Y N d t u v = (if dir N d t then Y v u else Y u v))) ∧
    (∀ N : ℕ, (0 : ℝ) < 1) ∧
    (∃ C : ℝ, ∀ N (d : Dyad N) (t : Fin (R.T N d)), 0 < (1 : ℝ)/1 ∧ (1 : ℝ)/1 ≤ C) ∧
    (1 : Mat 1).PosDef ∧
    Tendsto (fun N => wholeWeightedH R (fun (_ : Dyad N) _ => (1 : ℝ)/1)) atTop (𝓝 1) := by
  refine ⟨R_standing, (by intros; norm_num [R]), ?_, (by intro; norm_num), ?_, Matrix.PosDef.one, ?_⟩
  · refine ⟨fun _ _ => ![1], fun _ _ => 0, fun _ _ _ => false, ?_, ?_⟩
    · intro N d a b hab
      change Fin 1 at a b
      exact Subsingleton.elim a b
    · intros; exact ⟨rfl, rfl⟩
  · exact ⟨1, by intros; norm_num⟩
  · apply Filter.Tendsto.congr' (f₁ := fun _ : ℕ => (1 : Mat 1)) _ tendsto_const_nhds
    filter_upwards [eventually_ge_atTop 2] with N hN
    simpa using (R_weighted_H N hN).symm
