abbrev P : ClusterPopulation where
  M := 4

  mt_pos := by decide
  mt_lt := by decide

abbrev s : Finset (Fin P.M) := {0, 1}

theorem admissible_P : 0 < P.mt ∧ P.mt < P.M ∧ (∀ j, 0 < P.n j) ∧
    2 ≤ P.mt ∧ 2 ≤ controlCount P ∧ populationSize P = 10 ∧
    clusterVariance (fun j => (P.n j : ℝ)) = 5/4 ∧
    s ∈ assignments P := by
  norm_num [P, s, controlCount, populationSize, clusterVariance, clusterMean,
    Fin.sum_univ_succ, assignments, Finset.mem_powersetCard]
  exact P.n_pos

theorem affine_witness_P : armMeanSize P true s = 3/2 ∧ armMeanSize P false s = 7/2 ∧
    horvitzThompson P s = -(1:ℝ)/5 ∧
    horvitzThompson (affinePopulation P 1 2) s = -(6:ℝ)/5 ∧
    horvitzThompson (affinePopulation P 1 2) s ≠ 2 * horvitzThompson P s := by
  have hcontrols : controls s = ({2,3} : Finset (Fin 4)) := by decide
  norm_num [armMeanSize, armSize, armSum, hcontrols, controlCount,
    horvitzThompson, htTotal, populationSize, affinePopulation, Fin.sum_univ_succ,
    P, s]

example : ∃ (Q : ClusterPopulation) (b0 b1 : ℝ) (a : Finset (Fin Q.M)),
    a ∈ assignments Q ∧
      horvitzThompson (affinePopulation Q b0 b1) a ≠ b1 * horvitzThompson Q a := by
  refine ⟨P, 1, 2, s, ?_, affine_witness_P.2.2.2.2⟩
  decide

def L : FiniteLaw (Fin 2) := uniformLaw Finset.univ ⟨0, Finset.mem_univ _⟩
def u : Fin 2 → ℝ := ![1, 3]
def v : Fin 2 → ℝ := ![1, 2]

example : (∀ ω ∈ L.support, 0 < v ω) ∧ expectation L v = 3/2 ∧
    designCovariance L (fun ω => u ω / v ω) v = 1/8 ∧
    expectation L (fun ω => u ω / v ω) =
      (expectation L u - designCovariance L (fun ω => u ω / v ω) v) /
        expectation L v := by
  constructor
  · intro ω hω; fin_cases ω <;> norm_num [v]
  · norm_num [expectation, designCovariance, L, uniformLaw, u, v,
      Fin.sum_univ_succ]

theorem E_P (f : Finset (Fin 4) → ℝ) :
    expectation (clusterLaw P) f =
      (f {0,1} + f {0,2} + f {0,3} + f {1,2} + f {1,3} + f {2,3}) / 6 := by
  simp +decide [expectation, clusterLaw, uniformLaw, assignments_P, Finset.sum_insert]
  ring

example : expectation (clusterLaw P) (differenceInMeans P) = 1/24 ∧
    averageTreatmentEffect P = 0 ∧

abbrev X : Fin 1 → Fin P.M → ℝ := fun _ => ![2, -1, 4, 0]
abbrev ka : Fin 1 → ℝ := fun _ => -2

-- Nonconstant fixed adjustments; verify both totals, the ATE and centering.
example : expectation (clusterLaw P) (rajTotal P false (3/2)) = 3 ∧
    expectation (clusterLaw P) (rajTotal P true (3/2)) = 3 ∧
    expectation (clusterLaw P) (desRaj P (3/2)) = 0 ∧
    expectation (clusterLaw P) (rajCovariateTotal P X 1 ka false) = 3 ∧
    expectation (clusterLaw P) (rajCovariateTotal P X 1 ka true) = 3 ∧
    expectation (clusterLaw P) (desRajCovariates P X 1 ka) = 0 ∧
    expectation (clusterLaw P) (fun u =>
      3 * (armMeanSize P true u - (populationSize P : ℝ)/P.M)) = 0 ∧
    expectation (clusterLaw P) (fun u =>
      centeredArmMeanW3 P false (fun j => (P.n j : ℝ)) u +
        ∑ l, ka l * centeredArmMeanW3 P false (X l) u) = 0 := by
  norm_num [E_P, rajTotal, rajPotential, differenced, armSum, controlCount,
    desRaj, rajCovariateTotal, rajCovariatePotential, desRajCovariates,
    clusterMean, populationSize, armMeanSize, armSize, centeredArmMeanW3,
    Fin.sum_univ_succ, X, ka, P]

abbrev Pequal : ClusterPopulation :=

example : ∃ (Q : ClusterPopulation) (_Z : Fin 0 → Fin Q.M → ℝ) (_b : Fin 0 → ℝ),
    clusterVariance (fun j => (Q.n j : ℝ)) = 0 := by
  refine ⟨Pequal, Fin.elim0, Fin.elim0, ?_⟩
  norm_num [Pequal, P, clusterVariance, clusterMean, Fin.sum_univ_succ]

example (k : ℝ) (u : Finset (Fin Pequal.M)) :
    desRaj Pequal k u = horvitzThompson Pequal u := by
  have hr (a : Bool) : rajPotential Pequal a k =
      (if a then Pequal.y1 else Pequal.y0) := by
    funext j
    norm_num [rajPotential, differenced, clusterMean, Pequal, P, Fin.sum_univ_succ]
  simp only [desRaj, rajTotal, hr, horvitzThompson, htTotal]

abbrev Blocks : Fin 2 → ClusterPopulation := fun _ => P

theorem block_card : (blockLaw Blocks).support.card = 36 := by
  decide

-- A positive-probability off-diagonal draw rules out a repeated-draw law.
example : ∃ z : BlockAssignments Blocks,
    z ∈ (blockLaw Blocks).support ∧ (z 0).val ≠ (z 1).val ∧
      (blockLaw Blocks).weight z = 1/36 := by
  have h01 : ({0,1} : Finset (Fin 4)) ∈ assignments P := by decide
  have h23 : ({2,3} : Finset (Fin 4)) ∈ assignments P := by decide
  let z : BlockAssignments Blocks := fun b =>
    if b = 0 then ⟨{0,1}, h01⟩ else ⟨{2,3}, h23⟩
  refine ⟨z, ?_, ?_, ?_⟩
  · exact Finset.mem_univ _
  · simp [z]; decide
  · change 1 / ((blockLaw Blocks).support.card : ℝ) = 1/36
    rw [block_card]
    norm_num

-- The blockwise conservativeness premise is satisfiable by the actual statistic.
example : 0 < (2 : ℕ) ∧ blockedSize Blocks = 20 ∧
    (∀ b : Fin 2,
      designVariance (clusterLaw (Blocks b)) (horvitzThompson (Blocks b)) ≤
        expectation (clusterLaw (Blocks b))
          (combinedVarianceEstimator (Blocks b) (Blocks b).y0 (Blocks b).y1)) := by
  constructor
  · decide
  constructor
  · norm_num [blockedSize, Blocks, populationSize, P, Fin.sum_univ_succ]
  · intro b
    change designVariance (clusterLaw P) (horvitzThompson P) ≤
      expectation (clusterLaw P) (combinedVarianceEstimator P P.y0 P.y1)
    rw [ht_variance_P, expected_variance_estimator_P]
    norm_num

end
