-- T4: a valid design and a nonconstant population with an interior CDF value.
def balanced : Fin 4 → ℝ := ![0, 0, 1, 1]

example : fourUnitDesign.N = 4 ∧ fourUnitDesign.m = 2 ∧ fourUnitDesign.k = 2 ∧
    4 ≤ fourUnitDesign.N ∧ 2 ≤ fourUnitDesign.m ∧ 2 ≤ fourUnitDesign.k ∧
    fourUnitDesign.m + fourUnitDesign.k ≤ fourUnitDesign.N := by
  norm_num [fourUnitDesign]

example : empiricalCDF balanced 0 = (1 / 2 : ℝ) := by
  unfold empiricalCDF finiteMean Finset.expect
  rw [Fin.sum_univ_succ, Fin.sum_univ_succ, Fin.sum_univ_succ, Fin.sum_univ_succ]
  norm_num [balanced, NNRat.smul_def]

-- T10: a nested population with bounded variances; arms stay fixed while N grows.
def zeroOneSequence : ExperimentSequence where
  design q := {
    N := q + 4
    m := 2
    k := 2
    population_min := by omega
    treatment_min := by omega
    control_min := by omega
    sample_le := by omega }
  size _ := rfl
  y1 _ := 0
  y0 _ := 1

example :
    (∀ᶠ q in atTop, (zeroOneSequence.design q).m = 2) ∧
    (∀ᶠ q in atTop, (zeroOneSequence.design q).k = 2) ∧
    (∃ M : ℝ, ∀ q, finiteVariance (rowOutcomes zeroOneSequence true q) ≤ M ∧
      finiteVariance (rowOutcomes zeroOneSequence false q) ≤ M) := by
  refine ⟨Filter.Eventually.of_forall (fun _ => rfl),
    Filter.Eventually.of_forall (fun _ => rfl), 0, ?_⟩
  intro q
  haveI : Nonempty (Fin (q + 4)) := ⟨⟨0, by omega⟩⟩
  change finiteVariance (fun _ : Fin (q + 4) => (0 : ℝ)) ≤ 0 ∧
    finiteVariance (fun _ : Fin (q + 4) => (1 : ℝ)) ≤ 0
  simp [finiteVariance, finiteMean, Fintype.expect_const]

-- The fixed-sample estimator limit also admits nonconstant samples.
example : ∃ (m k : ℕ) (a : Fin m → ℝ) (b : Fin k → ℝ),
    2 ≤ m ∧ 2 ≤ k ∧ sampleVariance a = 1 / 2 ∧ sampleVariance b = 1 / 2 := by
  refine ⟨2, 2, ![0, 1], ![0, 1], ?_⟩
  norm_num [sampleVariance, finiteMean, Finset.expect, Fin.sum_univ_succ]
