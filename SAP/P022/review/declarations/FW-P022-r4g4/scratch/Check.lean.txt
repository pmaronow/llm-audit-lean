lemma zero_row (t : Bool) (q : ℕ) :
    rowOutcomes zeroWitness t q = fun _ => (0 : ℝ) := by
  cases t <;> rfl

lemma empirical_const {s : ℕ} (hs : 0 < s) (z x : ℝ) :
    empiricalCDF (fun _ : Fin s => z) x = if z ≤ x then 1 else 0 := by
  letI : Nonempty (Fin s) := Fin.pos_iff_nonempty.mp hs
  simp [empiricalCDF, finiteMean]

lemma origin_cdf (t : Bool) (x : ℝ) :
    lawCDF originLaw t x = if 0 ≤ x then 1 else 0 := by
  cases t <;> by_cases hx : 0 ≤ x <;>
    simp [lawCDF, originLaw, hx]

lemma zero_population_cdf (t : Bool) (q : ℕ) :
    empiricalCDF (rowOutcomes zeroWitness t q) = lawCDF originLaw t := by
  funext x
  rw [zero_row, empirical_const (by have := (zeroWitness.design q).population_min; omega)]
  exact (origin_cdf t x).symm

lemma zero_weak : weakPopulationConvergence zeroWitness originLaw := by
  intro f
  have hf : ∫ p, f p ∂(originLaw : Measure (ℝ × ℝ)) = f (0,0) := by
    simp [originLaw]
  rw [hf]
  have hfun : (fun q => finiteMean (fun i : Fin (zeroWitness.design q).N =>
      f (zeroWitness.y1 i.val, zeroWitness.y0 i.val))) = fun _ : ℕ => f (0,0) := by
    funext q
    letI : Nonempty (Fin (zeroWitness.design q).N) :=
      Fin.pos_iff_nonempty.mp (by have := (zeroWitness.design q).population_min; omega)
    change (𝔼 _i : Fin (zeroWitness.design q).N, f (0,0)) = f (0,0)
    exact Fintype.expect_const _
  rw [hfun]
  exact tendsto_const_nhds

lemma half_limit : Tendsto
    (fun q : ℕ => (((q + 4) / 2 : ℕ) : ℝ) / ((q : ℝ) + 4)) atTop (𝓝 (1 / 2 : ℝ)) := by
  have hmod : Tendsto (fun q : ℕ => (((q + 4) % 2 : ℕ) : ℝ) / ((q : ℝ) + 4))
      atTop (𝓝 (0 : ℝ)) := by
    simpa only [Function.comp_def, id_eq, Nat.cast_add, Nat.cast_ofNat] using
      (tendsto_mod_div_atTop_nhds_zero_nat (m := 2) (by decide)).comp
        ((tendsto_add_atTop_iff_nat 4).2 tendsto_id)
  have h := ((tendsto_const_nhds (x := (1 : ℝ))).sub hmod).div_const (2 : ℝ)
  norm_num at h
  convert h using 1
  funext q
  have hd : (((q + 4) % 2 : ℕ) : ℝ) + 2 * (((q + 4) / 2 : ℕ) : ℝ) = (q : ℝ) + 4 := by
    exact_mod_cast Nat.mod_add_div (q + 4) 2
  have hn : (q : ℝ) + 4 ≠ 0 := by positivity
  field_simp
  nlinarith

lemma full_allocation (q : ℕ) :
    (((zeroWitness.design q).m : ℝ) + (zeroWitness.design q).k) /
      (zeroWitness.design q).N = 1 := by
  change ((((q + 4) / 2 : ℕ) : ℝ) + ((q + 4 - (q + 4) / 2 : ℕ) : ℝ)) /
    ((q + 4 : ℕ) : ℝ) = 1
  have hh : (((q + 4) / 2 : ℕ) : ℝ) + ((q + 4 - (q + 4) / 2 : ℕ) : ℝ) =
      ((q + 4 : ℕ) : ℝ) := by
    exact_mod_cast Nat.add_sub_of_le (Nat.div_le_self (q + 4) 2)
  rw [hh, div_self (by positivity)]

lemma zero_sample_cdf (t : Bool) (q : ℕ) (π : Assignment (zeroWitness.design q)) :
    sampledCDF zeroWitness t q π = lawCDF originLaw t := by
  funext x
  cases t
  · change empiricalCDF (fun _ : Fin (zeroWitness.design q).k => (0 : ℝ)) x =
      lawCDF originLaw false x
    rw [empirical_const (by have := (zeroWitness.design q).control_min; omega), origin_cdf]
  · change empiricalCDF (fun _ : Fin (zeroWitness.design q).m => (0 : ℝ)) x =
      lawCDF originLaw true x
    rw [empirical_const (by have := (zeroWitness.design q).treatment_min; omega), origin_cdf]

lemma seven_zero (d : Design) (C : ℝ) (π : Assignment d) :
    sevenTerms d (fun _ => 0) (fun _ => 0) C π = fun _ => 0 := by
  have hN := empirical_const (s := d.N) (by have := d.population_min; omega) (0 : ℝ)
  have hM := empirical_const (s := d.m) (by have := d.treatment_min; omega) (0 : ℝ)
  have hK := empirical_const (s := d.k) (by have := d.control_min; omega) (0 : ℝ)
  have ht : empiricalCDF (observedTreatment d (fun _ => 0) π) =
      empiricalCDF (fun _ : Fin d.N => 0) := by
    funext x
    change empiricalCDF (fun _ : Fin d.m => 0) x = empiricalCDF (fun _ : Fin d.N => 0) x
    rw [hM x, hN x]
  have hc : empiricalCDF (observedControl d (fun _ => 0) π) =
      empiricalCDF (fun _ : Fin d.N => 0) := by
    funext x
    change empiricalCDF (fun _ : Fin d.k => 0) x = empiricalCDF (fun _ : Fin d.N => 0) x
    rw [hK x, hN x]
  have hl : lastIntegralTerm d (fun _ => 0) (fun _ => 0) C π = 0 := by
    simp [lastIntegralTerm, ht, hc]
  funext j
  fin_cases j <;> simp [sevenTerms, finiteMean, observedTreatment, observedControl, hl]

lemma zero_earlier (C : ℝ) (ν : Fin 8 → ℝ) (hν : ∀ i, 0 < ν i) :

  have hpos : 0 < correctedAllocation ν 0 ⟨j.val, by omega⟩ := by
    fin_cases j <;> dsimp [correctedAllocation] <;>
      first | exact hν _ | exact add_pos (hν _) (hν _)
  simp only [zero_row, seven_zero, abs_zero, not_le.mpr hpos, scaled_false]
  exact bot_le
