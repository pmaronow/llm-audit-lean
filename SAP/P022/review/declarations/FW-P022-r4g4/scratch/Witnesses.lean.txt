lemma empirical_all_le {s : ℕ} (hs : 0 < s) (v : Fin s → ℝ) (x : ℝ)
    (h : ∀ i, v i ≤ x) : empiricalCDF v x = 1 := by
  letI : Nonempty (Fin s) := Fin.pos_iff_nonempty.mp hs
  unfold empiricalCDF finiteMean
  have hf : (fun i : Fin s => if v i ≤ x then (1 : ℝ) else 0) = fun _ => 1 := by
    funext i; exact if_pos (h i)
  rw [hf]
  exact Fintype.expect_const _

lemma reciprocal_below (t : Bool) (q : ℕ)
    (i : Fin (reciprocalWitness.design q).N) :
    rowOutcomes reciprocalWitness t q i < reciprocalThreshold q := by
  have hi : i.val < q + 4 := by simpa [reciprocalWitness] using i.isLt
  have hi' : (i.val : ℝ) + 1 ≤ (q : ℝ) + 4 := by exact_mod_cast hi
  have hd1 : 0 < (i.val : ℝ) + 1 := by positivity
  have hd2 : 0 < 2 * ((q : ℝ) + 4) := by positivity
  have hfrac : 1 / (2 * ((q : ℝ) + 4)) < 1 / ((i.val : ℝ) + 1) := by
    apply (div_lt_div_iff₀ hd2 hd1).mpr
    nlinarith [(show (0 : ℝ) ≤ (q : ℝ) by positivity)]
  cases t <;> simpa [rowOutcomes, reciprocalWitness, reciprocalThreshold] using neg_lt_neg hfrac

example (q : ℕ) (t : Bool) :
    empiricalCDF (rowOutcomes reciprocalWitness t q) (reciprocalThreshold q) = 1 := by
  apply empirical_all_le (by have := (reciprocalWitness.design q).population_min; omega)
  intro i; exact (reciprocal_below t q i).le

example (q : ℕ) (t : Bool) (π : Assignment (reciprocalWitness.design q)) :
    sampledCDF reciprocalWitness t q π (reciprocalThreshold q) = 1 := by
  cases t
  · apply empirical_all_le (by have := (reciprocalWitness.design q).control_min; omega)
    intro i
    exact (reciprocal_below false q (π ⟨(reciprocalWitness.design q).m + i.val,
      by have := (reciprocalWitness.design q).sample_le; omega⟩)).le
  · apply empirical_all_le (by have := (reciprocalWitness.design q).treatment_min; omega)
    intro i
    exact (reciprocal_below true q (π ⟨i.val,
      by have := (reciprocalWitness.design q).sample_le; omega⟩)).le

example (q : ℕ) (t : Bool) : lawCDF originLaw t (reciprocalThreshold q) = 0 := by
  have hx : reciprocalThreshold q < 0 := by
    unfold reciprocalThreshold
    exact neg_lt_zero.mpr (by positivity)
  cases t <;> simp [lawCDF, originLaw, not_le.mpr hx]

lemma perturbed_bound (i : ℕ) (a : ℝ) (ha0 : 0 ≤ a) (ha1 : a ≤ 1) :
    |a - 1 / ((i : ℝ) + 3)| < 1 := by
  have hp : 0 < 1 / ((i : ℝ) + 3) := by positivity
  have hlt : 1 / ((i : ℝ) + 3) < 1 := by
    apply (div_lt_one (by positivity)).mpr
    nlinarith [(show (0 : ℝ) ≤ (i : ℝ) by positivity)]
  rw [abs_lt]
  constructor <;> linarith

example (i : ℕ) : |perturbedCornerWitness.y1 i| < 1 ∧ |perturbedCornerWitness.y0 i| < 1 := by
  constructor
  · apply perturbed_bound i (cornerWitness.y1 i)
    · dsimp [cornerWitness]; split <;> norm_num
    · dsimp [cornerWitness]; split <;> norm_num
  · apply perturbed_bound i (cornerWitness.y0 i)
    · dsimp [cornerWitness]; split <;> norm_num
    · dsimp [cornerWitness]; split <;> norm_num

lemma perturbed_below (t : Bool) (q : ℕ)
    (i : Fin (perturbedCornerWitness.design q).N) :
    rowOutcomes perturbedCornerWitness t q i < perturbedThreshold q := by
  have hi : i.val < q + 4 := by
    simpa [perturbedCornerWitness, cornerWitness, reciprocalWitness] using i.isLt
  have hi' : (i.val : ℝ) + 3 ≤ (q : ℝ) + 6 := by exact_mod_cast (by omega : i.val + 3 ≤ q + 6)
  have hd1 : 0 < (i.val : ℝ) + 3 := by positivity
  have hd2 : 0 < 2 * ((q : ℝ) + 6) := by positivity
  have hfrac : 1 / (2 * ((q : ℝ) + 6)) < 1 / ((i.val : ℝ) + 3) := by
    apply (div_lt_div_iff₀ hd2 hd1).mpr
    nlinarith [(show (0 : ℝ) ≤ (q : ℝ) by positivity)]
  cases t <;> dsimp [rowOutcomes, perturbedCornerWitness, cornerWitness, perturbedThreshold] <;>
    split <;> linarith

lemma perturbed_threshold_interior (q : ℕ) :
    0 < perturbedThreshold q ∧ perturbedThreshold q < 1 := by
  have hp : 0 < 1 / (2 * ((q : ℝ) + 6)) := by positivity
  have hlt : 1 / (2 * ((q : ℝ) + 6)) < 1 := by
    apply (div_lt_one (by positivity)).mpr
    nlinarith [(show (0 : ℝ) ≤ (q : ℝ) by positivity)]
  dsimp [perturbedThreshold]
  constructor <;> linarith

example (q : ℕ) (t : Bool) : lawCDF cornerLaw t (perturbedThreshold q) = 1/2 := by
  obtain ⟨hp, hlt⟩ := perturbed_threshold_interior q
  cases t <;>
    norm_num [lawCDF, cornerLaw, Measure.add_apply, Measure.smul_apply,
      hp.le, not_le.mpr hlt, ENNReal.toReal_add, ENNReal.toReal_inv]
