theorem linear_twoParty : TwoParty linearX linearD := by
  refine ⟨?_, ?_, ?_, ?_⟩
  all_goals intro i; fin_cases i <;> norm_num [linearX, linearD]

theorem linear_condition1 : Condition1 (1 / 200 : ℝ) linearX := by
  constructor
  · exact ⟨3, by norm_num [linearX]⟩
  · exact ⟨2, by norm_num [linearX]⟩

theorem linear_fullRank : FullRank (polynomialDesign 1 linearX linearD) := by
  intro c h
  have h0 := h 0
  have h1 := h 1
  have h3 := h 3
  norm_num [predict, polynomialDesign, linearX, linearD, Fin.sum_univ_succ] at h0 h1 h3
  funext j
  fin_cases j
  · change c (0 : Fin (1+2)) = 0
    linear_combination (-1 / 2 : ℝ) * h0 + (3 / 2 : ℝ) * h1
  · change c (1 : Fin (1+2)) = 0
    linear_combination (101 / 200 : ℝ) * h0 + (-301 / 200 : ℝ) * h1 + 1 * h3
  · change c (2 : Fin (1+2)) = 0
    linear_combination -2 * h0 + 2 * h1

-- A genuine finite-design conditional-mean instance for T3, with all its guards.
example : ∃ (P : Measure ℝ) (Y : Fin 6 → ℝ → ℝ),
    P univ = 1 ∧ (∀ i, Integrable (Y i) P) ∧
    (∀ i, (∫ ω, Y i ω ∂P) = (1/2 : ℝ) + (1/10)*linearD i + linearX i/20) ∧
    (∀ i, Integrable (fun ω => (Y i ω - linearY i)^2) P) ∧
    (∀ i l, i ≠ l → (∫ ω, (Y i ω-linearY i)*(Y l ω-linearY l) ∂P) = 0) := by
  refine ⟨Measure.dirac 0, (fun i _ => linearY i), ?_, ?_, ?_, ?_, ?_⟩
  · simp
  · intro i; exact integrable_const _
  · intro i; fin_cases i <;> norm_num [linearY, linearD, linearX]
  · intro i; simp
  · intro i l hil; simp

-- All the design/count/agreement/coarsened-fit guards of T3 are satisfiable together.
example : TwoParty linearX linearD ∧
    FullRank (polynomialDesign 1 linearX linearD) ∧
    Condition1 (1/200 : ℝ) linearX ∧
    (0 < ∑ i ∈ close (1/200 : ℝ) linearX, linearD i) ∧
    (0 < ∑ i ∈ close (1/200 : ℝ) linearX, (1-linearD i)) ∧
    (∀ i ∈ close (1/200 : ℝ) linearX, (fun _ : Fin 6 => (0 : ℝ)) i = 0) ∧
    CoarsenedOLS (1/200 : ℝ) linearX linearD (fun _ => 0) 0 := by
  refine ⟨linear_twoParty, linear_fullRank, linear_condition1,
    linear_counts.1, linear_counts.2, ?_, ?_⟩
  · intro i hi; rfl
  · refine ⟨fun _ => 0, ?_⟩
    simp [IsBinOLS]
