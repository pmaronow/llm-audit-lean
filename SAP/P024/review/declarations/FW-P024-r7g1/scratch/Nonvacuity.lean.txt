def balancedPopulation (k : ℕ) : ClusterPopulation (k+2) where
  size := fun _ => 1
  positive := fun _ => by norm_num
  y0 := fun _ _ => 0
  y1 := fun _ _ => 1

def fairDesign (k : ℕ) : Design (Finset (Fin (k+2))) :=
  bernoulliDesign (k+2) (1/2) (by norm_num) (by norm_num)

lemma balanced_max (k : ℕ) : maxClusterWeight (balancedPopulation k) =
    1 / ((k : ℝ)+2) := by
  unfold maxClusterWeight
  apply Finset.sup'_eq_of_forall
  intro j hj
  simp [clusterWeight, balancedPopulation]

-- T5: all hypotheses of the infinite-sequence consistency branch jointly hold.
example : (0 : ℝ) ≤ 1 ∧ (0 : ℝ) < 1/4 ∧
    (∀ k j i, |(balancedPopulation k).y0 j i| ≤ 1 ∧
      |(balancedPopulation k).y1 j i| ≤ 1) ∧
    (∀ k, independentClusterLaw (fairDesign k) (1/2) ∨
      ∃ m : ℕ, 0 < m ∧ m < k+2 ∧ (1/2 : ℝ) = (m : ℝ)/(k+2) ∧
        completeClusterLaw (fairDesign k) m) ∧
    (∀ _k : ℕ, (1/4 : ℝ) ≤ 1/2 ∧ (1/2 : ℝ) ≤ 1-1/4) ∧
    Tendsto (fun k => maxClusterWeight (balancedPopulation k)) atTop (nhds 0) := by
  refine ⟨by norm_num, by norm_num, ?_, ?_, ?_, ?_⟩
  · intro k j i
    dsimp [balancedPopulation]
    norm_num
  · intro k
    exact Or.inl (fun _ => rfl)
  · intro k
    norm_num
  · simp_rw [balanced_max]
    simpa only [Function.comp_def, Nat.cast_add, Nat.cast_ofNat] using
      (tendsto_one_div_atTop_nhds_zero_nat (𝕜 := ℝ)).comp
        (tendsto_add_atTop_nat 2)

def completePair : Design (Finset (Fin 2)) where
  mass := fun s => if s.card = 1 then 1/2 else 0
  nonneg := fun s => by split_ifs <;> norm_num
  total := by
    have hU : (Finset.univ : Finset (Finset (Fin 2))) = {∅, {0}, {1}, {0,1}} := by
      decide
    rw [hU]
    rw [Finset.sum_insert (by decide : (∅ : Finset (Fin 2)) ∉
      ({{0}, {1}, {0,1}} : Finset (Finset (Fin 2))))]
    rw [Finset.sum_insert (by decide : ({0} : Finset (Fin 2)) ∉
      ({{1}, {0,1}} : Finset (Finset (Fin 2))))]
    rw [Finset.sum_insert (by decide : ({1} : Finset (Fin 2)) ∉
      ({{0,1}} : Finset (Finset (Fin 2))))]
    norm_num
