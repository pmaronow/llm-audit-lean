-- The replication law is a nonzero probability law with independent coordinates.
private theorem genuine_iid {I K : Type*} [Fintype K] [MeasurableSpace K]
    (d : FiniteDesign I K) : ReplicationsIID d := by
  constructor
  · exact iIndepFun_infinitePi
      (P := fun _ : ℕ => (designPMF d).toMeasure)
      (X := fun _ x => x) (by fun_prop)
  · intro r
    exact (measurePreserving_eval_infinitePi
      (fun _ : ℕ => (designPMF d).toMeasure) r).map_eq
