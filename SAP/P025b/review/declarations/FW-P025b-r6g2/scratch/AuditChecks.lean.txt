-- T21: a random design, not a repeated dependent draw or a zero measure.
def coinDesign : FiniteDesign (Fin 2) (Fin 2) where
  sample := fun i => {i}
  probability := fun _ => 1/2
  nonnegative := by intro i; norm_num
  total := by norm_num [Fin.sum_univ_two]

lemma coin_inclusion (i : Fin 2) : inclusion coinDesign i = 1/2 := by
  fin_cases i <;>
    norm_num [inclusion, designExpectation, coinDesign, Fin.sum_univ_two]

lemma coin_classTotal (j : Fin 2) : classTotal coinDesign id j = 1/2 := by
  unfold classTotal
  rw [Finset.sum_filter]
  fin_cases j <;>
    norm_num [Fin.sum_univ_two, coin_inclusion]

lemma coin_separation : separation (![0,1] : Fin 2 → ℝ)
    (classTotal coinDesign id) ((1/2)/(1/4)) := by
  intro k
  fin_cases k <;>
    norm_num [upperValue, lowerValue, ratio, upperWeights, lowerWeights,
      Fin.sum_univ_two, coin_classTotal]

example : (0 < (2 : ℕ)) ∧ (0 < (2 : ℕ)) ∧
    Function.Surjective (id : Fin 2 → Fin 2) ∧
    StrictMono (![0,1] : Fin 2 → ℝ) ∧
    (0 : ℝ) < 1/4 ∧ (1/4 : ℝ) ≤ 1/2 ∧ (1/2 : ℝ) ≤ 1 ∧
    (∀ i, (1/4 : ℝ) ≤ inclusion coinDesign i ∧ inclusion coinDesign i ≤ 1/2) ∧
    separation (![0,1] : Fin 2 → ℝ) (classTotal coinDesign id) ((1/2)/(1/4)) := by
  refine ⟨by norm_num, by norm_num, Function.surjective_id, ?_,
    by norm_num, by norm_num, by norm_num, ?_, coin_separation⟩
  · intro i j hij
    fin_cases i <;> fin_cases j <;> norm_num at *
  · intro i
    rw [coin_inclusion]
    norm_num
