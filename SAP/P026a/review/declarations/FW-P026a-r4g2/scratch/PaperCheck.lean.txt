-- The T6 and T7 objects are concrete; verify the setting without using their theorems.
example : isProbability paperTableDesign := by
  norm_num [isProbability, paperTableDesign, Fintype.sum_prod_type]

example : hasPositiveMarginals paperTableDesign := by
  unfold hasPositiveMarginals
  decide +kernel

example : directedSupport paperTableDesign := by
  unfold directedSupport
  decide +kernel

example : ∀ i j : Fin 10, i ≠ j →
    0 < jointInclusion paperTableDesign true true i j ∧
    0 < jointInclusion paperTableDesign false false i j := by
  decide +kernel

example : paperTableLinearResidual =
    ![1,7/8,1/4,-1/2,-15/31,19/93,1/3,-76/93,-76/93,-79/93] := by
  decide +kernel

example : (∀ k, fiberTotal paperTableCluster paperTableY k = paperTableClusterY k ∧
    fiberTotal paperTableCluster (fun i => paperTableY i-1/2) k = paperTablePriorResidual k) := by
  decide +kernel

example : (∑ i, (paperTableCenteredX i)^2) = (331/10 : ℚ) := by
  decide +kernel

example : ∀ w : Fin 6 × Fin 15, 0 < ∑ i, (paperFeTreatmentResidual w i)^2 := by
  decide +kernel

-- A genuine supported allocation, with exactly two treated clusters in each block.

  decide +kernel
