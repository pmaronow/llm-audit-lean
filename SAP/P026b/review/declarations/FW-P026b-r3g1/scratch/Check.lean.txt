noncomputable def fairOne : UnitExperiment (Fin 1) where
  weight := fun _ => 1/2
  nonnegative := by intro w; norm_num
  normalized := by norm_num
  positive := by
    intro i
    have hs (b : Bool) : (1/2 : ℝ) ≤
        ∑ w : Fin 1 → Bool, (1/2 : ℝ) * (if w i = b then 1 else 0) := by
      simpa using (Finset.single_le_sum
        (f := fun w : Fin 1 → Bool => (1/2 : ℝ) * (if w i = b then 1 else 0))
        (s := univ) (a := fun _ => b)
        (by intro w hw; split_ifs <;> norm_num) (Finset.mem_univ _))
    constructor
    · have := hs true; linarith
    · have := hs false; linarith

-- T13 and each universal block of T16 have an actual admissible experiment.
example : probabilityDesign fairOne.design ∧ positiveMarginals fairOne.design := by
  constructor
  · exact ⟨fairOne.nonnegative, fairOne.normalized⟩
  · exact fairOne.positive

-- The T15 assignment law has exactly the claimed positive marginals.

  norm_num [probabilityDesign, positiveMarginals, marginal, expectation,

  constructor <;> intro i <;> fin_cases i <;> norm_num

-- The three-unit fits solve the ordinary, intercept-inclusive least squares problem.
example : ∀ w : Fin 3,
    IsLeastSquaresFitted 2
      (designWithIntercept 1 (fun (i : Fin 3) (_ : Fin 1) => (i.val : ℝ)))
      (fun i => (observed oneOfThree (fun _ => 1) (fun _ => 0) w i : ℚ))
      (fun i => (packageFit w i : ℝ)) := by
  intro w
  fin_cases w
  all_goals
    constructor
  case «0».left =>
    refine ⟨![5/6, -1/2], ?_⟩
    ext i
    fin_cases i <;> norm_num [designWithIntercept, packageFit, Matrix.mulVec,
      dotProduct, Fin.sum_univ_succ]
  case «0».right =>
    intro β
    norm_num [designWithIntercept, packageFit, observed, oneOfThree,
      Matrix.mulVec, dotProduct, Fin.sum_univ_succ]
    nlinarith [sq_nonneg (β 0 + β 1 - 1/3), sq_nonneg (β 1 + 1/2)]
  case «1».left =>
    refine ⟨![1/3, 0], ?_⟩
    ext i
    fin_cases i <;> norm_num [designWithIntercept, packageFit, Matrix.mulVec,
      dotProduct, Fin.sum_univ_succ]
  case «1».right =>
    intro β
    norm_num [designWithIntercept, packageFit, observed, oneOfThree,
      Matrix.mulVec, dotProduct, Fin.sum_univ_succ]
    nlinarith [sq_nonneg (β 0 + β 1 - 1/3), sq_nonneg (β 1)]
  case «2».left =>
    refine ⟨![-1/6, 1/2], ?_⟩
    ext i
    fin_cases i <;> norm_num [designWithIntercept, packageFit, Matrix.mulVec,
      dotProduct, Fin.sum_univ_succ]
  case «2».right =>
    intro β
    norm_num [designWithIntercept, packageFit, observed, oneOfThree,
      Matrix.mulVec, dotProduct, Fin.sum_univ_succ]
    nlinarith [sq_nonneg (β 0 + β 1 - 1/3), sq_nonneg (β 1 - 1/2)]
