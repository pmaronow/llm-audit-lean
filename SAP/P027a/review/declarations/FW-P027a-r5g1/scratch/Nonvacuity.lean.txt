-- Two independent Rademacher errors on a four-point probability space.
def uniform4 : Measure (Fin 4) := (4 : ℝ≥0∞)⁻¹ • Measure.count

instance : IsProbabilityMeasure uniform4 := by
  constructor
  norm_num [uniform4, Measure.smul_apply]
  exact ENNReal.inv_mul_cancel (by norm_num) (by norm_num)

abbrev K2 := Σ _ : Fin 1 × Fin 2, Fin 1

def error2 (k : K2) (ξ : Fin 4) : ℝ :=
  if k.1.2 = 0 then (if ξ.val < 2 then -1 else 1)
  else (if ξ.val % 2 = 0 then -1 else 1)

def centered2 (k : K2) : ℝ := if k.1.2 = 0 then -1 else 1

theorem t2_finite_hypotheses :
    (0 < (1:ℕ)) ∧ (0 < (1:ℕ)) ∧
    Valid (phaseOriginal 1 1 (by norm_num)) ∧
    Positive (phaseOriginal 1 1 (by norm_num)) ∧
    (∑ k : K2, centered2 k) = 0 ∧
    0 < variance (phaseOriginal 1 1 (by norm_num))
      (htTotal (phaseOriginal 1 1 (by norm_num)) centered2) := by
  refine ⟨by norm_num, by norm_num, ?_, ?_, ?_, ?_⟩
  · constructor
    · intro g
      rcases g with ⟨i,b⟩
      fin_cases b <;>
        norm_num [phaseOriginal, liftDesign, phaseGrouped, oneDraw, phaseProb]
    · norm_num [phaseOriginal, liftDesign, phaseGrouped, oneDraw, phaseProb,
        Fintype.sum_prod_type, Fin.sum_univ_succ]
  · intro k
    rcases k with ⟨⟨i,b⟩,j⟩
    fin_cases i
    fin_cases b <;>
      norm_num [pi, expect, indicator, phaseOriginal, liftDesign, phaseGrouped,
        oneDraw, phaseProb, Fintype.sum_prod_type, Fin.sum_univ_succ]
  · norm_num [centered2, Fintype.sum_sigma, Fintype.sum_prod_type, Fin.sum_univ_succ]
  · norm_num [variance, expect, htTotal, pi, indicator, centered2,
      phaseOriginal, liftDesign, phaseGrouped, oneDraw, phaseProb,
      Fintype.sum_sigma, Fintype.sum_prod_type, Fin.sum_univ_succ]

theorem t2_model_nonvacuity :
    ∃ (μ : Measure (Fin 4)) (ε : K2 → Fin 4 → ℝ) (β σ2 : ℝ),

  exact ⟨uniform4, error2, 1, 1, inferInstance, t2_noise, by norm_num, by norm_num⟩

-- An almost-sure cluster partition with a null atom on which its two members
-- disagree. This verifies the support guard has real content.
def nullAtomDesign : Design (Σ _ : Fin 1, Fin 2) (Fin 2) where
  weight ω := if ω = 0 then 1 else 0
  selected ω k := decide (ω = 0 ∨ k.2 = 0)

theorem t4_hypotheses :
    (∀ g : Fin 1, 0 < (fun _ : Fin 1 => 2) g) ∧
    Valid nullAtomDesign ∧ Positive nullAtomDesign ∧
    WholeAEClusters (fun _ : Fin 1 => 2) nullAtomDesign ∧
    nullAtomDesign.selected 1 ⟨0,0⟩ ≠ nullAtomDesign.selected 1 ⟨0,1⟩ := by
  refine ⟨by norm_num, ?_, ?_, ?_, by decide⟩
  · constructor
    · intro ω
      fin_cases ω <;> norm_num [nullAtomDesign]
    · norm_num [nullAtomDesign, Fin.sum_univ_succ]
  · intro k
    rcases k with ⟨g,j⟩
    fin_cases g
    fin_cases j <;>
      norm_num [pi, expect, indicator, nullAtomDesign, Fin.sum_univ_succ]
  · intro g i j
    fin_cases g
    fin_cases i <;> fin_cases j <;>
      norm_num [expect, nullAtomDesign, Fin.sum_univ_succ]

def unequalSizes (g : Fin 2) : ℕ := if g = 0 then 2 else 1

theorem t4_unequal_link :
    (∀ g, 0 < unequalSizes g) ∧
    Valid (liftDesign cluster2 unequalSizes) ∧
    Positive (liftDesign cluster2 unequalSizes) ∧
    WholeAEClusters unequalSizes (liftDesign cluster2 unequalSizes) ∧
    (∃ g h : Fin 2, g ≠ h ∧ joint cluster2 g h = 0 ∧
      unequalSizes g ≠ unequalSizes h) := by
  refine ⟨?_, ?_, ?_, ?_, ?_⟩
  · intro g
    fin_cases g <;> norm_num [unequalSizes]
  · simpa [Valid, liftDesign] using t3_design.2.2.1
  · intro k
    rcases k with ⟨g,j⟩
    fin_cases g <;>
      norm_num [pi, expect, indicator, liftDesign, cluster2, Fin.sum_univ_succ]
  · simp [WholeAEClusters, expect, liftDesign]
  · refine ⟨0,1, by decide, ?_, by decide⟩
    norm_num [joint, expect, indicator, cluster2, Fin.sum_univ_succ]
