noncomputable section

def auditPoint : Measure Unit := Measure.dirac ()

instance : IsProbabilityMeasure auditPoint := by
  dsimp [auditPoint]
  infer_instance

def auditZeroNoise {K : Type} : K → Unit → ℝ := fun _ _ => 0

-- The common-second-moment and uncorrelated hypotheses are jointly satisfiable.
example {K : Type} : NoiseMoments auditPoint (auditZeroNoise (K := K)) 0 := by
  classical
  constructor
  · intro k
    simp [auditZeroNoise, outcomeE]
  · intro k l
    simp [auditZeroNoise, outcomeE]

example {K : Type} : UncorrelatedNoise auditPoint (auditZeroNoise (K := K)) := by
  constructor
  · intro k
    simp [auditZeroNoise, outcomeE]
  · intro k l
    simp [auditZeroNoise, outcomeE]
