def design3 : Design (Fin 3) (Fin 2) where
  weight _ := 1 / 2
  nonneg _ := by norm_num
  total := by norm_num [Fin.sum_univ_succ]
  selected ω k := decide (if ω = 0 then k ≠ 2 else k = 2)

def design2 : Design (Fin 2) (Fin 2) where
  weight _ := 1 / 2
  nonneg _ := by norm_num
  total := by norm_num [Fin.sum_univ_succ]
  selected ω k := decide (ω = k)

lemma design3_positive : Positive design3 := by
  intro k
  fin_cases k <;> norm_num [pi, expect, indicator, design3, Fin.sum_univ_succ]

lemma design2_positive : Positive design2 := by
  intro k
  fin_cases k <;> norm_num [pi, expect, indicator, design2, Fin.sum_univ_succ]

-- T20: an actual witness for its complete existential conjunct, with both
-- differences from the true variance checked independently of the challenge.
example : ∃ (D : Design (Fin 3) (Fin 2)) (yPos yNeg : Fin 3 → ℝ),
    Positive D ∧ 0 < biasA D yPos ∧ biasA D yNeg < 0 ∧
    0 < expect D (htVarianceEstimator D yPos) - SAP.P027b.variance D (htTotal D yPos) ∧
    expect D (htVarianceEstimator D yNeg) - SAP.P027b.variance D (htTotal D yNeg) < 0 := by
  refine ⟨design3, fun _ => 1, ![1, 1, -1/4], design3_positive, ?_⟩
  simp only [biasA, htVarianceEstimator, SAP.P027b.variance, htTotal, inclusionCov,
    pi, joint, expect, indicator, design3, Fin.sum_univ_succ]
  norm_num

-- T21: positivity and conjugacy have a nontrivial zero-pair instance.
example : ∃ (D : Design (Fin 2) (Fin 2)) (y : Fin 2 → ℝ)
    (a b : Fin 2 → Fin 2 → ℝ), Positive D ∧
    (∃ k l, k ≠ l ∧ joint D k l = 0) ∧
    (∀ k l, k ≠ l ∧ joint D k l = 0 → (a k l).HolderConjugate (b k l)) ∧
    biasA D y ≠ 0 := by
  refine ⟨design2, ![1, 2], fun _ _ => 2, fun _ _ => 2,
    design2_positive, ⟨0, 1, by decide, ?_⟩, ?_, ?_⟩
  · norm_num [joint, expect, indicator, design2, Fin.sum_univ_succ]
  · intro k l _
    exact Real.HolderConjugate.two_two
  · norm_num [biasA, joint, expect, indicator, design2, Fin.sum_univ_succ]

-- T22: all hypotheses hold with nonzero, opposite outcomes and a genuine
-- zero-inclusion pair; both the variance and expected estimate equal 36.
example : ∃ (D : Design (Fin 2) (Fin 2)) (y : Fin 2 → ℝ)
    (a b : Fin 2 → Fin 2 → ℝ), Positive D ∧
    (∃ k l, k ≠ l ∧ joint D k l = 0) ∧
    (∀ k l, k ≠ l ∧ joint D k l = 0 → (a k l).HolderConjugate (b k l)) ∧
    (∀ k l, k ≠ l ∧ joint D k l = 0 → |y k| ^ (a k l) = |y l| ^ (b k l)) ∧
    (∀ k l, k ≠ l ∧ joint D k l = 0 → -y k * y l = |y k| * |y l|) ∧
    SAP.P027b.variance D (htTotal D y) = 36 ∧
    expect D (conservativeEstimator D y a b) = 36 := by
  refine ⟨design2, ![3, -3], fun _ _ => 2, fun _ _ => 2,
    design2_positive, ⟨0, 1, by decide, ?_⟩, ?_, ?_, ?_, ?_⟩
  · norm_num [joint, expect, indicator, design2, Fin.sum_univ_succ]
  · intro k l _
    exact Real.HolderConjugate.two_two
  · intro k l _
    fin_cases k <;> fin_cases l <;> norm_num
  · intro k l h
    fin_cases k <;> fin_cases l <;> norm_num at *
  · norm_num [conservativeEstimator, youngCorrection, htVarianceEstimator,
      SAP.P027b.variance, htTotal, inclusionCov, pi, joint, expect, indicator, design2,
      Fin.sum_univ_succ, Real.rpow_two]

-- T19: fixed-population and decomposition premises, with q=4, m=2 and
-- a centered nonzero signal vector.
example : ∃ (r m : ℕ) (z : (Σ _ : Fin r × Fin 2, Fin m) → ℝ),
    0 < r ∧ 2 ≤ m ∧ (∑ k, z k) = 0 ∧ ∃ k, z k ≠ 0 := by
  refine ⟨2, 2, fun k => if k.1.2 = 0 then -4 else 4, by norm_num,
    by norm_num, ?_, ⟨⟨(0, 0), 0⟩, by norm_num⟩⟩
  norm_num [Fintype.sum_sigma, Fintype.sum_prod_type, Fin.sum_univ_succ]

-- Product Gaussian coordinates give the positive-noise hypotheses of T19.
lemma gaussian_coordinates (K : Type) [Fintype K] (s : ℝ) :
    GaussianNoise (Measure.pi (fun _ : K => gaussianReal 0 s.toNNReal))
      (fun k (ξ : K → ℝ) => ξ k) s := by
  constructor
  · intro k
    exact ⟨(measurable_pi_apply k).aemeasurable,
      (measurePreserving_eval (fun _ : K => gaussianReal 0 s.toNNReal) k).map_eq⟩
  · exact iIndepFun_pi (X := fun _ => id) (fun _ => aemeasurable_id)

-- The same construction applies to each of the four row populations.
example (row : Fin 4) :
    let U := Σ _ : Fin (a3HighR row) × Fin 2, Fin (a3HighM row)
    let μ := Measure.pi (fun _ : U => gaussianReal 0 (17.91:ℝ).toNNReal)
    IsProbabilityMeasure μ ∧ GaussianNoise μ (fun k ξ => ξ k) 17.91 ∧
    0 < a3HighR row ∧ 2 ≤ a3HighM row ∧ (0:ℝ) < 17.91 := by
  dsimp
  refine ⟨inferInstance, gaussian_coordinates _ _, ?_, ?_, by norm_num⟩ <;>
    fin_cases row <;> norm_num [a3HighR, a3HighM]

-- Moment-only model: zero variance is allowed and its assumptions are real.
