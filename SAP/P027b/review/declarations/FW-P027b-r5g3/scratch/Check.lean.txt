import SAP.P027b.Defs

noncomputable section
open scoped BigOperators
open SAP.P027b

namespace ReviewT24

-- Draw exactly one of two units, each with probability one half.
def twoPointDesign : Design (Fin 2) (Fin 2) where
  weight _ := 1 / 2
  nonneg _ := by norm_num
  total := by norm_num [Fin.sum_univ_two]
  selected ω k := decide (ω = k)

def outcomes (k : Fin 2) : ℝ := if k = 0 then -2 else 3

lemma positive : Positive twoPointDesign := by
  intro k
  fin_cases k <;>
    norm_num [pi, expect, indicator, twoPointDesign, Fin.sum_univ_two]

lemma zeroJoint : joint twoPointDesign 0 1 = 0 := by
  norm_num [joint, expect, indicator, twoPointDesign, Fin.sum_univ_two]

lemma conjugate : (3 : ℝ).HolderConjugate (3 / 2) := by
  constructor <;> norm_num

-- Every substantive hypothesis is satisfied, the conjugate pair need not be (2,2),
-- and the zero-pair correction is active on the chosen positive-probability sample.
example :
    ∃ (D : Design (Fin 2) (Fin 2)) (y : Fin 2 → ℝ) (ω : Fin 2) (a b : ℝ),
      Positive D ∧ a.HolderConjugate b ∧ D.weight ω > 0 ∧
      joint D 0 1 = 0 ∧ y 0 = -2 ∧ y 1 = 3 ∧
      quadraticCorrection D y ω = 8 ∧ c2Estimator D y ω = 16 := by
  refine ⟨twoPointDesign, outcomes, 0, 3, 3 / 2,
    positive, conjugate, ?_, zeroJoint, ?_, ?_, ?_, ?_⟩
  all_goals
    norm_num [twoPointDesign, outcomes, quadraticCorrection, c2Estimator,
      htVarianceEstimator, inclusionCov, joint, pi, expect, indicator,
      Fin.sum_univ_two]

-- Check the other sample as well; neither the estimator nor its correction is zero.
example : quadraticCorrection twoPointDesign outcomes 1 = 18 ∧
    c2Estimator twoPointDesign outcomes 1 = 36 := by
  norm_num [twoPointDesign, outcomes, quadraticCorrection, c2Estimator,
    htVarianceEstimator, inclusionCov, joint, pi, expect, indicator,
    Fin.sum_univ_two]

-- The claimed attaining pair is actually a member of the comparison class.
example : (2 : ℝ).HolderConjugate 2 := Real.HolderConjugate.two_two

end ReviewT24
