def coin : Measure Bool :=
  bernoulliMeasure true false ⟨1/2, by constructor <;> norm_num⟩
instance : IsProbabilityMeasure coin := by unfold coin; infer_instance
def square : Measure (Bool × Bool) := coin.prod coin
instance : IsProbabilityMeasure square := by unfold square; infer_instance
def cube : Measure ((Bool × Bool) × Bool) := square.prod coin
instance : IsProbabilityMeasure cube := by unfold cube; infer_instance

lemma mass_sum {α : Type*} [Fintype α] [MeasurableSpace α]
    [MeasurableSingletonClass α] [DiscreteMeasurableSpace α]
    (μ : Measure α) [IsFiniteMeasure μ] (E : Set α) :
    μ.real E = ∑ i, if i ∈ E then μ.real {i} else 0 := by
  rw [← integral_indicator_one (by measurability), integral_fintype Integrable.of_finite]
  simp [Set.indicator_apply, smul_eq_mul, mul_ite]

lemma on_sum {α : Type*} [Fintype α] [MeasurableSpace α]
    [MeasurableSingletonClass α] [DiscreteMeasurableSpace α]
    (μ : Measure α) [IsFiniteMeasure μ] (f : α → ℝ) (E : Set α) :
    (∫ i in E, f i ∂μ) = ∑ i, if i ∈ E then μ.real {i} * f i else 0 := by
  rw [← integral_indicator (by measurability), integral_fintype Integrable.of_finite]
  simp [Set.indicator_apply, smul_eq_mul, mul_ite]

@[simp] lemma coin_singleton (b : Bool) : coin.real {b} = 1/2 := by
  cases b <;> norm_num [coin, bernoulliMeasure_real_apply _ (measurableSet_singleton _)]

lemma prod_real_singleton {α β : Type*} [MeasurableSpace α] [MeasurableSpace β]
    (μ : Measure α) (ν : Measure β) [SFinite ν] (a : α) (b : β) :
    (μ.prod ν).real {(a,b)} = μ.real {a} * ν.real {b} := by
  simp only [measureReal_def, ← Set.singleton_prod_singleton, Measure.prod_prod,
    ENNReal.toReal_mul]

@[simp] lemma square_singleton (b : Bool × Bool) : square.real {b} = 1/4 := by
  rcases b with ⟨b,c⟩
  norm_num [square, prod_real_singleton]
@[simp] lemma cube_singleton (b : (Bool × Bool) × Bool) : cube.real {b} = 1/8 := by
  rcases b with ⟨b,c⟩
  norm_num [cube, prod_real_singleton]

lemma coin_mass (E : Set Bool) : coin.real E =
    (if true ∈ E then (1/2:ℝ) else 0) + (if false ∈ E then (1/2:ℝ) else 0) := by
  rw [mass_sum]
  simp
lemma square_mass (E : Set (Bool × Bool)) : square.real E =
    ∑ b : Bool, ∑ c : Bool, if (b,c) ∈ E then (1/4:ℝ) else 0 := by
  rw [mass_sum]
  simp [Fintype.sum_prod_type]
lemma cube_mass (E : Set ((Bool × Bool) × Bool)) : cube.real E =
    ∑ b : Bool, ∑ c : Bool, ∑ d : Bool, if ((b,c),d) ∈ E then (1/8:ℝ) else 0 := by
  rw [mass_sum]
  simp [Fintype.sum_prod_type]

-- T8: a full-compliance model satisfying every structure hypothesis.
def fullModel : CausalModel Bool where
  μ := coin
  probability := inferInstance
  S := fun _ => 2
  Y := fun d _ => realBit d
  Z := id
  measurableS := measurable_const
  measurableY := fun _ => measurable_const
  integrableY := fun _ => integrable_const _
  measurableZ := measurable_id
  randomAssignment := indepFun_const_left _ _
  armPositive := by
    intro z
    rw [coin_mass]
    cases z <;> norm_num
  nonzeroFirstStage := by
    norm_num [treatment, realBit, coin_mass]

example : boundedOutcomes fullModel ∧ complierProbability fullModel = 1 ∧
    (∀ z, armMean fullModel z = realBit z) ∧
    (∀ d, complierMean fullModel d = realBit d) := by
  constructor
  · intro d ω; cases d <;> norm_num [fullModel, realBit]
  constructor
  · norm_num [complierProbability, complierEvent, fullModel, coin_mass]
  constructor
  · intro z
    simp only [armMean, eventMean, armEvent, observedY, observedD, fullModel]
    rw [on_sum, coin_mass]
    cases z <;> norm_num [Fintype.sum_bool, treatment, realBit]
  · intro d
    norm_num [complierMean, eventMean, complierEvent, fullModel, coin_mass]

-- T8 and T10: genuine independent fair assignment in a partial-compliance family.
def levelsModel (t : ℝ) : CausalModel (Bool × Bool) where
  μ := square
  probability := inferInstance
  S := fun ω => if ω.1 then 2 else 1
  Y := fun _ ω => if ω.1 then t else 3/2-t
  Z := Prod.snd
  measurableS := Measurable.of_discrete
  measurableY := fun _ => Measurable.of_discrete
  integrableY := fun _ => Integrable.of_finite
  measurableZ := measurable_snd
  randomAssignment := indepFun_prod (X := fun b : Bool =>
    ((if b then 2 else 1 : Fin 3), (if b then t else 3/2-t),
      (if b then t else 3/2-t))) Measurable.of_discrete measurable_id
  armPositive := by
    intro z
    rw [square_mass]
    cases z <;> norm_num [Fintype.sum_bool]
  nonzeroFirstStage := by
    rw [integral_fintype Integrable.of_finite]
    norm_num [Fintype.sum_prod_type, Fintype.sum_bool, treatment, realBit]

lemma levels_bounded (t : ℝ) (ht : t ∈ Set.Icc (1/2:ℝ) 1) :
    boundedOutcomes (levelsModel t) := by
  intro d ω
  dsimp [levelsModel]
  split_ifs <;> constructor <;> linarith [ht.1, ht.2]

lemma levels_values (t : ℝ) : complierProbability (levelsModel t) = 1/2 ∧
    (∀ z, armMean (levelsModel t) z = 3/4) ∧
    cace (levelsModel t) = 0 ∧ (∀ d, complierMean (levelsModel t) d = t) := by
  constructor
  · simp only [complierProbability, complierEvent, levelsModel]
    rw [square_mass]
    norm_num [Fintype.sum_bool]
  constructor
  · intro z
    simp only [armMean, eventMean, observedY, observedD, armEvent, levelsModel]
    rw [on_sum, square_mass]
    cases z <;> norm_num [Fintype.sum_prod_type, Fintype.sum_bool] <;> ring
  constructor
  · simp [cace, eventMean, levelsModel]
  · intro d
    simp only [complierMean, eventMean, complierEvent, levelsModel]
    rw [on_sum, square_mass]
    norm_num [Fintype.sum_prod_type, Fintype.sum_bool]
    ring

-- This exactly verifies the two-model existential of T10 without importing its theorem.
example : nonidentificationOfLevels := by
  refine ⟨levelsModel (1/2), levelsModel 1, levels_bounded _ (by norm_num),
    levels_bounded _ (by norm_num), ?_⟩
  have h0 := levels_values (1/2)
  have h1 := levels_values 1
  exact ⟨h0.1, h1.1, h0.2.1 false, h0.2.1 true, h1.2.1 false, h1.2.1 true,
    h0.2.2.1, h1.2.2.1, h0.2.2.2 false, h0.2.2.2 true,
    h1.2.2.2 false, h1.2.2.2 true⟩
