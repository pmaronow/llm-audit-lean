-- T6: a genuine probability law for every independent uniform coordinate,
-- without invoking any Challenge theorem.
local instance uniform_probability : IsProbabilityMeasure uniformCovariateLaw := by
  unfold uniformCovariateLaw
  apply cond_isProbabilityMeasure_of_finite
  · norm_num [Real.volume_Ioo]
  · norm_num [Real.volume_Ioo]

abbrev UniformIndex := Fin 1 × Fin 3
abbrev UniformOmega := UniformIndex → ℝ
def uniformProduct : Measure UniformOmega := Measure.pi (fun _ : UniformIndex => uniformCovariateLaw)

example : IsProbabilityMeasure uniformProduct := by
  unfold uniformProduct
  infer_instance

example :
    (∀ i : Fin 1, ∀ j : Fin 3, Measurable (fun ω : UniformOmega => ω (i,j))) ∧
    iIndepFun (fun ij : UniformIndex => fun ω : UniformOmega => ω ij) uniformProduct ∧
    (∀ i : Fin 1, ∀ j : Fin 3,
      Measure.map (fun ω : UniformOmega => ω (i,j)) uniformProduct = uniformCovariateLaw) := by
  refine ⟨fun i j => measurable_pi_apply (i,j), ?_, ?_⟩
  · exact iIndepFun_pi (X := fun _ => id) (fun _ => aemeasurable_id)
  · intro i j
    exact (measurePreserving_eval (fun _ : UniformIndex => uniformCovariateLaw) (i,j)).map_eq

-- T10: a positive distinguished group with a nonzero within-group term.
def groupExample (i : Fin 4) : Bool := decide (i.val < 2)
def scoreExample : Fin 4 → ℝ := ![0,1,2,3]
def indicatorExample : Fin 4 → ℝ := ![0,1,2,2]

example :
    0 < finiteProbability (fun _ : Fin 4 => (1/4 : ℝ)) (fun i => groupExample i = true) ∧
    (∀ i : Fin 4, groupExample i ≠ true →
      ∀ j : Fin 4, groupExample j = groupExample i → indicatorExample j = indicatorExample i) := by
  constructor
  · norm_num [finiteProbability, groupExample, Fin.sum_univ_succ]
  · intro i hi j hj
    fin_cases i <;> fin_cases j <;> norm_num [groupExample, indicatorExample] at *

example : finiteExpectation (fun _ : Fin 4 => (1/4 : ℝ))
    (fun i => (scoreExample i - finiteGroupMean (fun _ => (1/4 : ℝ)) scoreExample groupExample (groupExample i)) *
      (indicatorExample i - finiteGroupMean (fun _ => (1/4 : ℝ)) indicatorExample groupExample (groupExample i))) = 1/8 := by
  norm_num [finiteExpectation, finiteGroupMean, finiteConditionalMean, finiteProbability,
    scoreExample, indicatorExample, groupExample, Fin.sum_univ_succ]
