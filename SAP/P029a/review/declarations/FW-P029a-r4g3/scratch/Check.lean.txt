-- Independence of an arbitrary map of a finite latent variable and its
-- independently sampled Boolean assignment, checked without Challenge stubs.
lemma product_independent {L A : Type*} [Fintype L]
    (p : L → ℝ) (hp : ∑ i, p i = 1) (e : ℝ) (U : L → A) :
    FiniteIndependent (bernoulliProductMass p e) (fun i => U i.1) Prod.snd := by
  intro a b
  have hsum : ∀ x, p x * e + p x * (1 - e) = p x := by
    intro x
    ring
  have htrue : (∑ x, p x * e) = e := by
    rw [← Finset.sum_mul, hp]
    ring
  have hfalse : (∑ x, p x * (1 - e)) = 1 - e := by
    rw [← Finset.sum_mul, hp]
    ring
  cases b <;>
    simp [finiteProbability, bernoulliProductMass, Fintype.sum_prod_type,
      Fintype.sum_bool, hsum, htrue, hfalse] <;>
    rw [Finset.sum_mul] <;>
    apply Finset.sum_congr rfl <;>
    intro x hx <;> split_ifs <;> ring

    (∀ i, Y1 i = β0 + β1 + (∑ k, coeff k * Q i k) + u i) ∧

  refine ⟨fun _ => 5, fun _ => 6, fun _ => 2, fun _ _ => 1,
    0, 1, fun _ => 3, ?_⟩
  norm_num
