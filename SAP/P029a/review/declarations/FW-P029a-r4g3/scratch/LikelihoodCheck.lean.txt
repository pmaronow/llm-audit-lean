-- T13: all four specified points really maximize the Bernoulli likelihood.
example : ∀ d, twoReceiptFit d ∈ closedReceiptTriangle ∧
    ∀ v ∈ closedReceiptTriangle, twoReceiptLikelihood d v.1 v.2 ≤
      twoReceiptLikelihood d (twoReceiptFit d).1 (twoReceiptFit d).2 := by
  rintro ⟨d1,d0⟩
  cases d1 <;> cases d0 <;>
    refine ⟨by norm_num [twoReceiptFit, closedReceiptTriangle], ?_⟩
  all_goals
    rintro ⟨p,q⟩ hv
    change 0 ≤ q ∧ q ≤ p ∧ p ≤ 1 at hv
    rcases hv with ⟨hq0,hqp,hp1⟩
    have hp0 : 0 ≤ p := hq0.trans hqp
    have hq1 : q ≤ 1 := hqp.trans hp1
    have hprod1 := mul_nonneg (sub_nonneg.mpr hp1) hq0
    have hprod2 := mul_nonneg (sub_nonneg.mpr hp1) (sub_nonneg.mpr hqp)
    have hprod3 := mul_nonneg hp0 hq0
    have hprod4 := mul_nonneg hp0 (sub_nonneg.mpr hq1)
    norm_num [twoReceiptLikelihood, bernoulliContribution, twoReceiptFit]
    nlinarith [sq_nonneg (p - 1/2)]
