-- T15: verify the actual six-atom witness directly from its definitions.
lemma valid_six : ValidLaw sixMass := by
  constructor
  · rintro ⟨i,z⟩
    fin_cases i <;> cases z <;> norm_num [sixMass, randomized, typeMass]
  · norm_num [sixMass, randomized, typeMass, Fintype.sum_prod_type, Fin.sum_univ_succ]

lemma independent_six : FIndep sixMass
    (fun i => ((liftUnit constantUnit i).x, (liftUnit constantUnit i).d0,
      (liftUnit constantUnit i).d1, (liftUnit constantUnit i).y0,
      (liftUnit constantUnit i).y1)) Prod.snd := by
  intro a z
  cases z <;>
    norm_num [fP, sixMass, randomized, typeMass, liftUnit, constantUnit,
      Fintype.sum_prod_type, Fin.sum_univ_succ] <;>
    split_ifs <;> norm_num

example : Type7Quantile (fun i : Fin 2 => if i = 0 then (1/4 : ℝ) else 3/4)
    (1/2) (1/2) := by
  refine ⟨Equiv.refl _, ?_, ⟨0, ?_, ?_, ?_⟩⟩
  · intro i j hij
    fin_cases i <;> fin_cases j <;> norm_num at *
  · norm_num
  · norm_num
  · norm_num [nextOrderIndex]

example : belowCount (fun i : Fin 2 => if i = 0 then (1/4 : ℝ) else 3/4) (1/2) = 1 ∧
    weightChange (1/4) (1/2) = 2 := by
  constructor
  · have h : (Finset.univ.filter (fun i : Fin 2 =>
        (if i = 0 then (1/4 : ℝ) else 3/4) < 1/2)) = {0} := by
      ext i
      fin_cases i <;> norm_num
    change (Finset.univ.filter _).card = 1
    rw [h]
    rfl
  · norm_num [weightChange]
