private def legalFour : Finset (Fin 4) := {0, 1}

example : Admissible legalFour := by
  change 1 < legalFour.card ∧ 1 < legalFourᶜ.card
  constructor
  · decide
  · have h : legalFourᶜ = ({2, 3} : Finset (Fin 4)) := by
      ext i
      fin_cases i <;> decide
    rw [h]
    decide
example : legalFour.card = 2 := by decide

example : Fintype.card (Fin 4) = 2 * legalFour.card := by decide
