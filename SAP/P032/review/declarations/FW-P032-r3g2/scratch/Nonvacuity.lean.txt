-- These checks use definitions and Mathlib only, never the Challenge stubs.

lemma law_probability {S : Type} [Fintype S] [MeasurableSpace S]
    (w : FiniteDistribution S) : IsProbabilityMeasure (distributionMeasure w) := by
  constructor
  simp only [distributionMeasure, Measure.finsetSum_apply, Measure.smul_apply,
    Measure.dirac_apply_of_mem (Set.mem_univ _), smul_eq_mul, mul_one]
  rw [← ENNReal.ofReal_sum_of_nonneg (fun s _ => w.nonneg s), w.total]
  norm_num

lemma law_atom {S : Type} [Fintype S] [DecidableEq S] [MeasurableSpace S]
    [MeasurableSingletonClass S] (w : FiniteDistribution S) (a : S) :
    distributionMeasure w {a} = ENNReal.ofReal (w.weight a) := by
  simp [distributionMeasure, Measure.finsetSum_apply, Measure.smul_apply,
    Measure.dirac_apply, Set.indicator_apply, eq_comm]

lemma law_iid {S : Type} [Fintype S] [DecidableEq S] [MeasurableSpace S]
    [MeasurableSingletonClass S] (w : FiniteDistribution S) :
    iidRealizationExists w.weight := by
  letI := law_probability w
  refine ⟨ℕ → S, inferInstance, Measure.infinitePi (fun _ : ℕ => distributionMeasure w),
    fun b ω => ω b, inferInstance, ?_, ?_, ?_⟩
  · intro b
    fun_prop
  · exact iIndepFun_infinitePi (X := fun _ x => x) (fun _ => measurable_id)
  · intro b a
    have h := (measurePreserving_eval_infinitePi (fun _ : ℕ => distributionMeasure w) b).map_eq
    have ha := congrArg (fun ν : Measure S => ν {a}) h
    rw [Measure.map_apply (by fun_prop) (measurableSet_singleton a)] at ha
    exact ha.trans (law_atom w a)

-- T5: a genuine full-rank, positive-definite, nonincident covariance witness.
lemma pair_psd : (pairCovariance (-1)).PosSemidef := by
  rw [Matrix.posSemidef_iff_dotProduct_mulVec]
  constructor
  · ext i j
    fin_cases i <;> fin_cases j <;> norm_num [pairCovariance, Matrix.IsHermitian]
  · intro v
    simp [pairCovariance, dotProduct, Matrix.mulVec, Fin.sum_univ_two, star_trivial]
    nlinarith [sq_nonneg (v 0 - v 1), sq_nonneg (v 0 + v 1)]

lemma pair_bread : olsBread pairIntercept = (fun _ _ => (1/2 : ℝ)) := by
  unfold olsBread
  apply Matrix.inv_eq_left_inv
  ext i j
  fin_cases i
  fin_cases j
  norm_num [Matrix.mul_apply, Matrix.transpose_apply, pairIntercept, Fin.sum_univ_succ]
