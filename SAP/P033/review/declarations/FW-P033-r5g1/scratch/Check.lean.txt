    by norm_num, by norm_num, by norm_num, ?_, ?_⟩

    have hv : (⟨(1:ℝ)^2, sq_nonneg (1:ℝ)⟩ : NNReal) = 1 := by

      norm_num

-- All hypotheses of the independent-cluster plug-in identity, with positive
-- common variance. This is an explicit product of two standard normals.
def testPi : Measure (Fin 2 → ℝ) := Measure.pi (fun _ => gaussianReal 0 1)

example : IsProbabilityMeasure testPi ∧
    (∀ g : Fin 2, MemLp (fun ω : Fin 2 → ℝ => ω g) 2 testPi) ∧
    iIndepFun (fun g (ω : Fin 2 → ℝ) => ω g) testPi ∧
    (∀ g : Fin 2, (∫ ω, ω g ∂testPi) = 0) ∧
    (∀ g : Fin 2, variance (fun ω : Fin 2 → ℝ => ω g) testPi = 1) := by
  have hlaw (g : Fin 2) : HasLaw (fun ω : Fin 2 → ℝ => ω g)
      (gaussianReal 0 1) testPi :=
    (measurePreserving_eval (fun _ : Fin 2 => gaussianReal 0 1) g).hasLaw
  refine ⟨inferInstanceAs (IsProbabilityMeasure (Measure.pi
    (fun _ : Fin 2 => gaussianReal 0 1))), ?_, ?_, ?_, ?_⟩
  · intro g
    exact (hlaw g).memLp (memLp_id_gaussianReal' 2 (by norm_num))
  · exact iIndepFun_pi (X := fun _ => id) (fun _ => aemeasurable_id)
  · intro g
    rw [(hlaw g).integral_eq, integral_id_gaussianReal]
  · intro g
    rw [(hlaw g).variance_eq, variance_id_gaussianReal]
    rfl

-- The separate finite-type second-moment premise is satisfiable too.
example : ∃ Z : Fin 10 → Unit → ℝ, ∀ g,
    finiteMean (fun ω => (Z g ω - sampleMean (fun j => Z j ω))^2) = (9/10:ℝ) := by
  let t : ℝ := Real.sqrt (9/10)
  have ht : t^2 = 9/10 := Real.sq_sqrt (by norm_num)
  let v : Fin 10 → ℝ := fun g => if g.val < 5 then t else -t
  have hm : sampleMean v = 0 := by
    norm_num [v, sampleMean, Fin.sum_univ_succ] <;> ring
  have hv (g : Fin 10) : (v g)^2 = 9/10 := by
    dsimp [v]
    split_ifs <;> simpa only [neg_sq] using ht
  refine ⟨fun g _ => v g, ?_⟩
  intro g
  change finiteMean (fun _ : Unit => (v g - sampleMean v)^2) = 9/10
  rw [hm]
  simpa [finiteMean] using hv g
