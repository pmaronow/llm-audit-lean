instance : MeasurableSpace Bool := ⊤

noncomputable def μ : Measure Bool :=
  (1 / 2 : ℝ≥0∞) • (Measure.dirac false + Measure.dirac true)

instance : IsProbabilityMeasure μ := by
  constructor
  norm_num [μ]
  apply (ENNReal.toReal_eq_toReal_iff' (by finiteness) (by finiteness)).mp
  rw [ENNReal.toReal_add (by finiteness) (by finiteness)]
  norm_num

def P : Set Bool := {true}
def X : Bool → Fin 1 → ℝ := fun _ _ => 1
def γ : Fin 1 → ℝ := fun _ => 3
noncomputable def adjustedY (ω : Bool) : ℝ := 5 + 5 * peers P ω
noncomputable def unadjustedY (ω : Bool) : ℝ := 2 + 5 * peers P ω

-- Every hypothesis of the adjusted probability-model conjunct, with nonzero
-- slope and nonzero covariate contribution, and both arms of probability 1/2.
example :
    MeasurableSet P ∧ Measurable X ∧
    Integrable (fun _ : Bool => (0 : ℝ)) μ ∧
    Integrable (covariateContribution X γ) μ ∧
    adjustedY =ᵐ[μ] (fun ω => linearPredictor P X 2 5 γ ω + 0) ∧
    μ[(fun _ : Bool => (0 : ℝ)) | regressorSigma P X] =ᵐ[μ] (fun _ => 0) ∧
    μ.real P = 1 / 2 := by
  refine ⟨MeasurableSet.of_discrete, measurable_const, integrable_zero _ _ _, ?_, ?_, ?_, ?_⟩
  · simp [covariateContribution, X, γ]
  · filter_upwards with ω
    simp [adjustedY, linearPredictor, covariateContribution, X, γ]
    ring
  · simp
  · norm_num [μ, P, Measure.real]

-- Every hypothesis of the unadjusted probability-model conjunct.
example :
    MeasurableSet P ∧ Integrable (fun _ : Bool => (0 : ℝ)) μ ∧
    unadjustedY =ᵐ[μ] (fun ω => 2 + 5 * peers P ω + 0) ∧
    μ[(fun _ : Bool => (0 : ℝ)) | assignmentSigma P] =ᵐ[μ] (fun _ => 0) ∧
    0 < μ.real P ∧ μ.real P < 1 := by
  refine ⟨MeasurableSet.of_discrete, integrable_zero _ _ _, ?_, ?_, ?_, ?_⟩
  · exact Filter.Eventually.of_forall (by intro ω; simp [unadjustedY])
  · simp
  · norm_num [μ, P, Measure.real]
  · norm_num [μ, P, Measure.real]

-- A genuine witness for T1's existential conclusion, checked directly from
-- squared residuals without importing the Challenge theorem.
example : 0 < ({0} : Finset (Fin 2)).card ∧
    0 < ({0} : Finset (Fin 2))ᶜ.card ∧
    ∃! b : ℝ × ℝ, ∀ c : ℝ × ℝ,
      rss ({0} : Finset (Fin 2)) ![4, 1] b.1 b.2 ≤
        rss ({0} : Finset (Fin 2)) ![4, 1] c.1 c.2 := by
  refine ⟨by decide, by decide, (1, 3), ?_, ?_⟩
  · intro c
    simp [rss, peers, Fin.sum_univ_succ]
    nlinarith [sq_nonneg (4 - c.1 - c.2), sq_nonneg (1 - c.1)]
  · intro b hb
    have h := hb (1, 3)
    simp [rss, peers, Fin.sum_univ_succ] at h
    have h0 : (1 - b.1) ^ 2 = 0 := by
      nlinarith [sq_nonneg (4 - b.1 - b.2)]
    have h1 : (4 - b.1 - b.2) ^ 2 = 0 := by
      nlinarith [sq_nonneg (1 - b.1)]
    have e0 := (sq_eq_zero_iff).mp h0
    have e1 := (sq_eq_zero_iff).mp h1
    apply Prod.ext <;> dsimp <;> linarith

def smallArm : Finset (Fin 4) := {0, 1}
def smallY : Fin 4 → ℝ := ![0, 1, 2, 4]
def Y₁ : Fin 4 → ℝ := ![0, 1, 1, 0]
def Y₀ : Fin 4 → ℝ := ![0, 0, 1, 0]

-- Every hypothesis of T1's least-squares conjunct and T2's sandwich conjunct.
example : 0 < smallArm.card ∧ 0 < smallArmᶜ.card ∧
    2 ≤ smallArm.card ∧ 2 ≤ smallArmᶜ.card := by
  decide

-- Every randomization hypothesis, nonempty support, and heterogeneous effects.
example : 2 ≤ (2 : ℕ) ∧ 2 ≤ (2 : ℕ) ∧ 2 + 2 = Fintype.card (Fin 4) ∧
    (assignments (ι := Fin 4) 2).Nonempty ∧
    ¬ (∃ c : ℝ, ∀ i, unitEffect Y₁ Y₀ i = c) := by
  refine ⟨by norm_num, by norm_num, by norm_num, ?_, ?_⟩
  · refine ⟨smallArm, ?_⟩
    simp [assignments, smallArm]
  · rintro ⟨c, hc⟩
    have h0 := hc 0
    have h1 := hc 1
    norm_num [unitEffect, Y₁, Y₀] at h0 h1
    linarith

-- The zero-gap case is also genuinely possible under the same design.
example : 2 ≤ (2 : ℕ) ∧ 2 ≤ (2 : ℕ) ∧ 2 + 2 = Fintype.card (Fin 4) ∧
    ∃ c : ℝ, ∀ i : Fin 4, unitEffect (fun i => Y₀ i + 2) Y₀ i = c := by
  refine ⟨by norm_num, by norm_num, by norm_num, 2, ?_⟩
  intro i
  simp [unitEffect]

-- Every hypothesis of the separate binary single-arm variance identity.
example : 2 ≤ smallArm.card ∧
    (∀ i ∈ smallArm, Y₁ i = 0 ∨ Y₁ i = 1) ∧
    (∑ i ∈ smallArm, Y₁ i) = (1 : ℝ) := by
  norm_num [smallArm, Y₁]

def tableArm : Finset (Fin 189) := Finset.univ.filter (fun i => i.val < 94)
def tableY (k₀ : ℕ) (i : Fin 189) : ℝ :=
  if i.val < 10 ∨ (94 ≤ i.val ∧ i.val < 94 + k₀) then 1 else 0

-- Both real table count inputs satisfy every count hypothesis of T1 and T2.
example : tableArm.card = 94 ∧ tableArmᶜ.card = 95 := by
  decide

example (k₀ : ℕ) : ∀ i : Fin 189, tableY k₀ i = 0 ∨ tableY k₀ i = 1 := by
  intro i
  unfold tableY
  split_ifs <;> simp

example : (∑ i ∈ tableArm, tableY 10 i) = (10 : ℝ) ∧
    (∑ i ∈ tableArmᶜ, tableY 10 i) = (10 : ℝ) ∧
    (∑ i ∈ tableArm, tableY 7 i) = (10 : ℝ) ∧
    (∑ i ∈ tableArmᶜ, tableY 7 i) = (7 : ℝ) := by
  norm_num [tableY]
  norm_cast
