local instance uniform_probability : IsProbabilityMeasure uniformDraw := by
  constructor
  simp [uniformDraw, Real.volume_Ioo]
local instance exponential_probability : IsProbabilityMeasure (expMeasure 1) :=
  isProbabilityMeasure_expMeasure (by norm_num)

-- All of bootstrapRandomComparison's input assumptions, on one sample space,
-- with no audited theorem used as a lemma.
noncomputable def mixedInput (j : Fin 2 × Bool) : Measure ℝ :=
  if j.2 then expMeasure 1 else uniformDraw
local instance mixed_probability (j : Fin 2 × Bool) : IsProbabilityMeasure (mixedInput j) := by
  unfold mixedInput
  split <;> infer_instance
noncomputable def commonLaw : Measure ((Fin 2 × Bool) → ℝ) := Measure.pi mixedInput
local instance common_probability : IsProbabilityMeasure commonLaw := by
  unfold commonLaw
  infer_instance

def U (i : Fin 2) (x : (Fin 2 × Bool) → ℝ) : ℝ := x (i,false)
def E (i : Fin 2) (x : (Fin 2 × Bool) → ℝ) : ℝ := x (i,true)

example : IsProbabilityMeasure commonLaw ∧
    (∀ i, Measurable (U i)) ∧ (∀ i, Measurable (E i)) ∧
    iIndepFun U commonLaw ∧ iIndepFun E commonLaw ∧
    (∀ i, Measure.map (U i) commonLaw = uniformDraw) ∧
    (∀ i, Measure.map (E i) commonLaw = expMeasure 1) := by
  have h : iIndepFun (fun j (x : (Fin 2 × Bool) → ℝ) => x j) commonLaw :=
    iIndepFun_pi (X := fun _ => id) (fun _ => aemeasurable_id)
  refine ⟨inferInstance, (fun i => measurable_pi_apply (i,false)),
    (fun i => measurable_pi_apply (i,true)), ?_, ?_, ?_, ?_⟩
  · exact iIndepFun.precomp (g := fun i : Fin 2 => (i,false))
      (fun _ _ heq => congrArg Prod.fst heq) h
  · exact iIndepFun.precomp (g := fun i : Fin 2 => (i,true))
      (fun _ _ heq => congrArg Prod.fst heq) h
  · intro i
    exact (measurePreserving_eval mixedInput (i,false)).map_eq
  · intro i
    exact (measurePreserving_eval mixedInput (i,true)).map_eq

-- A positive-variation, nonsingular branch of responseIdentification.
noncomputable def oneDesign : Design (Fin 1) where
  weight := fun _ => 1
  nonneg := fun _ => by norm_num
  total := by simp
noncomputable def X (_k : Fin 1) (_w : Fin 1) (i : Fin 1 × Fin 4) : ℝ :=
  if i.2 = 0 then 1 else 0
noncomputable def Y (_w : Fin 1) (i : Fin 1 × Fin 4) : ℝ :=
  if i.2 = 0 then 2 else 0

example :
    (∀ w i, Y w i = (0 : ℝ) + (∑ k : Fin 1, (2 : ℝ)*X k w i) + 0) ∧
    (∀ k : Fin 1, mean oneDesign (fun w => centeredBlockCross (X k w) (fun _ => 0)) = 0) ∧
    Function.Injective (fun v : Fin 1 → ℝ => fun k =>
      ∑ l, v l * mean oneDesign (fun w => centeredBlockCross (X k w) (X l w))) := by
  refine ⟨?_, ?_, ?_⟩
  · intro w i
    simp [Y, X]
  · intro k
    simp [mean, centeredBlockCross]
  · intro v v' h
    have h0 := congrFun h 0
    norm_num [mean, oneDesign, centeredBlockCross, X, Fin.sum_univ_succ] at h0
    funext k
    fin_cases k
    simpa using h0

-- The explicit inverse coefficient is on a genuine nonsingular fit.
example : wlsCoefficient (![1,2] : Fin 2 → ℝ)
    (fun _ _ => 1 : Matrix (Fin 2) (Fin 1) ℝ) ![1,3] = ![7/3] := by
  ext k
  fin_cases k
  norm_num [wlsCoefficient, gramWeighted, rhsWeighted, Matrix.inv_def,
    Matrix.det_fin_one, Matrix.adjugate_fin_one, Matrix.mulVec, dotProduct, Fin.sum_univ_succ]
