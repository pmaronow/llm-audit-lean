abbrev Row := Fin 1 × Fin 4

noncomputable def toyDesign : Design (Fin 6 × Fin 2) where
  weight := fun _ => 1/12
  nonneg := fun _ => by norm_num
  total := by norm_num [Fintype.sum_prod_type, Fin.sum_univ_succ]

def toyD (w : Fin 6 × Fin 2) (i : Row) : Bool :=
  decide (i.2 = (treatmentPair w.1).1 ∨ i.2 = (treatmentPair w.1).2)

noncomputable def toySign (j : Fin 2) : ℝ := if j = 0 then 1/4 else -1/4
noncomputable def toyC (w : Fin 6 × Fin 2) (i : Row) : ℝ :=
  1/2 + if i.2 = 0 then toySign w.2 else 0
noncomputable def toyR (w : Fin 6 × Fin 2) (i : Row) : ℝ :=
  (if toyD w i then (1 : ℝ) else 0) * (toyC w i - 1/2)
noncomputable def toyW (_ : Fin 1) (d : Row → Bool) (i : Row) : ℝ :=
  (if d i then (1 : ℝ) else 0) * (1/2)
noncomputable def toyY (w : Fin 6 × Fin 2) (i : Row) : ℝ :=
  toyW 0 (toyD w) i + toyR w i

example : ∀ w i, 0 ≤ toyC w i ∧ toyC w i ≤ 1 := by
  intro w i
  simp only [toyC, toySign]
  split_ifs <;> norm_num

example : ∀ w i, toyY w i = (if toyD w i then (1 : ℝ) else 0) * toyC w i := by
  intro w i
  unfold toyY toyW toyR
  ring

lemma toy_randomized : CompositionRandomized toyDesign toyD toyR := by
  intro f i
  unfold mean
  rw [Fintype.sum_prod_type]
  apply Finset.sum_eq_zero
  intro a ha
  have hD : toyD (a, (1 : Fin 2)) = toyD (a, 0) := rfl
  norm_num [toyDesign, toyR, toyC, toySign, Fin.sum_univ_succ, hD]

lemma toy_variation : mean toyDesign (fun w => centeredBlockCross (toyR w) (toyR w)) = 3/128 := by

example : 0 < mean toyDesign (fun w => centeredBlockCross (toyR w) (toyR w)) := by

-- Positive nonuniform weights and a nonempty, full-rank regression design.
def fitX : Matrix (Fin 2) (Fin 1) ℝ := fun _ _ => 1
noncomputable def fitW : Fin 2 → ℝ := ![1,2]
noncomputable def fitY : Fin 2 → ℝ := ![1,3]

lemma fit_rank : Function.Injective fitX.mulVec := by
  intro v v' h
  have h0 := congrFun h 0
  funext k
  fin_cases k
  simpa [Matrix.mulVec, dotProduct, fitX, Fin.sum_univ_succ] using h0

example : (∀ i, 0 < fitW i) ∧ Function.Injective fitX.mulVec ∧
    (gramWeighted fitW fitX).det = 3 ∧
    WLS fitW fitX fitY (fun _ => 7/3) ∧
    normalizedWeights fitW = ![1/3,2/3] := by
  refine ⟨?_, fit_rank, ?_, ?_, ?_⟩
  · intro i; fin_cases i <;> norm_num [fitW]
  · norm_num [gramWeighted, fitW, fitX, Fin.sum_univ_succ, Matrix.det_fin_one]
  · intro k; fin_cases k
    norm_num [fitW, fitX, fitY, Fin.sum_univ_succ]
  · ext i; fin_cases i <;>
      norm_num [normalizedWeights, fitW, Fin.sum_univ_succ]
