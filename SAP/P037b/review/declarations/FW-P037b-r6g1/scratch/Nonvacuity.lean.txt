/- T18: two independent Rademacher coordinates, n=2, k=1, an intercept
   design, nonzero residuals, sigma squared = 1. -/
noncomputable def testMu : Measure (Fin 4) := (4 : ℝ≥0∞)⁻¹ • Measure.count

instance : IsProbabilityMeasure testMu := by
  constructor
  norm_num [testMu, Measure.smul_apply]
  exact ENNReal.inv_mul_cancel (by norm_num) (by norm_num)

noncomputable def testV : Fin 4 → Fin 2 → ℝ :=
  ![![1,1], ![1,-1], ![-1,1], ![-1,-1]]

def interceptX : Matrix (Fin 2) (Fin 1) ℝ := fun _ _ => 1

lemma intercept_injective : Function.Injective interceptX.mulVec := by
  intro a b h
  funext j
  fin_cases j
  have hj := congrFun h (0 : Fin 2)
  simpa [interceptX, Matrix.mulVec, dotProduct, Fin.sum_univ_succ] using hj

lemma test_integral (f : Fin 4 → ℝ) :
    (∫ ω, f ω ∂testMu) = (∑ ω, f ω)/4 := by
  simp [testMu, integral_smul_measure, integral_count, div_eq_mul_inv, mul_comm]

example :
    Function.Injective interceptX.mulVec ∧ 0 < (2 : ℕ) ∧
    (∀ i, Integrable (fun ω => testV ω i) testMu) ∧
    (∀ i j, Integrable (fun ω => testV ω i * testV ω j) testMu) ∧
    vectorMean testMu testV = 0 ∧
    (∀ i j, (∫ ω, testV ω i * testV ω j ∂testMu) =
      if i = j then (1 : ℝ) else 0) ∧
    0 ≤ (1 : ℝ) ∧
    (∀ i : Fin 2, 0 ≤ (![0,1] : Fin 2 → ℝ) i ∧
      (![0,1] : Fin 2 → ℝ) i ≤ 1) ∧
    0 < ∑ i : Fin 2, ((![1,2] : Fin 2 → ℝ) i)^2 := by
  refine ⟨intercept_injective, by norm_num, ?_, ?_, ?_, ?_, by norm_num, ?_, ?_⟩
  · intro i
    exact Integrable.of_finite
  · intro i j
    exact Integrable.of_finite
  · funext i
    fin_cases i <;>
      norm_num [vectorMean, test_integral, testV, Fin.sum_univ_succ]
  · intro i j
    fin_cases i <;> fin_cases j <;>
      norm_num [test_integral, testV, Fin.sum_univ_succ]
  · intro i
    fin_cases i <;> norm_num
  · norm_num [Fin.sum_univ_succ]

/- T24: n=3, p=0, rows (1,0,0), (1,1,0), (1,1,1).
   This is an actual full-rank design of the stipulated form. -/
def recenterX : Matrix (Fin 3) (Fin 3) ℝ := fun i j =>
  if j.val = 0 then 1 else if j.val = 1 then
    (![0,1,1] : Fin 3 → ℝ) i else
    (![0,1,1] : Fin 3 → ℝ) i * (![0,0,1] : Fin 3 → ℝ) i

example : Function.Injective recenterX.mulVec := by
  intro a b h
  have h0 := congrFun h (0 : Fin 3)
  have h1 := congrFun h (1 : Fin 3)
  have h2 := congrFun h (2 : Fin 3)
  simp [recenterX, Matrix.mulVec, dotProduct, Fin.sum_univ_succ] at h0 h1 h2
  funext i
  fin_cases i <;> dsimp <;> linarith

example : Function.Injective (Matrix.mulVec
    ((fun i j => if h0 : j.val = 0 then 1 else if h1 : j.val = 1 then
      (![0,1,1] : Fin 3 → ℝ) i else if h2 : j.val = 2 then
      (![0,1,1] : Fin 3 → ℝ) i * (![0,0,1] : Fin 3 → ℝ) i
      else (fun (_ : Fin 3) (j : Fin 0) => Fin.elim0 j) i
        ⟨j.val-3, by omega⟩) : Matrix (Fin 3) (Fin (3+0)) ℝ)) := by
  have heq :
      ((fun i j => if h0 : j.val = 0 then 1 else if h1 : j.val = 1 then

        (![0,1,1] : Fin 3 → ℝ) i * (![0,0,1] : Fin 3 → ℝ) i

          ⟨j.val-3, by omega⟩) : Matrix (Fin 3) (Fin (3+0)) ℝ) = recenterX := by

  intro a b h
  have h0 := congrFun h (0 : Fin 3)
  have h1 := congrFun h (1 : Fin 3)
  have h2 := congrFun h (2 : Fin 3)
  simp [recenterX, Matrix.mulVec, dotProduct, Fin.sum_univ_succ] at h0 h1 h2

/- T20: a design with two equally likely assignments, two units, and both
   exposures positive. Fixed inclusion retains both units. -/
noncomputable def twoDesign : ProbabilityDesign (Fin 2) where
  mass := fun _ => 1/2
  positive := by intro z; norm_num
  total := by norm_num [Fin.sum_univ_succ]

def testD (z i : Fin 2) : Fin 2 := if i = 0 then z else 1-z

    0 < (Finset.univ.filter (fun _i : Fin 2 => true = true)).card := by

    fin_cases d <;> fin_cases i <;>
      norm_num [eventProbability, twoDesign, testD, Fin.sum_univ_succ]
