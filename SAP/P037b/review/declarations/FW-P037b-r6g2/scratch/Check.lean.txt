def w1Schedule (j : Fin 10000) : Fin 5 :=
  if j.val < 5000 then 0 else if j.val < 5005 then 1
  else if j.val < 7500 then 2 else 3

def w1Count (k : Fin 4) : ℕ :=
  (Finset.univ.filter (fun j : Fin 10000 => w1Schedule j = k.castSucc)).card

example : ∀ k : Fin 4, w1Count k = (![5000,5,2495,2500] : Fin 4 → ℕ) k := by
  native_decide

def w1Draw (j : Fin 10000) : Bool × Finset (Fin 12) :=
  if j.val < 5000 then (false, avoid12) else if j.val < 5005 then (true, avoid12)
  else if j.val < 7500 then (true, nonref12) else (true, referent12)

theorem w1Draw_supported (j : Fin 10000) : (w1Draw j).2 ∈ seedSubsets 12 6 := by
  unfold w1Draw
  split_ifs <;> norm_num [seedSubsets, avoid12, nonref12, referent12]

theorem w1Draw_labels (j : Fin 10000) :
    (if (w1Draw j).1 then localTreatedLabel ∅ {0,1,2,3,4,5} {0} (w1Draw j).2
      else 0) = w1Schedule j := by
  unfold w1Draw w1Schedule
  split_ifs <;> norm_num [localTreatedLabel, avoid12, nonref12, referent12,
    Finset.disjoint_left] at *

-- The complete T21 existential frequency witness, including its support clause.
example : ∃ z : Fin 10000 → Bool × Finset (Fin 12),
    (∀ j, (z j).2 ∈ seedSubsets 12 6) ∧
    (∀ k : Fin 4, ((Finset.univ.filter (fun j =>
      (if (z j).1 then localTreatedLabel ∅ {0,1,2,3,4,5} {0} (z j).2
        else 0) = k.castSucc)).card : ℝ) / 10000 =
      (![1/2,1/2000,499/2000,1/4] : Fin 4 → ℝ) k) := by
  refine ⟨w1Draw, w1Draw_supported, ?_⟩
  have counts : ∀ k : Fin 4, w1Count k = (![5000,5,2495,2500] : Fin 4 → ℕ) k := by
    native_decide
  intro k
  simp_rw [w1Draw_labels]
  change (w1Count k : ℝ) / 10000 = _
  rw [counts]
  fin_cases k <;> norm_num

-- A balanced schedule used with the actual legal four-block cutoff design.

  if j.val < 2500 then 0 else if j.val < 5000 then 1
  else if j.val < 7500 then 2 else 3

  (Finset.univ.filter (fun j : Fin 10000 => balancedSchedule j = k.castSucc)).card

theorem balancedCounts : ∀ k : Fin 4, balancedCount k = 2500 := by native_decide

def referentBlock12 : CutoffBlockW2 := ⟨referent12, by norm_num [referent12]⟩

  else if j.val < 5000 then (true, fun _ => avoidBlock12)
  else if j.val < 7500 then (true, fun _ => nonrefBlock12)

theorem balancedDraw_labels (j : Fin 10000) (i : Fin 2) :

-- Nonvacuity of every hypothesis of T31's general half-marginal identity,
-- using the actual independent two-of-four selections for eight schools.

def schools31 (z : Fin 6 × Fin 6) (i : Fin 8) : Bool :=
  if i.val < 4 then decide (Fin.ofNat 4 i.val ∈ treatmentPair z.1)
  else decide (Fin.ofNat 4 i.val ∈ treatmentPair z.2)

example : (∀ z, 0 ≤ mass31 z) ∧ (∑ z, mass31 z) = 1 ∧
    (∀ i, (∑ z, mass31 z * (if schools31 z i then (1 : ℝ) else 0)) = 1/2) ∧
    0 < (4 : ℕ) := by
  refine ⟨fun z => by norm_num [mass31], ?_, ?_, by norm_num⟩
  · norm_num [mass31, Fintype.sum_prod_type, Fin.sum_univ_succ]
  · intro i
    fin_cases i <;> norm_num [mass31, schools31, treatmentPair, Fin.ofNat,
      Fintype.sum_prod_type, Fin.sum_univ_succ]
