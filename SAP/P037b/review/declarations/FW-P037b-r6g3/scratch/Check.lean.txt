private def added (z : Bool) : ℕ := if z then 1 else 0

    (∀ z : Bool, 0 < (1/2 : ℝ)) ∧
    (∑ z : Bool, (1/2 : ℝ)) = 1 ∧
    (∀ z : Bool, (0 : ℕ) ≤ 0) ∧
    (∀ z : Bool, added z ≤ added z) ∧
    (∃ z : Bool, nonseedExposureLabel 0 0 = 0 ∧

    (∃ z : Bool, nonseedExposureLabel 0 0 ≠ 2 ∧

  norm_num [Fintype.sum_bool, added, nonseedExposureLabel]
