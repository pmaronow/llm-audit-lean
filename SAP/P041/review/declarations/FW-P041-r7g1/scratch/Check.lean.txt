lemma rates {q : ℝ} (hq : 0 < q) (hq1 : q < 1) :
    0 < positiveHazard q ∧ 0 < negativeHazard q ∧
    positiveHazard q / negativeHazard q = 2/3 := by
  have hl : Real.log q < 0 := Real.log_neg hq hq1
  have hn : Real.log q ≠ 0 := ne_of_lt hl
  constructor
  · dsimp [positiveHazard]
    exact div_pos (mul_pos_of_neg_of_neg (by norm_num) hl) (by norm_num)
  constructor
  · dsimp [negativeHazard]
    exact div_pos (mul_pos_of_neg_of_neg (by norm_num) hl) (by norm_num)
  · dsimp [positiveHazard, negativeHazard]
    field_simp

lemma contrast {q : ℝ} (hq : 0 < q) : twelveHourContrast q = q^2*(1-q) := by
  have ha : -(-2 * Real.log q / 12) * 12 = 2 * Real.log q := by ring
  have hn : -(-3 * Real.log q / 12) * 12 = 3 * Real.log q := by ring
  have he2 : Real.exp (2 * Real.log q) = q^2 := by
    simpa only [Nat.cast_ofNat, Real.exp_log hq] using Real.exp_nat_mul (Real.log q) 2
  have he3 : Real.exp (3 * Real.log q) = q^3 := by
    simpa only [Nat.cast_ofNat, Real.exp_log hq] using Real.exp_nat_mul (Real.log q) 3
  unfold twelveHourContrast exponentialSurvival positiveHazard negativeHazard
  rw [ha, hn, he2, he3]
  ring

lemma binomial_pos {n k : ℕ} {p : ℝ}
    (hk : k ≤ n) (hp : 0 < p) (hp1 : p < 1) :
    0 < binomialCountProbability n k p := by
  have hc : (0:ℝ) < (n.choose k : ℝ) := by exact_mod_cast Nat.choose_pos hk
  unfold binomialCountProbability
  exact mul_pos (mul_pos hc (pow_pos hp _)) (pow_pos (sub_pos.mpr hp1) _)
