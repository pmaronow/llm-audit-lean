-- Binary survivor patterns satisfying all twelve horizon hypotheses at once.
-- In each cell, earlier indexed facts survive; the thresholds decrease in h.
def pCut (h : Fin 4) (b : Bool) : Fin 3 → ℕ :=
  (if b then ![![20,50,35],![18,48,32],![16,44,30],![12,34,22]]
   else ![![17,41,30],![14,35,26],![12,32,24],![8,24,17]]) h
def pAlive (h : Fin 4) (i : Fin 300) : Bool :=
  decide (i.val % 50 < pCut h (pTreat i) (pStudy i))
def cAlive (h : Fin 4) (i : Fin 200) : Bool :=
  decide (i.val % 50 <
    (![![12,25,23,42],![9,24,17,39],![7,23,12,35],![4,14,7,20]] :
      Fin 4 → Fin 4 → ℕ) h (c200 i))
def fAlive (h : Fin 4) (i : Fin 151) : Bool :=
  decide ((if fTreat i then i.val else i.val - 75) <
    (if fTreat i then (![72,72,72,68] : Fin 4 → ℕ) h
     else (![75,75,75,73] : Fin 4 → ℕ) h))
def realBinary {n : ℕ} (p : Fin n → Bool) : Fin n → ℝ :=
  fun i => if p i then 1 else 0

lemma armSum_count {n : ℕ} (t p : Fin n → Bool) (b : Bool) :
    ArmTotalW1 t b (realBinary p) =
      ((Finset.univ.filter (fun i => t i = b ∧ p i = true)).card : ℝ) := by
  simp only [ArmTotalW1, realBinary, Finset.sum_boole, Finset.filter_filter]

lemma studySum_count {n r : ℕ} (v : Fin n → Fin r) (p : Fin n → Bool) (w : Fin r) :
    StudyTotalW1 v w (realBinary p) =
      ((Finset.univ.filter (fun i => v i = w ∧ p i = true)).card : ℝ) := by
  simp only [StudyTotalW1, realBinary, Finset.sum_boole, Finset.filter_filter]

example (h : Fin 4) :
    ArmTotalW1 pTreat true (realBinary (pAlive h)) = S11TreatedSurvivorsW1 0 h ∧
    ArmTotalW1 pTreat false (realBinary (pAlive h)) = S11ControlSurvivorsW1 0 h := by
  simp only [armSum_count]
  fin_cases h <;> norm_cast

example (h : Fin 4) :

example (h : Fin 4) :

example :
