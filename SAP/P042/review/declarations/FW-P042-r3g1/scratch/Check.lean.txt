abbrev G : Measure ℝ := gaussianReal 0 1
abbrev GP (ι : Type) [Fintype ι] : Measure (ι → ℝ) := Measure.pi (fun _ : ι => G)

lemma gLp : MemLp (fun x : ℝ => x) 2 G := memLp_id_gaussianReal 2
lemma gVar : variance (fun x : ℝ => x) G = 1 := by simp

lemma evalLp {ι : Type} [Fintype ι] (i : ι) :
    MemLp (fun x : ι → ℝ => x i) 2 (GP ι) := by
  exact gLp.comp_measurePreserving (measurePreserving_eval (fun _ : ι => G) i)

lemma evalVar {ι : Type} [Fintype ι] (i : ι) :
    variance (fun x : ι → ℝ => x i) (GP ι) = 1 := by
  simpa using (measurePreserving_eval (fun _ : ι => G) i).variance_fun_comp
    (f := fun x : ℝ => x) measurable_id.aemeasurable

lemma evalIndep {ι : Type} [Fintype ι] {i j : ι} (h : i ≠ j) :
    IndepFun (fun x : ι → ℝ => x i) (fun x : ι → ℝ => x j) (GP ι) := by
  exact (iIndepFun_pi (μ := fun _ : ι => G) (X := fun _ => id)
    (fun _ => measurable_id.aemeasurable)).indepFun h

lemma evalCov {ι : Type} [Fintype ι] {i j : ι} (h : i ≠ j) :
    covariance (fun x : ι → ℝ => x i) (fun x : ι → ℝ => x j) (GP ι) = 0 :=
  (evalIndep h).covariance_eq_zero (evalLp i) (evalLp j)

-- T4: real random variables with exactly the stipulated marginal variances and covariance 0.
def roboD (j : Fin 2) (x : Fin 2 → ℝ) : ℝ := SAP.P042.roboSE j * x j

example : ∃ D : Fin 2 → (Fin 2 → ℝ) → ℝ,
    (∀ j, MemLp (D j) 2 (GP (Fin 2))) ∧
    (∀ j, variance (D j) (GP (Fin 2)) = SAP.P042.roboVariances j) ∧
    covariance (D 0) (D 1) (GP (Fin 2)) = 0 := by
  refine ⟨roboD, ?_, ?_, ?_⟩
  · intro j; exact (evalLp j).const_mul _
  · intro j
    change variance (fun x : Fin 2 → ℝ => SAP.P042.roboSE j * x j) (GP (Fin 2)) = _
    rw [variance_const_mul, evalVar]
    simp [SAP.P042.roboVariances]
  · change covariance (fun x : Fin 2 → ℝ => SAP.P042.roboSE 0 * x 0)
      (fun x : Fin 2 → ℝ => SAP.P042.roboSE 1 * x 1) (GP (Fin 2)) = _
    rw [covariance_const_mul_left, covariance_const_mul_right,
      evalCov (by decide : (0 : Fin 2) ≠ 1)]
    simp

-- T5 witness 1: two copies of a N(0,1) variable. Every H1/H2 premise holds;
-- pooled actual variance is 1 and the diagonal formula is 1/2.
example : ∃ D : Fin 2 → ℝ → ℝ,
    (0 < (2:ℕ)) ∧ (∀ j, MemLp (D j) 2 G) ∧
    (∀ j, 0 < variance (D j) G) ∧
    variance (SAP.P042.pooled
      (SAP.P042.normalizedWeight (fun j => variance (D j) G)) D) G = 1 ∧
    (SAP.P042.totalWeight (fun j => variance (D j) G))⁻¹ = (1:ℝ)/2 := by
  refine ⟨fun _ x => x, by norm_num, fun _ => gLp, ?_, ?_, ?_⟩
  · intro j; simp
  · have hpool : SAP.P042.pooled
        (SAP.P042.normalizedWeight (fun _ : Fin 2 => variance (fun x : ℝ => x) G))
        (fun _ : Fin 2 => fun x : ℝ => x) = (fun x : ℝ => x) := by
      funext x
      simp [SAP.P042.pooled, SAP.P042.normalizedWeight, SAP.P042.totalWeight,
        SAP.P042.precisionWeight]
    rw [hpool]; simp
  · norm_num [SAP.P042.totalWeight, SAP.P042.precisionWeight, Fin.sum_univ_two]

-- T5 witness 2: nonempty disjoint singleton index pools with that same variable.
-- Their difference is identically zero; the asserted sum of diagonal variances is 2.
example : ∃ (N A : Finset (Fin 2)) (D : Fin 2 → ℝ → ℝ),
    N.Nonempty ∧ A.Nonempty ∧ Disjoint N A ∧
    (∀ j, MemLp (D j) 2 G) ∧ (∀ j, 0 < variance (D j) G) ∧
    variance (fun x =>
      SAP.P042.pooled (SAP.P042.subsetWeight N (fun j => variance (D j) G)) D x -
      SAP.P042.pooled (SAP.P042.subsetWeight A (fun j => variance (D j) G)) D x) G = 0 ∧
    (∑ j ∈ N, (variance (D j) G)⁻¹)⁻¹ +
      (∑ j ∈ A, (variance (D j) G)⁻¹)⁻¹ = (2:ℝ) := by
  refine ⟨{0}, {1}, fun _ x => x, by decide, by decide, by decide,
    fun _ => gLp, ?_, ?_, ?_⟩
  · intro j; simp
  · simp [SAP.P042.pooled, SAP.P042.subsetWeight]
    exact variance_zero G
  · norm_num
