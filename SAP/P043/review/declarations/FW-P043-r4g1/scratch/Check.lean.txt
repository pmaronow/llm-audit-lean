example {Ω : Type*} [Fintype Ω] (P : Population Ω) : FairAssignmentClaims P := by
  refine ⟨?_, ?_, ?_, ?_, ?_⟩
  · intro z
    cases z <;> norm_num [armProb]
  · intro z i
    have hw := P.nonneg i
    cases z <;> dsimp [jointMass, armProb] <;> positivity
  · simp only [jointMass, ← Finset.mul_sum, P.total, mul_one]
    norm_num [armProb, Fintype.sum_bool]
  · intro z
    simp only [jointMass, ← Finset.mul_sum, P.total, mul_one]
    cases z <;> norm_num [armProb]
  · intro A z
    simp only [jointMass, eventProb, mean, mul_assoc, Finset.mul_sum]

-- T2: an inhabited probability space with all five finite expectations,
-- nondegenerate assignment, and nonzero mediated/direct effects.

  unfold auditProb
  infer_instance

def auditP : unitInterval := ⟨1/2, by constructor <;> norm_num⟩

-- T5: all outer hypotheses together, with two positive crossed margins.
def auditIV : IVModel Bool :=
  ⟨fun z i => if z then (if i then 10 else 9) else 8,
   fun k i => if i then 2 + 2 * indicator (9 ≤ k) + indicator (10 ≤ k)
     else 3 + indicator (9 ≤ k)⟩

example : (0 < (1/2 : ℝ) ∧ (1/2 : ℝ) < 1) ∧
    (∀ i, auditIV.knowledge false i ≤ 10 ∧ auditIV.knowledge true i ≤ 10) ∧
    (∀ i, auditIV.knowledge false i ≤ auditIV.knowledge true i) ∧
    0 < mean (uniformPopulation Bool)
      (fun i => (auditIV.knowledge true i : ℝ) - auditIV.knowledge false i) := by
  norm_num [auditIV, mean, uniformPopulation, Fintype.sum_bool]

-- T6: inhabited observations and nonreference/reference strata, and q=0.
example : let s : Fin 2 → Fin 2 := id
    let γ : Fin 1 → ℝ := fun _ => 3
    stratumEffect s γ 0 = 3 ∧ stratumEffect s γ 1 = 0 ∧
    (∑ k : Fin 1, stratumDummy s 0 k) = 1 ∧
    (∑ k : Fin 1, stratumDummy s 1 k) = 0 := by
  dsimp only
  simp only [stratumEffect, Fin.sum_univ_one]
  norm_num [stratumDummy]

example (i : Fin 1) :
    stratumEffect (fun _ : Fin 1 => (0 : Fin 1)) (fun k : Fin 0 => Fin.elim0 k) i = 0 ∧
    stratumCoefficient (fun k : Fin 0 => Fin.elim0 k) (0 : Fin 1) = 0 := by
  constructor
  · simp [stratumEffect]
  · change stratumCoefficient (fun k : Fin 0 => Fin.elim0 k) (Fin.last 0) = 0
    rw [stratumCoefficient, Fin.lastCases_last]
