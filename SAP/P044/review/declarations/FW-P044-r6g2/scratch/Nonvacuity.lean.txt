noncomputable def reviewHalf : unitInterval := ⟨1/2, by constructor <;> norm_num⟩

-- A genuine nondegenerate random-bank model, R=3 and p=1/2, with independence.
example : ∃ (μ : Measure (Bool × ℕ)) (_hμ : IsProbabilityMeasure μ)
    (D : Bool × ℕ → Bool) (C : Bool × ℕ → ℕ),
    HasLaw D (bernoulliMeasure true false reviewHalf) μ ∧
    HasLaw C (binomial 3 reviewHalf) μ ∧ IndepFun D C μ ∧
    0 < (reviewHalf : ℝ) ∧ (reviewHalf : ℝ) < 1 := by
  let μ := (bernoulliMeasure true false reviewHalf).prod (binomial 3 reviewHalf)
  refine ⟨μ, inferInstance, Prod.fst, Prod.snd, ?_, ?_, ?_, ?_, ?_⟩
  · exact ⟨measurable_fst.aemeasurable, by simp [μ]⟩
  · exact ⟨measurable_snd.aemeasurable, by simp [μ]⟩
  · exact indepFun_prod (μ := bernoulliMeasure true false reviewHalf)
      (ν := binomial 3 reviewHalf) measurable_id measurable_id
  · norm_num [reviewHalf]
  · norm_num [reviewHalf]
