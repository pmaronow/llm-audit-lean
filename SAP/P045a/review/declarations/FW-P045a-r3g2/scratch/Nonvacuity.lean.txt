def uniform2 : Design (Assignment 2) where
  mass := fun _ => 1/4
  nonneg := by intro z; norm_num
  total := by norm_num [Assignment, Fintype.card_fun]

def edgeNetwork : Network 2 where
  adj := fun i j => decide (i ≠ j)
  symmetric := by intro i j; simp [ne_comm]
  loopless := by intro i; simp

lemma prob_pos_at {Ω : Type} [Fintype Ω] (p : Design Ω)
    (E : Ω → Prop) (w : Ω) (hw : E w) (hp : 0 < p.mass w) :
    0 < prob p.mass E := by
  classical
  unfold prob SAP.P045a.expect
  apply Finset.sum_pos'
  · intro v hv
    exact mul_nonneg (p.nonneg v) (by dsimp; split_ifs <;> norm_num)
  · exact ⟨w, mem_univ w, by simpa [hw] using hp⟩

-- All assumptions of the universal population-expectation part of T6,
-- with a genuinely nonconstant outcome schedule (the all-vs-none target is 2).
example : 0 < (2 : ℕ) ∧
    (∀ i z, allTreated edgeNetwork.adj z i →
      linear2 i z = linear2 i (fun _ => true)) ∧
    (∀ i z, allControl edgeNetwork.adj z i →
      linear2 i z = linear2 i (fun _ => false)) ∧
    (∀ i, 0 < prob uniform2.mass (fun z => allTreated edgeNetwork.adj z i)) ∧
    (∀ i, 0 < prob uniform2.mass (fun z => allControl edgeNetwork.adj z i)) ∧
    allVsNone linear2 = 2 := by
  have hT : ∀ i z, allTreated edgeNetwork.adj z i →
      linear2 i z = linear2 i (fun _ => true) := by
    intro i z h
    fin_cases i <;> cases h0 : z 0 <;> cases h1 : z 1 <;>
      simp [allTreated, neighborhoodCount, neighborhood, edgeNetwork,
        Fin.univ_succ, Finset.filter_insert, Finset.filter_singleton, h0, h1] at h <;>
      norm_num [h0, h1, linear2, bit]
  have hC : ∀ i z, allControl edgeNetwork.adj z i →
      linear2 i z = linear2 i (fun _ => false) := by
    intro i z h
    fin_cases i <;> cases h0 : z 0 <;> cases h1 : z 1 <;>
      simp [allControl, neighborhoodCount, neighborhood, edgeNetwork,
        Fin.univ_succ, Finset.filter_insert, Finset.filter_singleton, h0, h1] at h <;>
      norm_num [h0, h1, linear2, bit]
  refine ⟨by decide, hT, hC, ?_, ?_, ?_⟩
  · intro i
    apply prob_pos_at uniform2 _ (fun _ => true)
    · fin_cases i <;> norm_num [allTreated, neighborhoodCount, neighborhood,
        edgeNetwork, Fin.univ_succ]
    · norm_num [uniform2]
  · intro i
    apply prob_pos_at uniform2 _ (fun _ => false)
    · fin_cases i <;> norm_num [allControl, neighborhoodCount, neighborhood,
        edgeNetwork, Fin.univ_succ]
    · norm_num [uniform2]
  · norm_num [allVsNone, mean, linear2, bit, Fin.sum_univ_two]

-- All hypotheses of the general finite-event identity in T6.
example : ∃ p : Design Bool, ∃ E : Bool → Prop, ∃ Y : Bool → ℝ,
    (∃ _ : DecidablePred E, 0 < prob p.mass E) ∧ Y false ≠ Y true := by
  let p : Design Bool := ⟨fun _ => 1/2, by intro b; norm_num, by norm_num⟩
  refine ⟨p, (fun b => b = true), (fun b => if b then 3 else 1), ?_, by norm_num⟩
  refine ⟨inferInstance, ?_⟩
  apply prob_pos_at p (fun b => b = true) true rfl
  norm_num [p]

-- All premises of the matrix-bounds part of T9, including deletion and
-- addition antecedents, on actual two-by-two binary data.
def numericEdge (i j : ℕ) : ℝ := if i < 2 ∧ j < 2 ∧ i ≠ j then 1 else 0

example : (∀ i j, numericEdge i j = numericEdge j i) ∧
    (∀ i, numericEdge i i = 0) ∧
    (∀ i j, i < 2 → j < 2 → numericEdge i j = 0 ∨ numericEdge i j = 1) ∧
    (∀ i j, i < j → j < 2 → numericEdge i j = 0 ∨ numericEdge i j = 1) ∧
    (∀ i j, i < j → j < 2 → (0 : ℝ) ≤ numericEdge i j) ∧
    (∀ i j, i < j → j < 2 → numericEdge i j ≤ numericEdge i j) := by
  refine ⟨?_, ?_, ?_, ?_, ?_, ?_⟩
  · intro i j; simp [numericEdge, and_left_comm, and_comm, and_assoc, ne_comm]
  · intro i; simp [numericEdge]
  · intro i j hi hj; unfold numericEdge; split_ifs <;> simp
  · intro i j hij hj; unfold numericEdge; split_ifs <;> simp
  · intro i j hij hj; unfold numericEdge; split_ifs <;> norm_num
  · intro i j hij hj; exact le_rfl

def spilloverOut (_i : Fin 2) (d : Exposure) : ℝ :=
  match d with | .d11 => 2 | .d10 => 1 | .d01 => 1 | .d00 => 0

-- All hypotheses of the exposure and HT parts of T9. Taking S=T is allowed
-- and still gives strictly positive measured d00 and d01 probabilities and
-- strictly positive spillovers; no conditioning event is null.
example : 0 < (2 : ℕ) ∧
    (∀ i j, edgeNetwork.adj i j = true → edgeNetwork.adj i j = true) ∧
    (∀ i, 0 < exposureProb uniform2.mass (exposure edgeNetwork.adj) i .d00) ∧
    (∀ i, 0 < exposureProb uniform2.mass (exposure edgeNetwork.adj) i .d01) ∧
    (∀ i, spilloverOut i .d00 < spilloverOut i .d01) := by
  refine ⟨by decide, by intros; assumption, ?_, ?_, by intro i; norm_num [spilloverOut]⟩
  · intro i
    unfold exposureProb
    apply prob_pos_at uniform2 _ (fun _ => false)
    · simp [exposure]
    · norm_num [uniform2]
  · intro i
    unfold exposureProb
    apply prob_pos_at uniform2 _ (fun j => decide (j ≠ i))
    · fin_cases i <;> norm_num [exposure, edgeNetwork, Fin.exists_fin_two]
    · norm_num [uniform2]

-- T7 has no initial hypotheses. Its supplied single-edge witness is genuine.

  norm_num [singleEdge4, faultyCopy, hu, hl, List.idxOf_cons]
