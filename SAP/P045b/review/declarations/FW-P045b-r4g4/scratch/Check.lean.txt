import SAP.P045b.Defs

namespace ReviewT33
open SAP.P045b
open scoped BigOperators
noncomputable section

-- The two equiprobable assignments are (false,true) and (true,false).
def P : Design Bool where
  mass := fun _ => 1 / 2
  nonneg := by intro ω; norm_num
  total := by norm_num [Fintype.sum_bool]

def f (ω : Bool) (i : Fin 2) : Bool := if i = 0 then ω else !ω

def y (i : Fin 2) (d : Bool) : ℝ :=
  if i = 0 then (if d then 3 else 1) else (if d then 5 else 2)

-- Every substantive antecedent, together with Design's two structure proofs.
example :
    (∀ ω, 0 ≤ P.mass ω) ∧ (∑ ω, P.mass ω) = 1 ∧
    0 < (2 : ℕ) ∧ (false : Bool) ≠ true ∧
    (∀ i, 0 < exposureMarginal P.mass f i false) ∧
    (∀ i, 0 < exposureMarginal P.mass f i true) := by
  refine ⟨P.nonneg, P.total, by decide, by decide, ?_, ?_⟩ <;>
    intro i <;> fin_cases i <;>
    norm_num [exposureMarginal, P, f, Fintype.sum_bool]

-- Positive exact marginals, positive cross-unit cross-label joints,
-- zero cross-unit same-label joints, and zero same-unit cross-label joints.
example :
    (∀ (i : Fin 2) (d : Bool), exposureMarginal P.mass f i d = 1 / 2) ∧
    exposureJoint P.mass f 0 1 false false = 0 ∧
    exposureJoint P.mass f 0 1 true true = 0 ∧
    exposureJoint P.mass f 0 1 false true = 1 / 2 ∧
    (∀ i : Fin 2, exposureJoint P.mass f i i false true = 0) := by
  refine ⟨?_, ?_, ?_, ?_, ?_⟩
  · intro i d
    fin_cases i <;> cases d <;>
      norm_num [exposureMarginal, P, f, Fintype.sum_bool]
  · norm_num [exposureJoint, P, f, Fintype.sum_bool]
  · norm_num [exposureJoint, P, f, Fintype.sum_bool]
  · norm_num [exposureJoint, P, f, Fintype.sum_bool]
  · intro i
    fin_cases i <;> norm_num [exposureJoint, P, f, Fintype.sum_bool]

-- Direct arithmetic checks of the definitions, without using T33 or a Solution.
example :
    weightedContrast f y (exposureMarginal P.mass f) false true false = -4 ∧
    weightedContrast f y (exposureMarginal P.mass f) false true true = -1 ∧
    expect P.mass (weightedContrast f y (exposureMarginal P.mass f) false true) = -5 / 2 ∧
    exposureEffect y false true = -5 / 2 ∧
    momentVariance P.mass (weightedContrast f y (exposureMarginal P.mass f) false true) = 9 / 4 := by
  norm_num [weightedContrast, weightedMean, weightedTerm, exposureMarginal,
    exposureEffect, momentVariance, expect, P, f, y, Fintype.sum_bool, Fin.sum_univ_two]

example :
    varianceEstimate f y (exposureMarginal P.mass f) (exposureJoint P.mass f) false true false = 34 ∧
    varianceEstimate f y (exposureMarginal P.mass f) (exposureJoint P.mass f) false true true = 27 / 2 ∧
    expect P.mass (varianceEstimate f y (exposureMarginal P.mass f)
      (exposureJoint P.mass f) false true) = 95 / 4 ∧
    zeroSurplus y (exposureJoint P.mass f) false true = 43 / 2 := by
  norm_num [varianceEstimate, totalVarianceEstimate, totalCovarianceEstimate,
    zeroSurplus, exposureMarginal, exposureJoint, expect, P, f, y,
    Fintype.sum_bool, Fin.sum_univ_two]

-- A signed schedule on the same design has structural zeros but zero surplus.
def signedY (i : Fin 2) (_ : Bool) : ℝ := if i = 0 then 1 else -1

example :
    zeroSurplus signedY (exposureJoint P.mass f) false true = 0 ∧
    momentVariance P.mass
      (weightedContrast f signedY (exposureMarginal P.mass f) false true) = 4 ∧
    expect P.mass (varianceEstimate f signedY (exposureMarginal P.mass f)
      (exposureJoint P.mass f) false true) = 4 := by
  norm_num [zeroSurplus, momentVariance, weightedContrast, weightedMean, weightedTerm,
    varianceEstimate, totalVarianceEstimate, totalCovarianceEstimate,
    exposureMarginal, exposureJoint, expect, P, f, signedY,
    Fintype.sum_bool, Fin.sum_univ_two]

end
end ReviewT33
