-- T5: a genuine balanced learner with the singleton zero family. No T5 result is used.
def auditZeroFits (n : ℕ) : CrossFit n where
  fold := (RegressionExample.fits n).fold
  g _ _ _ := 0
  measurable_g _ := measurable_const
  opposite_fold _ _ _ _ := rfl

def auditZeroFamily (_ : Fin 0 → ℝ) (_ : ℝ) : ℝ := 0

lemma auditZeroBalanced : Balanced auditZeroFits := by
  exact (SAP.P046.T4.1.2.2.2.1 : SAP.P046.Balanced RegressionExample.fits)

lemma auditZeroConvergence (M : Study) (j : Bool) :
    WholeL2InProbability M auditZeroFits j (fun _ => 0) := by
  intro eps heps
  have hpos : 0 < ENNReal.ofReal eps := ENNReal.ofReal_pos.mpr heps
  simpa [WholeL2InProbability, wholeL2Sq, auditZeroFits, hpos.ne'] using
    (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))

-- Every hypothesis of WholeT5Family, with a specific valid continuous law.
example :
    ValidStudy ConditionalExample.study ∧
    Balanced auditZeroFits ∧
    (∀ theta, MemLp (fun o => auditZeroFamily theta (W o)) 2 ConditionalExample.study.law) ∧
    ZeroBenchmark ConditionalExample.study (Set.range auditZeroFamily) ∧
    (∀ n j, ∀ᵐ a ∂sampleLaw ConditionalExample.study n, ∃ theta,
      (auditZeroFits n).g j a = auditZeroFamily theta ∧ ∀ eta,
      empiricalRisk ConditionalExample.study
        (univ.filter (fun i => (auditZeroFits n).fold i ≠ j)) a (auditZeroFamily theta) ≤
      empiricalRisk ConditionalExample.study
        (univ.filter (fun i => (auditZeroFits n).fold i ≠ j)) a (auditZeroFamily eta)) ∧
    (fun _ : ℝ => (0 : ℝ)) ∈ Set.range auditZeroFamily ∧
    (∀ theta, risk ConditionalExample.study (fun _ => 0) ≤
      risk ConditionalExample.study (auditZeroFamily theta)) ∧
    (∀ j, WholeL2InProbability ConditionalExample.study auditZeroFits j (fun _ => 0)) := by
  refine ⟨SAP.P046.T2.1.1, auditZeroBalanced, ?_, ?_, ?_, ?_, ?_, ?_⟩
  · intro theta
    exact MemLp.zero
  · exact Or.inl ⟨(fun i => Fin.elim0 i), rfl⟩
  · intro n j
    exact Filter.Eventually.of_forall (fun a =>
      ⟨(fun i => Fin.elim0 i), rfl, fun eta => le_rfl⟩)
  · exact ⟨(fun i => Fin.elim0 i), rfl⟩
  · intro theta
    exact le_rfl
  · exact auditZeroConvergence ConditionalExample.study
