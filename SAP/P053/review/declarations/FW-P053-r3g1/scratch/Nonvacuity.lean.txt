-- T1: two positive arms, a nonconstant binary outcome vector, and a
-- nonconstant symmetric arm-sample statistic with a real codomain.
def symmetricStatistic (x1 x0 : Fin 2 → Bool) : ℝ :=
  (∑ i, binaryReal (x1 i)) - (∑ i, binaryReal (x0 i))

example : 0 < (2 : ℕ) ∧ 0 < (2 : ℕ) ∧

example (x1 x0 : Fin 2 → Bool) (p1 p0 : Equiv.Perm (Fin 2)) :
    symmetricStatistic (x1 ∘ p1) (x0 ∘ p0) = symmetricStatistic x1 x0 := by
  unfold symmetricStatistic
  change (∑ i, binaryReal (x1 (p1 i))) - (∑ i, binaryReal (x0 (p0 i))) = _
  rw [Equiv.sum_comp p1 (fun i => binaryReal (x1 i)),
    Equiv.sum_comp p0 (fun i => binaryReal (x0 i))]

example : symmetricStatistic (fun _ => true) (fun _ => false) = 2 ∧
    symmetricStatistic (fun _ => false) (fun _ => true) = -2 := by
  norm_num [symmetricStatistic, binaryReal]

-- T5: construct every S5 hypothesis independently. The effect is nonzero.
def locationFamily (k : ℕ) : Population where
  n1 := k+2
  n0 := k+2
  Y1 := fun _ => 1
  Y0 := fun _ => 0

lemma location_size_positive (k : ℕ) : 0 < (locationFamily k).size := by
  simp [locationFamily, Population.size]

lemma populationMean_const (P : Population) (h : 0 < P.size) (c : ℝ) :
    populationMean P (fun _ => c) = c := by
  have hn : (P.size : ℝ) ≠ 0 := by exact_mod_cast (Nat.ne_of_gt h)
  simp only [populationMean, sampleMean, Finset.sum_const, nsmul_eq_mul,
    Finset.card_univ, Fintype.card_fin]
  exact mul_div_cancel_left₀ c hn

lemma populationVariance_const (P : Population) (h : 0 < P.size) (c : ℝ) :
    populationVariance P (fun _ => c) = 0 := by
  have hm : sampleMean (Finset.univ : Finset (Fin P.size)) (fun _ => c) = c :=
    populationMean_const P h c
  simp [populationVariance, sampleVariance, centeredSS, hm]

lemma location_regularity : Regularity locationFamily (1/2) 0 0 0 1 := by
  constructor
  · exact tendsto_atTop_mono (fun k => by simp [locationFamily, Population.size]; omega)
      tendsto_id
  · exact Eventually.of_forall (fun k => by simp [locationFamily])
  · norm_num
  · norm_num
  · have heq : (fun k : ℕ => ((locationFamily k).n1 : ℝ)/(locationFamily k).size) =
        fun _ => (1/2 : ℝ) := by
      funext k
      have hn : (k : ℝ)+2 ≠ 0 := by positivity
      simp only [locationFamily, Population.size, Nat.cast_add, Nat.cast_ofNat]
      field_simp; ring
    rw [heq]
    exact tendsto_const_nhds
  · have heq : (fun k => populationVariance (locationFamily k) (locationFamily k).Y1) =
        fun _ => (0 : ℝ) := by
      funext k
      exact populationVariance_const _ (location_size_positive k) 1
    rw [heq]
    exact tendsto_const_nhds
  · have heq : (fun k => populationVariance (locationFamily k) (locationFamily k).Y0) =
        fun _ => (0 : ℝ) := by
      funext k
      exact populationVariance_const _ (location_size_positive k) 0
    rw [heq]
    exact tendsto_const_nhds
  · have heq : (fun k => populationVariance (locationFamily k) (unitEffects (locationFamily k))) =
        fun _ => (0 : ℝ) := by
      funext k
      change populationVariance (locationFamily k) (fun _ => (1 : ℝ)-0) = 0
      simpa using populationVariance_const (locationFamily k) (location_size_positive k) 1
    rw [heq]
    exact tendsto_const_nhds
  · have heq : (fun k => averageEffect (locationFamily k)) = fun _ => (1 : ℝ) := by
      funext k
      unfold averageEffect
      change populationMean (locationFamily k) (fun _ => (1 : ℝ)) -
        populationMean (locationFamily k) (fun _ => (0 : ℝ)) = 1
      rw [populationMean_const _ (location_size_positive k) 1,
        populationMean_const _ (location_size_positive k) 0]
      norm_num
    rw [heq]
    exact tendsto_const_nhds
  · refine ⟨1, by norm_num, ?_⟩
    intro k
    have hn : ((locationFamily k).size : ℝ) ≠ 0 := by
      exact_mod_cast (Nat.ne_of_gt (location_size_positive k))
    dsimp only [locationFamily]
    norm_num only [one_pow, zero_pow (by decide : 4 ≠ 0), add_zero,
      Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul, mul_one]
    change ((Finset.univ : Finset (Fin ((k+2)+(k+2)))).card : ℝ) /
      (((k+2)+(k+2) : ℕ) : ℝ) ≤ 1
    have hc : (Finset.univ : Finset (Fin ((k+2)+(k+2)))).card = (k+2)+(k+2) := by simp
    rw [hc]
    exact le_of_eq (div_self hn)

example : ∃ E : ℕ → Population, Regularity E (1/2) 0 0 0 1 :=
  ⟨locationFamily, location_regularity⟩
