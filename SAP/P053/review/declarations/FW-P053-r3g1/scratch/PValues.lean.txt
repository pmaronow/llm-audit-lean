noncomputable section

lemma success_bound {N n : ℕ} (y : Fin N → Bool) (z : Finset (Fin N))
    (hz : z ∈ support n) : successes y z ≤ n := by
  have hc : z.card = n := (Finset.mem_powersetCard.mp hz).2
  calc
    successes y z ≤ z.card := Finset.card_filter_le _ _
    _ = n := hc

lemma six_top_tau : binaryTau 3 3 sixY sixTop = 1 := by
  have ha : successes sixY sixTop = 3 := by decide
  have hM : totalSuccesses sixY = 3 := by decide
  norm_num [binaryTau, ha, hM]

example : SixInference := by
  classical
  have hM : totalSuccesses sixY = 3 := by decide
  have hc : (support (ι := Fin 6) 3).card = 20 := by decide
  have ht : (support (ι := Fin 6) 3).filter
        (fun w => binaryTau 3 3 sixY sixTop ≤ binaryTau 3 3 sixY w) =
      (support (ι := Fin 6) 3).filter (fun w => successes sixY w = 3) := by
    ext w
    by_cases hw : w ∈ support (ι := Fin 6) 3
    · have hb := success_bound sixY w hw
      simp only [Finset.mem_filter, hw, true_and, six_top_tau]
      interval_cases ha : successes sixY w <;> norm_num [binaryTau, hM, ha]
    · simp [hw]
  have hta : (support (ι := Fin 6) 3).filter
        (fun w => |binaryTau 3 3 sixY sixTop| ≤ |binaryTau 3 3 sixY w|) =
      (support (ι := Fin 6) 3).filter
        (fun w => successes sixY w = 0 ∨ successes sixY w = 3) := by
    ext w
    by_cases hw : w ∈ support (ι := Fin 6) 3
    · have hb := success_bound sixY w hw
      simp only [Finset.mem_filter, hw, true_and, six_top_tau]
      interval_cases ha : successes sixY w <;> norm_num [binaryTau, hM, ha]
    · simp [hw]
  have hct : ((support (ι := Fin 6) 3).filter (fun w => successes sixY w = 3)).card = 1 :=
    by decide
  have hcta : ((support (ι := Fin 6) 3).filter
      (fun w => successes sixY w = 0 ∨ successes sixY w = 3)).card = 2 := by decide
  have hp : upperP 3 (binaryTau 3 3 sixY) sixTop = 1/20 := by
    unfold upperP
    rw [ht, hct, hc]
    norm_num
  have hpa : upperP 3 (fun w => |binaryTau 3 3 sixY w|) sixTop = 1/10 := by
    unfold upperP
    rw [hta, hcta, hc]
    norm_num
  have htop : sixTop ∈ support 3 := by decide
  norm_num [SixInference, hc, htop, six_top_tau, hp, hpa]

-- On four units, check each possible observed count against all reference
-- counts. Every reference assignment is present in the original support.
lemma four_tau (z : Finset (Fin 4)) :
    binaryTau 2 2 fourY z = (successes fourY z : ℝ)-1 := by
  have hM : totalSuccesses fourY = 2 := by decide
  unfold binaryTau
  rw [hM]
  push_cast
  ring

lemma four_pvalues (z : Finset (Fin 4)) (hz : z ∈ support 2) :
    upperP 2 (binaryTau 2 2 fourY) z =
      (if successes fourY z = 2 then 1/6 else if successes fourY z = 1 then 5/6 else 1) ∧
    upperP 2 (fun w => |binaryTau 2 2 fourY w|) z =
      (if successes fourY z = 1 then 1 else 1/3) := by
  classical
  have hc : (support (ι := Fin 4) 2).card = 6 := by decide
  have ht : (support (ι := Fin 4) 2).filter
        (fun w => binaryTau 2 2 fourY z ≤ binaryTau 2 2 fourY w) =
      (support (ι := Fin 4) 2).filter (fun w => successes fourY z ≤ successes fourY w) := by
    ext w
    simp [four_tau]
  have hta : (support (ι := Fin 4) 2).filter
        (fun w => |binaryTau 2 2 fourY z| ≤ |binaryTau 2 2 fourY w|) =
      (support (ι := Fin 4) 2).filter
        (fun w => successes fourY z = 1 ∨ successes fourY w = 0 ∨ successes fourY w = 2) := by
    ext w
    by_cases hw : w ∈ support (ι := Fin 4) 2
    · have hbz := success_bound fourY z hz
      have hbw := success_bound fourY w hw
      simp only [Finset.mem_filter, hw, true_and]
      interval_cases ha : successes fourY z <;>
        interval_cases hb : successes fourY w <;> norm_num [four_tau, ha, hb]
    · simp [hw]
  unfold upperP
  rw [ht, hta, hc]
  have hbz := success_bound fourY z hz
  interval_cases ha : successes fourY z
  all_goals norm_num only [ha, OfNat.zero_ne_ofNat, OfNat.one_ne_ofNat,
    OfNat.ofNat_ne_one, ite_false, ite_true, true_or, false_or]
  all_goals norm_num [show ((support (ι := Fin 4) 2).filter
    (fun w => 0 ≤ successes fourY w)).card = 6 by decide,
    show ((support (ι := Fin 4) 2).filter
      (fun w => 1 ≤ successes fourY w)).card = 5 by decide,
    show ((support (ι := Fin 4) 2).filter
      (fun w => 2 ≤ successes fourY w)).card = 1 by decide,
    show ((support (ι := Fin 4) 2).filter
      (fun w => successes fourY w = 0 ∨ successes fourY w = 2)).card = 2 by decide,
    hc]

example : FourInference := by
  have htop : fourTop ∈ support 2 := by decide
  have hbottom : fourBottom ∈ support 2 := by decide
  have hat : successes fourY fourTop = 2 := by decide
  have hab : successes fourY fourBottom = 0 := by decide
  have hpt := four_pvalues fourTop htop
  have hpb := four_pvalues fourBottom hbottom
  norm_num [hat, hab] at hpt hpb
  refine ⟨htop, hbottom, ?_, ?_, hpt.1, hpt.2, hpb.1, hpb.2, ?_⟩
  · norm_num [four_tau, hat]
  · norm_num [four_tau, hab]
  · intro z hz hzt hzb
    have ha : successes fourY z = 1 := by
      have h : ∀ w ∈ support (ι := Fin 4) 2,
          w ≠ fourTop → w ≠ fourBottom → successes fourY w = 1 := by decide
      exact h z hz hzt hzb
    have hp := four_pvalues z hz
    norm_num [ha] at hp
    exact ⟨by norm_num [four_tau, ha], hp⟩

end
end AuditPValues
