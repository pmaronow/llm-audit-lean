def zeroFamily (k : ℕ) : Population where
  n1 := k + 2
  n0 := k + 2
  Y1 := fun _ => 0
  Y0 := fun _ => 0

example : Regularity zeroFamily (1/2) 0 0 0 0 := by
  have hsize : Tendsto (fun k => (zeroFamily k).size) atTop atTop := by
    exact tendsto_atTop_mono (fun k => by simp [zeroFamily, Population.size]; omega)
      tendsto_id
  refine ⟨hsize, Filter.Eventually.of_forall (fun k => by simp [zeroFamily]),
    by norm_num, by norm_num, ?_, ?_, ?_, ?_, ?_, ?_⟩
  · have heq : (fun k => ((zeroFamily k).n1 : ℝ)/(zeroFamily k).size) =
        (fun _ : ℕ => (1/2 : ℝ)) := by
      funext k
      dsimp [zeroFamily, Population.size]
      push_cast
      have hk : (0 : ℝ) < k + 2 := by positivity
      field_simp
      <;> ring
    rw [heq]
    exact tendsto_const_nhds
  · simpa [populationVariance, sampleVariance, centeredSS, sampleMean, zeroFamily]
      using (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))
  · simpa [populationVariance, sampleVariance, centeredSS, sampleMean, zeroFamily]
      using (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))
  · simpa [populationVariance, sampleVariance, centeredSS, sampleMean, unitEffects, zeroFamily]
      using (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))
  · simpa [averageEffect, populationMean, sampleMean, zeroFamily]
      using (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))
  · exact ⟨0, le_rfl, fun k => by simp [zeroFamily]⟩

lemma lower_card (a b : ℕ) (hab : a ≤ b) :
    ((Finset.univ : Finset (Fin b)).filter (fun i => i.val < a)).card = a := by
  classical
  calc
    _ = (Finset.univ : Finset (Fin a)).card :=
      Finset.card_bij (fun i hi => ⟨i.val, (Finset.mem_filter.mp hi).2⟩)
        (by intros; exact Finset.mem_univ _)
        (by intro x hx y hy h; apply Fin.ext; exact congrArg (fun t : Fin a => t.val) h)
        (by
          intro j _
          refine ⟨⟨j.val, lt_of_lt_of_le j.isLt hab⟩, ?_, ?_⟩
          · exact Finset.mem_filter.mpr ⟨Finset.mem_univ _, j.isLt⟩
          · rfl)
    _ = a := by simp

def binaryY (m : ℕ) (i : Fin (m+3*m)) : ℝ := if i.val < 2*m then 1 else 0
def sharpFamily (k : ℕ) : Population where
  n1 := k+2
  n0 := 3*(k+2)
  Y1 := binaryY (k+2)
  Y0 := binaryY (k+2)

lemma binary_mean (m : ℕ) (hm : 0 < m) :
    sampleMean Finset.univ (binaryY m) = 1/2 := by
  unfold sampleMean binaryY
  rw [Finset.sum_boole, lower_card (2*m) (m+3*m) (by omega)]
  simp only [Finset.card_univ, Fintype.card_fin]
  push_cast
  have hmR : (0 : ℝ) < m := by exact_mod_cast hm
  field_simp
  <;> ring

lemma binary_variance (m : ℕ) (hm : 0 < m) :
    sampleVariance Finset.univ (binaryY m) =
      ((m+3*m : ℕ) : ℝ)/(4*(((m+3*m : ℕ) : ℝ)-1)) := by
  unfold sampleVariance centeredSS
  rw [binary_mean m hm]
  have hi (i : Fin (m+3*m)) : (binaryY m i - 1/2)^2 = (1/4 : ℝ) := by
    unfold binaryY
    split <;> norm_num
  simp_rw [hi]
  simp only [Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]
  simp only [div_mul_eq_div_div]
  ring

theorem sharp_regularity : Regularity sharpFamily (1/4) (1/4) (1/4) 0 0 := by
  have hsize : Tendsto (fun k => (sharpFamily k).size) atTop atTop :=
    tendsto_atTop_mono (fun k => by dsimp [sharpFamily, Population.size]; omega) tendsto_id
  have hbig : Tendsto (fun k => ((sharpFamily k).size : ℝ)-1) atTop atTop := by
    apply tendsto_atTop_mono _ (tendsto_natCast_atTop_atTop (R := ℝ))
    intro k
    dsimp [sharpFamily, Population.size]
    push_cast
    have hk : (0 : ℝ) ≤ k := Nat.cast_nonneg k
    linarith
  have hinv := tendsto_inv_atTop_zero.comp hbig
  have hvar : Tendsto (fun k => sampleVariance Finset.univ (binaryY (k+2)))
      atTop (nhds (1/4 : ℝ)) := by
    have hf : (fun k => sampleVariance Finset.univ (binaryY (k+2))) =
        (fun k => (1/4 : ℝ) + (1/4)*(((sharpFamily k).size : ℝ)-1)⁻¹) := by
      funext k
      rw [binary_variance (k+2) (by omega)]
      dsimp [sharpFamily, Population.size]
      push_cast
      have hk : (0 : ℝ) ≤ k := Nat.cast_nonneg k
      have hd : (k : ℝ)+2+3*(k+2)-1 ≠ 0 := by linarith
      ring_nf at hd ⊢
      field_simp [hd]
      <;> ring
    rw [hf]
    simpa using (tendsto_const_nhds.add (tendsto_const_nhds.mul hinv))
  refine ⟨hsize, Filter.Eventually.of_forall (fun k => by dsimp [sharpFamily]; omega),
    by norm_num, by norm_num, ?_, hvar, hvar, ?_, ?_, ?_⟩
  · have hf : (fun k => ((sharpFamily k).n1 : ℝ)/(sharpFamily k).size) =
        (fun _ : ℕ => (1/4 : ℝ)) := by
      funext k
      dsimp [sharpFamily, Population.size]
      push_cast
      have hk : (0 : ℝ) < k+2 := by positivity
      field_simp
      <;> ring
    rw [hf]
    exact tendsto_const_nhds
  · simpa [populationVariance, sampleVariance, centeredSS, sampleMean, unitEffects, sharpFamily]
      using (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))
  · simpa [averageEffect, sharpFamily] using
      (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))
  · refine ⟨1, by norm_num, ?_⟩
    intro k
    have hpow (i : Fin ((k+2)+3*(k+2))) : binaryY (k+2) i ^ 4 = binaryY (k+2) i := by
      unfold binaryY
      split <;> norm_num
    change (∑ i : Fin ((k+2)+3*(k+2)),
      (binaryY (k+2) i ^ 4 + binaryY (k+2) i ^ 4)) / (((k+2)+3*(k+2) : ℕ) : ℝ) ≤ 1
    simp_rw [hpow]
    rw [Finset.sum_add_distrib, add_div]
    have hm := binary_mean (k+2) (by omega)
    simp only [sampleMean, Finset.card_univ, Fintype.card_fin] at hm
    rw [hm]
    norm_num

lemma centered_nonneg {ι : Type*} [DecidableEq ι] (s : Finset ι) (y : ι → ℝ) :
    0 ≤ centeredSS s y := Finset.sum_nonneg (fun _ _ => sq_nonneg _)

lemma centered_pos {ι : Type*} [DecidableEq ι] (s : Finset ι) (y : ι → ℝ)
    (a b : ι) (ha : a ∈ s) (hb : b ∈ s) (hne : y a ≠ y b) : 0 < centeredSS s y := by
  apply (Finset.sum_pos_iff_of_nonneg (fun _ _ => sq_nonneg _)).mpr
  by_cases h : y a = sampleMean s y
  · exact ⟨b, hb, sq_pos_of_ne_zero (sub_ne_zero.mpr (by intro he; exact hne (h.trans he.symm)))⟩
  · exact ⟨a, ha, sq_pos_of_ne_zero (sub_ne_zero.mpr h)⟩

theorem sharp_positive : PositiveOLS sharpFamily := by
  classical
  apply Filter.Eventually.of_forall
  intro k z hz
  change Finset (Fin ((k+2)+3*(k+2))) at z
  change z ∈ support (k+2) at hz
  let m := k+2
  have hm : 0 < m := by dsimp [m]; omega
  have hzcard : z.card = m := (Finset.mem_powersetCard.mp hz).2
  let low : Finset (Fin (m+3*m)) := Finset.univ.filter (fun i => i.val < 2*m)
  have hlcard : low.card = 2*m := lower_card (2*m) (m+3*m) (by omega)
  have hhcard : lowᶜ.card = 2*m := by
    rw [Finset.card_compl, Fintype.card_fin, hlcard]
    omega
  have hlo : ¬low ⊆ z := by
    intro hs
    have h := Finset.card_le_card hs
    rw [hlcard, hzcard] at h
    omega
  have hhi : ¬lowᶜ ⊆ z := by
    intro hs
    have h := Finset.card_le_card hs
    rw [hhcard, hzcard] at h
    omega
  obtain ⟨a, ha, haz⟩ := Finset.not_subset.mp hlo
  obtain ⟨b, hb, hbz⟩ := Finset.not_subset.mp hhi
  have hay : binaryY m a = 1 := by simp [binaryY, (Finset.mem_filter.mp ha).2]
  have hby : binaryY m b = 0 := by
    have : ¬b.val < 2*m := by simpa [low] using hb
    simp [binaryY, this]
  have hcp : 0 < centeredSS zᶜ (binaryY m) :=
    centered_pos zᶜ (binaryY m) a b (Finset.mem_compl.mpr haz) (Finset.mem_compl.mpr hbz)
      (by rw [hay, hby]; norm_num)
  have hobs : observed (sharpFamily k) z = binaryY m := by
    funext i
    simp [observed, sharpFamily, m]
  unfold olsVariance residualSS
  rw [hobs]
  apply mul_pos
  · apply div_pos
    · exact add_pos_of_nonneg_of_pos (centered_nonneg z (binaryY m)) hcp
    · dsimp [sharpFamily, Population.size]
      push_cast
      have hk : (0 : ℝ) ≤ k := Nat.cast_nonneg k
      linarith
  · dsimp [sharpFamily]
    positivity

example : Regularity sharpFamily (1/4) (1/4) (1/4) 0 0 ∧
    (∀ k, SharpNull (sharpFamily k)) ∧ (∀ k, ConstantEffects (sharpFamily k)) ∧
    Tendsto (fun k => averageEffect (sharpFamily k)) atTop (nhds 0) ∧
    0 < olsLimit (1/4) (1/4) (1/4) ∧ 0 < (1/4 : ℝ) ∧ PositiveOLS sharpFamily := by
  refine ⟨sharp_regularity, ?_, ?_, ?_, by norm_num [olsLimit], by norm_num, sharp_positive⟩
  · simp [SharpNull, sharpFamily]
  · simp [ConstantEffects, unitEffects, averageEffect, sharpFamily]
  · simpa [averageEffect, sharpFamily] using
      (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))

-- The same nondegenerate family supplies every outer and local hypothesis of T7.
example : ∃ E : ℕ → Population, 0 ≤ (1/2 : ℝ) ∧ 0 ≤ (1/2 : ℝ) ∧
    Regularity E (1/4) ((1/2)^2) ((1/2)^2) 0 0 ∧
    Tendsto (fun k => averageEffect (E k)) atTop (nhds 0) := by
  refine ⟨sharpFamily, by norm_num, by norm_num, ?_, ?_⟩
  · norm_num
    exact sharp_regularity
  · simpa [averageEffect, sharpFamily] using
      (tendsto_const_nhds : Tendsto (fun _ : ℕ => (0 : ℝ)) atTop (nhds 0))

-- The universal and constant-effect hypotheses of T8 also hold together.
example : ∃ E : ℕ → Population, Regularity E (1/4) (1/4) (1/4) 0 0 ∧
    (∀ k, ConstantEffects (E k)) := by
  refine ⟨sharpFamily, sharp_regularity, ?_⟩
  simp [ConstantEffects, unitEffects, averageEffect, sharpFamily]

end
