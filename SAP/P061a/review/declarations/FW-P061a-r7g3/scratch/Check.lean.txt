import SAP.P061a.Defs

open scoped BigOperators
open SAP.P061a

namespace T11Audit

-- Infinitely many feasible assignments, of which exactly three have mass.
noncomputable def design : AtomicDesign ℕ where
  mass z := if z ∈ ({0, 1, 2} : Finset ℕ) then (1 / 3 : ℝ) else 0
  mass_nonnegative := by
    intro z
    split_ifs <;> norm_num
  mass_summable := by
    apply summable_of_ne_finset_zero (s := ({0, 1, 2} : Finset ℕ))
    intro z hz
    simp [hz]
  mass_total := by
    rw [tsum_eq_sum (s := ({0, 1, 2} : Finset ℕ))]
    · norm_num
    · intro z hz
      simp [hz]

def exposure : Exposures ℕ 1 := fun _ z => if z = 0 then 0 else 1
def response : Responses ℕ 1 := fun _ z => (z : ℝ)

lemma mass_outside (z : ℕ) (hz : z ∉ ({0, 1, 2} : Finset ℕ)) :
    design.mass z = 0 := if_neg hz

lemma sum_three (f : ℕ → ℝ)
    (hf : ∀ z ∉ ({0, 1, 2} : Finset ℕ), f z = 0) :
    (∑' z, f z) = f 0 + f 1 + f 2 := by
  rw [tsum_eq_sum hf]
  simp [add_assoc]

lemma probability_zero (i : Fin 1) :
    exposureProbability design exposure i 0 = (1 / 3 : ℝ) := by
  unfold exposureProbability eventProbability
  rw [sum_three]
  · norm_num [exposure, design]
  · intro z hz
    rw [mass_outside z hz]
    simp

lemma probability_one (i : Fin 1) :
    exposureProbability design exposure i 1 = (2 / 3 : ℝ) := by
  unfold exposureProbability eventProbability
  rw [sum_three]
  · norm_num [exposure, design]
  · intro z hz
    rw [mass_outside z hz]
    simp

lemma positive_zero : positiveExposure design exposure 0 := by
  refine ⟨1 / 3, by norm_num, ?_⟩
  intro i
  rw [probability_zero]

lemma positive_one : positiveExposure design exposure 1 := by
  refine ⟨2 / 3, by norm_num, ?_⟩
  intro i
  rw [probability_one]

lemma finite_support : {z : ℕ | supported design z}.Finite := by
  apply (Finset.finite_toSet ({0, 1, 2} : Finset ℕ)).subset
  intro z hz
  by_contra hn
  change 0 < design.mass z at hz
  rw [mass_outside z hn] at hz
  exact (lt_irrefl 0) hz

-- This example supplies every explicit and structure-level hypothesis of T11.
example : 0 < (1 : ℕ) ∧ positiveExposure design exposure 1 ∧
    positiveExposure design exposure 0 ∧ {z : ℕ | supported design z}.Finite := by
  exact ⟨by norm_num, positive_one, positive_zero, finite_support⟩

-- Both distinct target cells exist; the exposure-one cell is not stable.
example : supported design 1 ∧ supported design 2 ∧
    exposure 0 1 = exposure 0 2 ∧ response 0 1 ≠ response 0 2 := by
  norm_num [supported, design, exposure, response]

example : ¬ supportStability design exposure response := by
  intro h
  have he := h 0 1 2 (by norm_num [supported, design])
    (by norm_num [supported, design]) (by norm_num [exposure])
  norm_num [response] at he

example : exposureMean design exposure response 0 1 = (3 / 2 : ℝ) := by
  unfold exposureMean mean
  rw [probability_one, sum_three]
  · norm_num [design, exposure, response]
  · intro z hz
    rw [mass_outside z hz]
    simp

example : conditionalExposureMean design exposure response 0 1 = (3 / 2 : ℝ) := by
  unfold conditionalExposureMean
  rw [probability_one, sum_three]
  · norm_num [design, exposure, response]
  · intro z hz
    rw [mass_outside z hz]
    simp

end T11Audit
