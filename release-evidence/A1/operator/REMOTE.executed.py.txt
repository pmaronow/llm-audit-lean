from pathlib import Path

def raw_file(value):
    path = Path(value)
    assert path.is_absolute() and path.resolve(strict=True) == path
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    try:
        before = os.fstat(fd)
        assert stat.S_ISREG(before.st_mode) and before.st_nlink == 1 and before.st_size <= 128 * 1024**2
        chunks, count = [], 0
        while True:
            chunk = os.read(fd, 1024**2)
            if not chunk: break
            count += len(chunk)
            assert count <= 128 * 1024**2
            chunks.append(chunk)
        after = os.fstat(fd)
        stamp = lambda s: (s.st_dev,s.st_ino,s.st_mode,s.st_nlink,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
        assert stamp(before) == stamp(after) and count == before.st_size
        return b''.join(chunks)
    finally:
        os.close(fd)


def h(path): return hashlib.sha256(raw_file(path)).hexdigest()
def status(rc): return rc if rc >= 0 else 128 - rc
